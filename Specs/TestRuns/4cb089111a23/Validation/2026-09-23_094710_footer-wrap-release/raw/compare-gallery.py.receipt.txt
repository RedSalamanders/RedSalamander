"""Read original harness PNGs and metrics; never modify application captures."""
import csv
import hashlib
import json
import math
from collections import defaultdict
from pathlib import Path

from PIL import Image, ImageChops

ROOT = Path('C:/RedSalamander.Perf/evidence/i26-ui')
BASE = ROOT / 'footer-wrap-baseline-20260923'
CAND = ROOT / 'footer-wrap-candidate-20260923'

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def load_metrics(folder):
    path = folder / 'runner-artifacts/perf/perf_metrics.jsonl'
    groups = defaultdict(list)
    rows = []
    for line in path.read_text(encoding='utf-8-sig').splitlines():
        value = json.loads(line)
        name = value.get('metric', '')
        if name.startswith('FileOps.Popup.'):
            # Redraw batches carry exact DPI/width; ordinary render events do not.
            key = name
            if name.endswith('Redraw8Us'):
                key += f"/{value['value0']}dpi/{value['value1']}dip"
            groups[key].append(value['durationUs'])
            if name.endswith('Redraw8Us') or name == 'FileOps.Popup.FooterActions.MeasureUs':
                rows.append(value)
    stats = {}
    for key, values in sorted(groups.items()):
        ordered = sorted(values)
        stats[key] = {'count': len(values), 'medianUs': ordered[math.ceil(len(values) * .5) - 1],
                      'p95Us': ordered[math.ceil(len(values) * .95) - 1], 'sumUs': sum(values)}
    return {'path': str(path), 'sha256': digest(path), 'bytes': path.stat().st_size, 'metrics': stats}, rows

def main():
    pair = {'scope': 'One serial baseline/candidate gallery pair. Not a Full, UIA or resource acceptance gate.',
            'baseline': {}, 'candidate': {}, 'captures': []}
    for label, folder in (('baseline', BASE), ('candidate', CAND)):
        pair[label], rows = load_metrics(folder)
        (CAND / f'{label}-focused-metrics.jsonl').write_text(
            ''.join(json.dumps(row, ensure_ascii=False) + '\n' for row in rows), encoding='utf-8')
    bg = BASE / 'runner-artifacts/fileops/visual-gallery'
    cg = CAND / 'runner-artifacts/fileops/visual-gallery'
    assert {p.name for p in bg.glob('*.png')} == {p.name for p in cg.glob('*.png')}
    for path in sorted(bg.glob('*.png')):
        candidate = cg / path.name
        with Image.open(path) as before, Image.open(candidate) as after:
            same_size = before.size == after.size
            box = ImageChops.difference(before.convert('RGB'), after.convert('RGB')).getbbox() if same_size else None
            pair['captures'].append({'file': path.name, 'baselineSha256': digest(path),
                'candidateSha256': digest(candidate), 'baselineSize': list(before.size),
                'candidateSize': list(after.size), 'sameSize': same_size,
                'samePixels': same_size and box is None, 'differenceBounds': box})
    pair['counts'] = {'captures': len(pair['captures']),
        'sameSize': sum(x['sameSize'] for x in pair['captures']),
        'samePixels': sum(x['samePixels'] for x in pair['captures'])}
    pair['manifestSame'] = (bg / 'scenarios.tsv').read_bytes() == (cg / 'scenarios.tsv').read_bytes()
    pair['displaysSame'] = (bg / 'displays.tsv').read_bytes() == (cg / 'displays.tsv').read_bytes()
    (CAND / 'gallery-comparison.json').write_text(json.dumps(pair, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({k: pair[k] for k in ('counts', 'manifestSame', 'displaysSame')}, indent=2))
    for key in pair['baseline']['metrics']:
        if key.endswith('MeasureUs') or 'Redraw8Us' in key:
            print(key, pair['baseline']['metrics'][key], pair['candidate']['metrics'].get(key))

if __name__ == '__main__':
    main()
