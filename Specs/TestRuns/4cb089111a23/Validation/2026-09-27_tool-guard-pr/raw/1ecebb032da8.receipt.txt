import hashlib,json,re,shutil,subprocess,xml.etree.ElementTree as ET
from datetime import datetime
from pathlib import Path
root=Path(__file__).parent
repo=Path('Z:/src/RedSalamander')
packet=repo/'Specs/TestRuns/4cb089111a23/Validation/2026-09-27_tool-guard-pr'
sha=lambda p:hashlib.sha256(Path('\\\\?\\'+str(p.absolute())).read_bytes()).hexdigest()
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
assert not packet.exists()
selected=[]; outcomes=[]
for version,config in [('v6','release'),('v8','release'),('v8','debug')]:
    d=root/f'current-{version}-pr-{config}-20260927'
    run=d/'pr-runner-artifacts'
    summary=run/'artifacts/selftest/last_run/run-all-tests-results.json'
    s=read(summary)
    assert s['requested_gate']=='PR' and s['counts']['executed']==13 and s['counts']['reused']==0
    expected=1 if version=='v6' else 0
    assert int((d/'pr-exit.txt').read_text().strip())==expected
    assert s['case_counts']['failed']==(2 if version=='v6' else 0) and s['case_counts']['skipped']==0
    assert s['repository_verdict']==('FAILED' if version=='v6' else 'PASSED'),s['repository_verdict']
    hashes=read(d/'pr-runner-copy-hashes.json')
    for item in hashes:
        assert sha(run/item['path'])==item['sha256'],item['path']
    snapshots=list(run.rglob('workspace-initial.json'))
    assert len(snapshots)==1
    after=list(run.rglob('workspace-after-tooling.pester.json'))
    assert len(after)==1
    assert read(snapshots[0])['snapshot_id']==read(after[0])['snapshot_id']
    selected += [d/'pr.log',d/'pr-exit.txt',d/'invocations.json',d/'candidate-inputs.json',
                 d/'pr-plan.json',d/'pr-runner-copy-hashes.json',d/'run.ps1',summary,snapshots[0],after[0]]
    if version=='v8':
        assert int((d/'build-exit.txt').read_text().strip())==0
        build=read(d/'build-receipt.json')
        assert build['receipt_id']==s['build_receipt_id']
        assert build['configuration']==config.title() and build['platform']=='x64'
        assert build['exit_code']==0 and build['target']=='solution' and build['tests_enabled']
        assert build['git_head']=='ccebd5af68eb45f2e9deed64a7063780e71eb1ce'
        before=read(snapshots[0])
        assert build['source_snapshot_id']==s['snapshot_id']==before['snapshot_id']
        frozen=read(d/'candidate-inputs.json')
        # Windows resource.h has retained lower-case driver spelling and canonical
        # Resource.h Git spelling. Require unique case-insensitive paths and exact bytes.
        bypath={r['path'].casefold():r['worktree_sha256'] for r in before['files']}
        assert len(bypath)==len(before['files'])==len({r['path'].casefold() for r in frozen})
        assert len(frozen)==len(bypath)==40
        for row in frozen:
            assert bypath[row['path'].casefold()]==row['sha256'],row['path']
        invocation=read(d/'invocations.json')
        assert datetime.fromisoformat(build['ended_utc'])<=datetime.fromisoformat(invocation['startedUtc'])
        selected += [d/'build.log',d/'build-exit.txt',d/'build-receipt.json']
        selected += [p for p in (d/'source').rglob('*') if p.is_file()]
    outcomes.append(dict(version=version,configuration=config,run_id=s['run_id'],
        build_receipt_id=s['build_receipt_id'],case_counts=s['case_counts'],counts=s['counts'],
        repository_verdict=s['repository_verdict'],copied_files_verified=len(hashes)))
v7=root/'current-v7-pr-release-20260927'
assert int((v7/'pr-exit.txt').read_text().strip())==4
selected += [v7/p for p in ['pr.log','pr-exit.txt','invocations.json','setup-recovery.json','candidate-inputs-setup-v1.json']]
focused=root/'tool-guards-v7-20260927'
xml=ET.parse(focused/'focused-pester.xml').getroot()
assert xml.attrib['total']=='226' and xml.attrib['failures']=='0' and xml.attrib['not-run']=='0'
assert read(focused/'validator-exits.json')==dict(toolInventory=0,specInventory=0)
selected += [p for p in focused.iterdir() if p.is_file()]
selected += [Path(__file__),root/'prepare-tool-guards-v7-20260927.py',root/'complete-tool-guards-v7-setup-20260927.py',root/'prepare-tool-guards-v8-20260927.py']
packet.mkdir(parents=True)
mapping=[]
for source in sorted(set(selected)):
    relative=source.relative_to(root).as_posix()
    # Compact deterministic archive IDs avoid reproducing native runner's nested path lengths.
    name=hashlib.sha256(relative.encode()).hexdigest()[:12]+'.receipt.txt'
    target=packet/'raw'/name
    target.parent.mkdir(exist_ok=True)
    assert not target.exists()
    shutil.copyfile(source,target)
    mapping.append(dict(source=str(source),path=target.relative_to(packet).as_posix(),sha256=sha(target)))
(packet/'raw/file-map.json.receipt.txt').write_text(json.dumps(mapping,indent=2)+'\n',encoding='utf-8')
(packet/'raw/review.json.receipt.txt').write_text(json.dumps(outcomes,indent=2)+'\n',encoding='utf-8')
(packet/'.gitattributes').write_bytes(b'raw/** -text -whitespace\nSHA256SUMS.txt -text -whitespace\n')
counts='; '.join(f"{o['configuration']} {o['case_counts']['passed']}/0/0" for o in outcomes if o['version']=='v8')
(packet/'README.md').write_text(f'''# Current File Operations consumer: repaired tooling guards and fresh PR qualification

The V6 Release PR run executes all thirteen entries but reports 822 passes and two Pester failures. One source contract still expected accepted-decision validation inline in IssuesPane after the shared `QualifyIssueFacts` helper took ownership. The phase inventory also counted the exact-only common presentation benchmark as a broad behavioral phase. The repair verifies the pane consumes qualified facts, checks the helper's task/generation/action guards, and keeps the explicit benchmark discoverable without inflating broad behavioral totals. Production code and the qualified DxUi pin are unchanged.

The focused tooling, source-contract and governance tests pass 226/0/0; tool and spec inventories have zero blocking findings. After fresh full-solution builds, the current consumer PR gates pass {counts}. Both runs execute all thirteen nonactivating entries with no reused result. `raw/review.json.receipt.txt` records exact run/build identities and counts. Root rehashes every copied runner file before curation. Full's foreground entries and the integrated I26 acceptance matrix remain separate required gates.

`SOURCE_INPUT_CHANGED` in V6 is the validation planner's affected-input classification, not a file mutation during Pester. Before/after snapshot identities agree. The failed original run is retained. V7's reused build attestation is rejected before native execution because tooling inputs changed (exit 4); V8 performs a new build rather than bypassing that guard. Two external setup-script errors before V7 native execution are also retained: Windows path separator substitution and a duplicate old manifest entry. The corrected manifest verifies forty unique inputs.

`raw/file-map.json.receipt.txt` maps compact archive names to original paths and SHA256 hashes. Full runner trees remain in the external I26 evidence roots, with complete copied-file hash inventories retained here. Every copied payload is byte-exact; archived source and scripts have `.receipt.txt` suffixes and are evidence, not build inputs. `SHA256SUMS.txt` binds this packet. No new UI pixel, native mixed-DPI, assistive-technology or paired performance claim follows from these tooling repairs.
''',encoding='utf-8')
files=sorted(p for p in packet.rglob('*') if p.is_file())
(packet/'SHA256SUMS.txt').write_bytes(''.join(sha(p)+'  '+p.relative_to(packet).as_posix()+'\n' for p in files).encode('utf-8'))
print(json.dumps(dict(payloads=len(files),outcomes=outcomes),indent=2))
