import hashlib,json,shutil
from pathlib import Path
root=Path(__file__).parent
repo=Path('C:/RsI26Final')
release=root/'current-v7-pr-release-20260927'
rows=json.loads((release/'candidate-inputs.json').read_text(encoding='utf-8-sig'))
unique={r['path']:r for r in rows}
for row in unique.values():
    assert hashlib.sha256((repo/row['path']).read_bytes()).hexdigest()==row['sha256'],row['path']
shutil.copyfile(release/'candidate-inputs.json',release/'candidate-inputs-setup-v1.json')
for config in ('release','debug'):
    old=root/f'current-v6-pr-{config}-20260927'
    new=root/f'current-v7-pr-{config}-20260927'
    new.mkdir(exist_ok=True)
    (new/'candidate-inputs.json').write_text(json.dumps(list(unique.values()),indent=2)+'\n',encoding='utf-8')
    driver=(old/'run.ps1').read_text(encoding='utf-8-sig').replace(old.as_posix(),new.as_posix())
    assert new.as_posix() in driver
    (new/'run.ps1').write_text(driver,encoding='utf-8')
    for relative in ['Tools/Modules/Testing/TestInventory.psm1','Tools/Tests/TestInventory.Tests.ps1',
                     'Tools/Tests/TestHarnessSourceContracts.Tests.ps1','Tools/tool-inventory.json',
                     'Specs/Testing/Testing_TestCoverage.md']:
        snap=new/'source'/relative
        snap.parent.mkdir(parents=True,exist_ok=True)
        shutil.copyfile(repo/relative,snap)
(release/'setup-recovery.json').write_text(json.dumps(dict(nativeExecutedBeforeRecovery=False,
    failures=['Driver template replacement compared Windows backslashes with forward-slash literals.',
              'Appending changed inputs duplicated an existing tool-inventory entry with its old hash.'],
    correction='Use normalized slash paths and replace manifest entries by source path; verify every final hash.',
    finalInputCount=len(unique)),indent=2)+'\n',encoding='utf-8')
(root/'current-v7-pr-sequence-20260927.ps1').write_text('''$ErrorActionPreference = 'Stop'
& "$PSScriptRoot/current-v7-pr-release-20260927/run.ps1"
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
& "$PSScriptRoot/current-v7-pr-debug-20260927/run.ps1"
exit $LASTEXITCODE
''',encoding='utf-8')
print(f'{len(unique)} verified inputs. Setup failures retained; no native test previously ran.')
