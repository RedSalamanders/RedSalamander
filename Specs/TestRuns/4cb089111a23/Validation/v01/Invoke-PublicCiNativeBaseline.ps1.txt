param([switch]$SingleCpu)
$ErrorActionPreference='Stop'
Set-StrictMode -Version Latest
$repo='E:\RedSalamander.Perf\integration-20261024'
$expected='582d644be5e6147acb0b8a8aae05511b75f64d17'
Set-Location -LiteralPath $repo
if((git rev-parse HEAD).Trim() -ne $expected -or (git status --porcelain)){throw 'Preserve the qualified native baseline.'}
Import-Module ./Tools/Modules/Build/ArtifactOperationLock.psm1 -Force
Import-Module ./Tools/Modules/Build/BuildEvidence.psm1 -Force
Import-Module ./Tools/Modules/Build/ProcessStreaming.psm1 -Force
Import-Module ./Tools/Modules/Testing/TestSandbox.psm1 -Force
$scope=@{kind='test';target='public CI focused native baseline';suite='PR';platform='x64';configuration='Debug'}
$lease=Enter-RSArtifactOperationLock -RepoRoot $repo -Operation 'Public CI focused native baseline' -Scope $scope
try {
    if($lease.WasAbandoned -or (Test-RSArtifactOperationContaminated -RepoRoot $repo -Scope $scope)){throw 'Contaminated baseline profile.'}
    Assert-RSNoResidualArtifactToolProcesses -RepoRoot $repo -Scope $scope
    $snapshot=Get-RSBuildSourceSnapshot -RepoRoot $repo
    $output=Join-Path $repo '.build\x64\Debug'
    $receipt=Read-RSBuildReceipt -BuildOutputDir $output -SchemaPath (Join-Path $repo 'Specs\Build\BuildReceipt.schema.json')
    $usable=Test-RSBuildReceiptForArtifactUse -Receipt $receipt -SourceSnapshot $snapshot -Platform x64 -Configuration Debug -RepoRoot $repo -RequiredRelativePath @('.build/x64/Debug/ViewerPETests.exe','.build/x64/Debug/PluginContractTests.exe')
    if(-not $usable){throw 'Baseline artifact identity differs.'}
    $records=[Collections.Generic.List[object]]::new()
    $cases=@(
        @{exe='ViewerPETests.exe';name='web-profile';args=@('--group=noninteractive','TestViewerWebVirtualHtmlUsesPrivateOriginAndEnforcesByteCaps')},
        @{exe='ViewerPETests.exe';name='lifecycle-captures';args=@('--group=noninteractive','TestViewerSpaceLifecycleCaptures')},
        @{exe='ViewerPETests.exe';name='directory-summary';args=@('--group=noninteractive','TestViewerSpaceDirectorySummaryProtocol')},
        @{exe='PluginContractTests.exe';name='s3-directory-transfer';args=@('--s3-directory-transfer-selftests')}
    )
    if($SingleCpu){$cases=@($cases|Where-Object name -EQ 'directory-summary')}
    $recordName=if($SingleCpu){'public-ci-native-baseline-singlecpu.json'}else{'public-ci-native-baseline.json'}
    if(Test-Path -LiteralPath (Join-Path $PSScriptRoot $recordName)){throw 'Preserve the existing native diagnostic receipt.'}
    foreach($case in $cases){
        $context=New-RSTestRunContext -RepoRoot $repo -TestRootOverride 'E:\RedSalamander.Perf'
        $temp=New-RSTestSandboxScratchDirectory -RepoRoot $repo -Harness public-ci-baseline -Case $case.name -RunId $context.RunId
        [void][IO.Directory]::CreateDirectory($context.ArtifactRoot)
        $log=Join-Path $context.ArtifactRoot ($case.name+'.log')
        $errorLog=Join-Path $context.ArtifactRoot ($case.name+'.stderr.log')
        $started=[DateTime]::UtcNow
        $file=Join-Path $output $case.exe
        $arguments=$case.args
        if($SingleCpu){$file=(Get-Command pwsh).Source;$arguments=@('-NoProfile','-File',(Join-Path $PSScriptRoot 'Invoke-OwnSingleCpuViewer.ps1'),'-Executable',(Join-Path $output $case.exe),'-Case','TestViewerSpaceDirectorySummaryProtocol')}
        $code=Invoke-RSStreamingProcess -FilePath $file -WorkingDirectory $output -Arguments $arguments -LogPath $log -ErrorLogPath $errorLog -AdditionalEnvironment @{REDSALAMANDER_TEST_ROOT=$context.TestRoot;REDSALAMANDER_TEST_RUN_ID=$context.RunId;TEMP=$temp;TMP=$temp} -TimeoutMilliseconds 600000
        $records.Add([ordered]@{case=$case.name;head=$expected;sourceSnapshot=$snapshot.SnapshotId;receipt=$receipt.receipt_id;exitCode=$code;startedUtc=$started.ToString('o');finishedUtc=[DateTime]::UtcNow.ToString('o');runId=$context.RunId;log=$log;errorLog=$errorLog})
        $records|ConvertTo-Json -Depth 6|Set-Content (Join-Path $PSScriptRoot $recordName)
        if((git rev-parse HEAD).Trim() -ne $expected -or (git status --porcelain)){throw 'Baseline source changed during testing.'}
    }
    if(@($records|Where-Object exitCode -NE 0).Count){exit 1}
} finally {Exit-RSArtifactOperationLock -Lock $lease}
