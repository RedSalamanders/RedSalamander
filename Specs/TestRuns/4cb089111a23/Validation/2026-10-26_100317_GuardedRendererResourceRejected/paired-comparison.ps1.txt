param([Parameter(Mandatory)][string]$ExpectedHead,[ValidatePattern('^[a-z0-9-]+$')][string]$CandidatePrefix='seed-memory-candidate',[ValidatePattern('^[a-z0-9-]+$')][string]$OutputLabel='pool')
$ErrorActionPreference='Stop'
Set-StrictMode -Version Latest
$baselineHead='fac05d5b70b9a5933b12093c1b2337f76aee8cdc'
function Resolve-EvidencePath([string]$Path) {
    if(Test-Path -LiteralPath $Path){return $Path}
    $prefix='E:\RedSalamander.Perf\runs\'
    if($Path.StartsWith($prefix,[StringComparison]::OrdinalIgnoreCase)){
        $archived='E:\RedSalamander.Perf\evidence\completed-runs\'+$Path.Substring($prefix.Length)
        if(Test-Path -LiteralPath $archived){return $archived}
    }
    throw "Retained evidence is missing: $Path"
}
function Read-SeedRecord([string]$Path,[string]$Head,[string]$Profile,[int]$Attempt,[string]$Cohort) {
    $r=Get-Content -LiteralPath $Path -Raw | ConvertFrom-Json
    if($r.source -ne $Head -or $r.configuration -ne $Profile -or $r.outcome -ne 'focused_pass_only' -or $r.exactCaseCounts -ne $true -or $r.nativePassed -ne 1 -or $r.nativeFailed -ne 0 -or $r.nativeSkipped -ne 0){throw 'Invalid exact-source native seed record.'}
    $nativePath=Resolve-EvidencePath $r.nativeResults
    if((Get-FileHash -LiteralPath $nativePath).Hash -ne $r.nativeResultsSha256){throw 'Native seed verdict digest differs.'}
    $native=Get-Content -LiteralPath $nativePath -Raw | ConvertFrom-Json
    if(@($native.cases).Count -ne 1 -or $native.cases[0].name -ne 'local_index_core_sqlite_cold_start_stale_root_refreshes_before_query' -or $native.cases[0].status -ne 'passed'){throw 'Wrong native seed scenario.'}
    $metrics=@()
    $metricSources=@()
    foreach($original in $r.metricsPaths){
        $metricPath=Resolve-EvidencePath $original
        $metricSources+=@{path=$metricPath;sha256=(Get-FileHash -LiteralPath $metricPath).Hash}
        foreach($line in [IO.File]::ReadLines($metricPath)){
            if($line.Contains('"metric":"search.ntfs.seed_mft_stages_us"')){$metrics+=($line | ConvertFrom-Json)}
        }
    }
    if($metrics.Count -ne 1 -or $metrics[0].hr -ne 0 -or $metrics[0].detail -notmatch '^enumeration=(\d+);projection=(\d+);retirement=(\d+);complete=true$'){throw 'One complete native MFT seed is required.'}
    $parts=@{enumerationUs=[long]$Matches[1];projectionUs=[long]$Matches[2];retirementUs=[long]$Matches[3]}
    $memoryPath=Resolve-EvidencePath $r.memorySamplesPath
    if((Get-FileHash -LiteralPath $memoryPath).Hash -ne $r.memorySamplesSha256){throw 'Owned memory sample digest differs.'}
    $samples=@(Get-Content -LiteralPath $memoryPath | ForEach-Object {$_ | ConvertFrom-Json})
    if($samples.Count -ne $r.memorySampleCount -or $samples.Count -lt 2){throw 'Incomplete memory sampling.'}
    $pids=@($samples.identity.pid | Sort-Object -Unique)
    if($pids.Count -ne 1 -or $pids[0] -ne $metrics[0].process){throw 'Samples do not belong to the native seed process.'}
    $peak=($samples | Measure-Object privateBytes -Maximum).Maximum
    if($peak -ne $r.sampledPeakPrivateBytes){throw 'Sampled peak does not recompute.'}
    [pscustomobject]@{cohort=$Cohort;head=$Head;profile=$Profile;attempt=$Attempt;record=$Path;recordSha256=(Get-FileHash -LiteralPath $Path).Hash;nativeResultsSha256=$r.nativeResultsSha256;durationUs=[long]$metrics[0].durationUs;enumerationUs=$parts.enumerationUs;projectionUs=$parts.projectionUs;retirementUs=$parts.retirementUs;mftEntries=[long]$metrics[0].value0;mftParents=[long]$metrics[0].value1;sampledPeakPrivateBytes=[long]$peak;sampleCount=$samples.Count;samplerSourceSha256=$r.memorySamplerSourceSha256;memorySamplesSha256=$r.memorySamplesSha256;metricSources=$metricSources}
}
function Median($Values){$sorted=@($Values | Sort-Object);return [double]$sorted[[int][Math]::Floor($sorted.Count/2)]}
$rows=@()
foreach($profile in @('ASan Debug','Release')){
    $suffix=if($profile -eq 'ASan Debug'){'asan'}else{'release'}
    foreach($attempt in 1..3){
        $rows+=Read-SeedRecord (Join-Path $PSScriptRoot "owned-residual-seed-memory-baseline-$suffix-$attempt.json") $baselineHead $profile $attempt baseline
        $rows+=Read-SeedRecord (Join-Path $PSScriptRoot "owned-residual-${CandidatePrefix}-$suffix-$attempt.json") $ExpectedHead $profile $attempt candidate
    }
}
if(@($rows.samplerSourceSha256 | Sort-Object -Unique).Count -ne 1){throw 'The sampling implementation changed between cohorts.'}
$comparisons=foreach($profile in @('ASan Debug','Release')){
    $baseline=@($rows | Where-Object {$_.profile -eq $profile -and $_.cohort -eq 'baseline'})
    $candidate=@($rows | Where-Object {$_.profile -eq $profile -and $_.cohort -eq 'candidate'})
    $baselineTime=Median $baseline.durationUs
    $candidateTime=Median $candidate.durationUs
    $baselineMemory=Median $baseline.sampledPeakPrivateBytes
    $candidateMemory=Median $candidate.sampledPeakPrivateBytes
    $timeRatio=$candidateTime/$baselineTime
    $memoryRatio=$candidateMemory/$baselineMemory
    $passed=$timeRatio -le 1.10 -and $memoryRatio -le 1.10
    if($profile -eq 'ASan Debug'){$passed=$passed -and $timeRatio -lt 1.0}
    [pscustomobject]@{profile=$profile;samplesPerCohort=3;baselineMedianUs=$baselineTime;candidateMedianUs=$candidateTime;durationRatio=$timeRatio;baselineMedianSampledPeakPrivateBytes=$baselineMemory;candidateMedianSampledPeakPrivateBytes=$candidateMemory;sampledPeakPrivateRatio=$memoryRatio;passed=$passed}
}
$result=[ordered]@{source=$ExpectedHead;baselineSource=$baselineHead;scope='Focused paired native seed wall-time and sampled memory; not Full qualification. Actual MFT record counts are retained per attempt and the volume is not a frozen image.';samplingIntervalMs=250;passed=@($comparisons | Where-Object {-not $_.passed}).Count -eq 0;comparisons=@($comparisons);records=$rows}
$result | ConvertTo-Json -Depth 8 | Set-Content (Join-Path $PSScriptRoot ('ntfs-seed-'+$OutputLabel+'-comparison.json'))
$comparisons | ConvertTo-Json -Depth 4
if(-not $result.passed){exit 1}
