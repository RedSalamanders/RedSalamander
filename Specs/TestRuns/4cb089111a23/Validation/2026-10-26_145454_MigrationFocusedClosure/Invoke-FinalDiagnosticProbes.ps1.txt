param(
    [Parameter(Mandatory)][string]$ExpectedHead,
    [string]$QualificationStateFile = 'quiescence-qualification.json',
    [string]$RuntimeScopeWaiverFile=''
)
$ErrorActionPreference = 'Stop'
Set-StrictMode -Version Latest
$repoPath = 'E:\RedSalamander.Perf\integration-20261024'
$evidenceDir = $PSScriptRoot
$state = Get-Content (Join-Path $evidenceDir $QualificationStateFile) -Raw | ConvertFrom-Json
if ($state.head -ne $ExpectedHead -or $state.outcome -ne 'passed' -or $state.phase -ne 'complete') {
    throw 'The declared exact-source qualification scope must pass before diagnostic probes.'
}
$qualificationOwner=Get-Process -Id $state.ownerPid -ErrorAction SilentlyContinue
if ($qualificationOwner -and (!$state.PSObject.Properties['ownerStartUtc'] -or $qualificationOwner.StartTime.ToUniversalTime() -eq ([datetime]$state.ownerStartUtc).ToUniversalTime())) {
    throw 'The serial qualification owner is still alive; do not overlap it.'
}
Set-Location -LiteralPath $repoPath
function Assert-FrozenSource {
    if ((git rev-parse HEAD) -ne $ExpectedHead -or (git status --porcelain)) { throw 'Candidate source is no longer frozen.' }
}
Assert-FrozenSource
$fullPhases = @($state.entries | Where-Object phase -Like '*-serial-x64-*-Full')
if ($fullPhases.Count -ne 3 -or @($fullPhases | Where-Object exitCode -NE 0).Count) {
    if(!$RuntimeScopeWaiverFile -or !$state.PSObject.Properties['fullRuntimeSkippedByUser'] -or !$state.fullRuntimeSkippedByUser){throw 'Missing passing Full profile or explicit user scope waiver.'}
    $waiverPath=Join-Path $evidenceDir $RuntimeScopeWaiverFile
    $waiver=Get-Content $waiverPath -Raw|ConvertFrom-Json
    if($waiver.head -ne $ExpectedHead -or !$waiver.skipFullRuntimeSuites -or $waiver.fullRuntimePassClaim -or
       (Get-FileHash $waiverPath).Hash -ne $state.scopeWaiverSha256){throw 'User runtime scope waiver differs.'}
}
$observer = 'Z:\src\RedMonitor\.build\x64\Release\TransportRoundtripTests.exe'
$observerReceiptPath = 'Z:\src\RedMonitor\.build\x64\Release\build-receipt.json'
$observerReceipt = Get-Content $observerReceiptPath -Raw | ConvertFrom-Json
$observerHash = (Get-FileHash -LiteralPath $observer -Algorithm SHA256).Hash
if ($observerHash -ne '585C7E7CFCC6073F90CFFD00F5FE422926066DD5E0C59C96962235D62B645981' -or
    $observerReceipt.sourceIdentity -ne 'd97c8b11c48d76a0fa9f1a93fd3c9362963dbf0a42cb6cd9adaea5748d051716') {
    throw 'Reviewed RedMonitor observer identity changed.'
}
$observerArtifact = @($observerReceipt.artifacts | Where-Object path -EQ 'TransportRoundtripTests.exe')
if ($observerArtifact.Count -ne 1 -or $observerArtifact[0].sha256 -ne $observerHash) { throw 'Observer receipt mismatch.' }
$outputDir = Join-Path $evidenceDir 'final-diagnostic-probes'
New-Item -ItemType Directory -Path $outputDir -Force | Out-Null
$records = [System.Collections.Generic.List[object]]::new()
Import-Module ./Tools/Modules/Build/ArtifactOperationLock.psm1 -Force
function Enter-DiagnosticArtifactLock([string]$profile) {
    $scope=@{kind='test';target='final diagnostic probe';suite='Full';configuration=$profile;platform='x64'}
    $lock=Enter-RSArtifactOperationLock -RepoRoot $repoPath -Operation 'Final diagnostic probe' -Scope $scope
    if($lock.WasAbandoned -or (Test-RSArtifactOperationContaminated -RepoRoot $repoPath -Scope $scope)) {
        Exit-RSArtifactOperationLock -Lock $lock
        throw 'Diagnostic build profile is contaminated.'
    }
    return $lock
}
function Get-ArtifactIdentity([string]$profile, [string]$name) {
    $profileDir = Join-Path $repoPath ('.build\x64\' + $profile)
    $receiptPath = Join-Path $profileDir 'build-receipt.json'
    $receipt = Get-Content $receiptPath -Raw | ConvertFrom-Json
    if ($receipt.git_head -ne $ExpectedHead -or $receipt.exit_code -ne 0 -or -not $receipt.tests_enabled) { throw "Invalid $profile receipt." }
    $relative = '.build/x64/' + $profile + '/' + $name
    $entry = @($receipt.outputs | Where-Object relative_path -EQ $relative)
    $path = Join-Path $profileDir $name
    $hash = (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash
    if ($entry.Count -ne 1 -or $hash -ne $entry[0].sha256) { throw "Artifact mismatch: $profile / $name" }
    [pscustomobject]@{ path=$path; sha256=$hash; receipt=$receipt.receipt_id; sourceSnapshot=$receipt.source_snapshot_id }
}
$oldNativePreference = $PSNativeCommandUseErrorActionPreference
$oldEtw = $env:REDSALAMANDER_DIAGNOSTICS_ETW
$oldAsanOptions = $env:ASAN_OPTIONS
$oldAsanDump = $env:ASAN_SAVE_DUMPS
try {
    $PSNativeCommandUseErrorActionPreference = $false
    # Repeat the unchanged owned gate on the baseline's exact C: fixture root.
    # The original service checks also run on E:, but cross-volume observations
    # alone must not be described as a paired latency comparison.
    Import-Module ./Tools/Modules/Testing/TestInvocation.psm1 -Force
    $buildEnv = Get-RSBuildEnvironmentOverrides -Suite Full
    foreach ($key in $buildEnv.Keys) { [Environment]::SetEnvironmentVariable($key,$buildEnv[$key],'Process') }
    Assert-FrozenSource
    $producer = Get-ArtifactIdentity 'ASan Debug' 'RedSalamander.exe'
    $log = Join-Path $outputDir 'paired-busy-volume-gates.log'
    $started = [DateTime]::UtcNow.ToString('o')
    & pwsh -NoProfile -File (Join-Path $evidenceDir 'Invoke-OwnedResidualProbe.ps1') -ExpectedHead $ExpectedHead -Label ('final-croot-warmup-'+$ExpectedHead.Substring(0,9)) -Suite Compare -Configuration 'ASan Debug' -TestRoot C:\RedSalamander.Perf -Repeat 3 -CaseFilter 'local_search_index_status_snapshot_does_not_wait_for_warmup,local_search_index_nowait_query_does_not_wait_for_warmup,local_search_index_opportunistic_query_does_not_wait_for_warmup' *> $log
    $code = $LASTEXITCODE
    $records.Add([pscustomobject]@{ kind='paired-owned-warmup-gates'; profile='ASan Debug'; exitCode=$code; producer=$producer; baselineRun='20261025T205025Z-30012-6cfdfa2480ec481a8a2c0af51d802617'; testRoot='C:\RedSalamander.Perf'; startedUtc=$started; finishedUtc=[DateTime]::UtcNow.ToString('o'); log=$log })
    $records | ConvertTo-Json -Depth 8 | Set-Content (Join-Path $outputDir 'probe-records.json')
    if ($code -ne 0) { throw "Paired warmup gates failed ($code); see $log" }
    Assert-FrozenSource
    foreach ($profile in @('Debug', 'Release', 'ASan Debug')) {
        Assert-FrozenSource
        $diagnosticLock=Enter-DiagnosticArtifactLock $profile
        try {
        $producer = Get-ArtifactIdentity $profile 'RedSalamander.exe'
        $common = Get-ArtifactIdentity $profile 'Common.dll'
        $env:REDSALAMANDER_DIAGNOSTICS_ETW = '1'
        $log = Join-Path $outputDir ('etw-' + $profile.Replace(' ', '-') + '.log')
        $started = [DateTime]::UtcNow.ToString('o')
        $probeOutput = @(& $observer $producer.path RedSalamander 2>&1)
        $code = $LASTEXITCODE
        $probeOutput | Set-Content -LiteralPath $log
        $records.Add([pscustomobject]@{ kind='real-etw-roundtrip'; profile=$profile; exitCode=$code; producer=$producer; common=$common; observerSha256=$observerHash; observerSourceIdentity=$observerReceipt.sourceIdentity; startedUtc=$started; finishedUtc=[DateTime]::UtcNow.ToString('o'); log=$log })
        $records | ConvertTo-Json -Depth 8 | Set-Content (Join-Path $outputDir 'probe-records.json')
        if ($code -ne 0 -or ($probeOutput -join "`n") -notmatch 'PASS: real ETW roundtrip with exact Unicode, severity, PID, TID and timestamp') { throw "Real ETW roundtrip failed for $profile." }
        } finally { Exit-RSArtifactOperationLock -Lock $diagnosticLock }
    }
    Assert-FrozenSource
    $diagnosticLock=Enter-DiagnosticArtifactLock 'ASan Debug'
    try {
    $canary = Get-ArtifactIdentity 'ASan Debug' 'PluginContractTests.exe'
    $env:ASAN_OPTIONS = if ($oldAsanOptions) { $oldAsanOptions + ':log_path=stderr' } else { 'log_path=stderr' }
    $env:ASAN_SAVE_DUMPS = ''
    $canaryLog = Join-Path $outputDir 'asan-canary.log'
    $started = [DateTime]::UtcNow.ToString('o')
    $canaryOutput = @(& $canary.path --asan-seed-heap-overflow 2>&1)
    $code = $LASTEXITCODE
    $canaryOutput | Set-Content -LiteralPath $canaryLog
    $text = $canaryOutput -join "`n"
    $detected = $code -ne 0 -and $text -match 'AddressSanitizer' -and $text -match 'heap-buffer-overflow'
    $records.Add([pscustomobject]@{ kind='asan-seeded-heap-overflow'; profile='ASan Debug'; exitCode=$code; expectedFailureDetected=$detected; artifact=$canary; startedUtc=$started; finishedUtc=[DateTime]::UtcNow.ToString('o'); log=$canaryLog })
    $records | ConvertTo-Json -Depth 8 | Set-Content (Join-Path $outputDir 'probe-records.json')
    if (-not $detected) { throw 'Seeded overflow did not produce the required AddressSanitizer heap-buffer-overflow diagnostic.' }
    } finally { Exit-RSArtifactOperationLock -Lock $diagnosticLock }
    Assert-FrozenSource
    [ordered]@{ head=$ExpectedHead; passed=$true; observerSha256=$observerHash; records=@($records.ToArray()); scope='Paired C-root owned warmup gates, three real ETW producer profiles and seeded x64 ASan rejection; installed helper handoff remains separate.' } |
        ConvertTo-Json -Depth 9 | Set-Content (Join-Path $outputDir 'result.json')
}
finally {
    $PSNativeCommandUseErrorActionPreference = $oldNativePreference
    $env:REDSALAMANDER_DIAGNOSTICS_ETW = $oldEtw
    $env:ASAN_OPTIONS = $oldAsanOptions
    $env:ASAN_SAVE_DUMPS = $oldAsanDump
}
