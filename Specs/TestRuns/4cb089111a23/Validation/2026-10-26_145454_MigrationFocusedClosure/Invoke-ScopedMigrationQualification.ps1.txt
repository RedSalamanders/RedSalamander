param([Parameter(Mandatory)][string]$ExpectedHead)
$ErrorActionPreference='Stop'
Set-StrictMode -Version Latest
$repoPath='E:\RedSalamander.Perf\integration-20261024'
$earlierHead='c4c7399b0ccf700d5f761be5245f6a1e3c513fea'
$short=$ExpectedHead.Substring(0,9)
$label='scoped-'+$short
$statePath=Join-Path $PSScriptRoot ($label+'-qualification.json')
if(Test-Path -LiteralPath $statePath){throw 'Preserve the earlier scoped qualification.'}
Set-Location -LiteralPath $repoPath
$records=[Collections.Generic.List[object]]::new()
$phase='waiting-for-builds'
$outcome='running'
$originalEightPassed=$false
$scopeWaiverPath=Join-Path $PSScriptRoot ('runtime-scope-waiver-'+$short+'.json')
$ownerStarted=(Get-Process -Id $PID).StartTime.ToUniversalTime().ToString('o')
function Assert-Frozen {if((git rev-parse HEAD) -ne $ExpectedHead -or (git status --porcelain)){throw 'Scoped qualification source changed.'}}
function Save-State {
    [ordered]@{head=$ExpectedHead;ownerPid=$PID;ownerStartUtc=$ownerStarted;phase=$phase;outcome=$outcome;fullRuntimeSkippedByUser=$true;fullRuntimePassClaim=$false;scopeWaiverFile=$scopeWaiverPath;scopeWaiverSha256=if(Test-Path $scopeWaiverPath){(Get-FileHash $scopeWaiverPath).Hash}else{$null};earlierComponentQualification=$earlierHead;originalEightFocusedPassed=$originalEightPassed;entries=$records.ToArray()}|ConvertTo-Json -Depth 7|Set-Content $statePath
}
function Invoke-Phase([string]$Name,[scriptblock]$Action){
    Assert-Frozen;$script:phase=$Name;Save-State
    $started=[DateTime]::UtcNow
    $log=Join-Path $PSScriptRoot ($label+'-'+$Name+'.log')
    $global:LASTEXITCODE=0
    & $Action *> $log
    $code=$LASTEXITCODE
    $records.Add([ordered]@{phase=$Name;exitCode=$code;startedUtc=$started.ToString('o');finishedUtc=[DateTime]::UtcNow.ToString('o');log=$log})
    Save-State;Assert-Frozen
    if($code -ne 0){throw "Scoped phase failed: $Name ($code); see $log"}
}
try {
    Assert-Frozen
    $earlierScope=Get-Content (Join-Path $PSScriptRoot runtime-scope-waiver-c4.json) -Raw|ConvertFrom-Json
    $priorResource=Get-Content (Join-Path $PSScriptRoot ntfs-seed-density-guarded-renderer-comparison.json) -Raw|ConvertFrom-Json
    $priorExports=Get-Content (Join-Path $PSScriptRoot final-migration-c4c7399b0-paired-comparison.json) -Raw|ConvertFrom-Json
    if(!$earlierScope.skipFullRuntimeSuites -or !$priorResource.passed -or !$priorExports.passed -or $priorResource.source -ne $earlierHead -or $priorExports.candidate -ne $earlierHead){throw 'Earlier green qualification or user scope instruction differs.'}
    $delta=@(git diff --name-only $earlierHead $ExpectedHead)
    if(@($delta|Where-Object {$_ -notmatch '^(Specs/|Changes/|CHANGELOG\.md$|\.gitattributes$|RedSalamander/SelfTest/Commands/Commands.Tests.FileOps\.cpp$|RedSalamander/FolderWindow.FileOperations.Popup\.(cpp|h)$)'}).Count){throw 'Changes exceed the focused file-operation binding repair; re-evaluate the scope.'}
    [ordered]@{head=$ExpectedHead;userInstruction=$earlierScope.userInstruction;recordedUtc=[DateTime]::UtcNow.ToString('o');skipFullRuntimeSuites=$true;fullRuntimePassClaim=$false;priorFailedFullsRemainFailed=$true;earlierGreenHead=$earlierHead;basis='Earlier exact C4 six builds, 54 focused native checks and 24 export/resource checks passed. Search allocation, renderer retirement, exports and dependency code are unchanged. Subsequent production changes are limited to popup focus binding; current six builds, all original-eight regressions and focused input/lifetime coverage in all x64 profiles must pass. Full runtime remains omitted at the user''s request.';changedPaths=$delta}|ConvertTo-Json -Depth 4|Set-Content $scopeWaiverPath
    Save-State
    for(;;){
        Assert-Frozen
        $lanes=@(foreach($platform in @('x64','ARM64')){Get-Content (Join-Path $PSScriptRoot ($label+'-build-'+$platform+'.json')) -Raw|ConvertFrom-Json})
        if(@($lanes|Where-Object outcome -EQ failed).Count){throw 'A current-source build lane failed.'}
        if(@($lanes|Where-Object outcome -NE passed).Count -eq 0){break}
        Start-Sleep -Seconds 10
    }
    foreach($lane in $lanes){foreach($entry in $lane.entries){$records.Add($entry)}}
    if(@($records|Where-Object {$_.phase -match '(x64|ARM64)-(ASan-Debug|Debug|Release)-build$' -and $_.exitCode -eq 0}).Count -ne 6){throw 'Six current-source build phases are required.'}
    Save-State
    Import-Module ./Tools/Modules/Build/BuildEvidence.psm1 -Force
    Import-Module ./Tools/Modules/Build/ArtifactOperationLock.psm1 -Force
    Invoke-Phase 'verify-current-build-artifacts' {
        $snapshot=Get-RSBuildSourceSnapshot -RepoRoot $repoPath
        foreach($platform in @('x64','ARM64')){foreach($configuration in @('Debug','Release','ASan Debug')){
            $scope=@{kind='test';target='verify scoped migration builds';suite='Full';platform=$platform;configuration=$configuration}
            $lock=Enter-RSArtifactOperationLock -RepoRoot $repoPath -Operation 'Verify scoped migration builds' -Scope $scope
            try {
                if($lock.WasAbandoned -or (Test-RSArtifactOperationContaminated -RepoRoot $repoPath -Scope $scope)){throw 'Build profile contaminated.'}
                $receipt=Read-RSBuildReceipt -BuildOutputDir (Join-Path $repoPath ('.build\'+$platform+'\'+$configuration)) -SchemaPath (Join-Path $repoPath 'Specs\Build\BuildReceipt.schema.json')
                if(!(Test-RSBuildReceiptForArtifactUse -Receipt $receipt -SourceSnapshot $snapshot -Platform $platform -Configuration $configuration -RepoRoot $repoPath -RequiredRelativePath @(('.build/'+$platform+'/'+$configuration+'/RedSalamander.exe')))){throw 'Current build/artifact identity differs.'}
                [ordered]@{platform=$platform;configuration=$configuration;receipt=$receipt.receipt_id;allArtifactHashesReverified=$true}|ConvertTo-Json -Compress
            }finally{Exit-RSArtifactOperationLock -Lock $lock}
        }}
        $global:LASTEXITCODE=0
    }
    $probe=Join-Path $PSScriptRoot Invoke-OwnedResidualProbe.ps1
    Invoke-Phase 'original-commands-ASan' {& pwsh -NoProfile -File $probe -ExpectedHead $ExpectedHead -Label ('original-commands-'+$short+'-asan') -Suite Commands -Configuration 'ASan Debug' -Repeat 2 -CaseFilter 'theme_cycle_overlay_retention,cmd_fileops_move_breadcrumb_durable_lifecycle,cmd_pane_fileops_conflict_two_pending_task_identity,cmd_viewer_space_close_does_not_wait_for_retired_model_disposal,cmd_viewer_space_renderer_retirement_is_bounded,cmd_viewer_space_dense_files_exposes_more_than_legacy_topk'}

        foreach($prefix in @('original-commands','original-search','original-selection')){

}catch{$outcome='failed';Save-State;throw}
