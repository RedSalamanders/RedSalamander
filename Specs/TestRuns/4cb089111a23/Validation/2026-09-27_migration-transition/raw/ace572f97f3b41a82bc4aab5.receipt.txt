// Included only by the opt-in documentation gallery. No operation is admitted.
struct FileOpsGalleryMonitor final
{
    RECT work{};
    std::wstring device;
};

[[nodiscard]] std::vector<FileOpsGalleryMonitor> GetFileOpsGalleryMonitors()
{
    std::vector<FileOpsGalleryMonitor> monitors;
    EnumDisplayMonitors(nullptr,
                        nullptr,
                        [](HMONITOR monitor, HDC, LPRECT, LPARAM context) noexcept -> BOOL
    {
        MONITORINFOEXW info{};
        info.cbSize = sizeof(info);
        if (GetMonitorInfoW(monitor, &info))
            reinterpret_cast<std::vector<FileOpsGalleryMonitor>*>(context)->push_back({info.rcWork, info.szDevice});
        return TRUE;
    },
                        reinterpret_cast<LPARAM>(&monitors));
    return monitors;
}

void PlaceFileOpsGalleryWindow(HWND hwnd, const FileOpsGalleryMonitor& monitor, int widthDip, int heightDip)
{
    SetWindowPos(hwnd, HWND_BOTTOM, monitor.work.left + 16, monitor.work.top + 16, 0, 0, SWP_NOSIZE | SWP_NOACTIVATE);
    const UINT dpi = GetDpiForWindow(hwnd);
    RECT rect{0, 0, MulDiv(widthDip, static_cast<int>(dpi), 96), MulDiv(heightDip, static_cast<int>(dpi), 96)};
    AdjustWindowRectExForDpi(
        &rect, static_cast<DWORD>(GetWindowLongPtrW(hwnd, GWL_STYLE)), FALSE, static_cast<DWORD>(GetWindowLongPtrW(hwnd, GWL_EXSTYLE)), dpi);
    SetWindowPos(hwnd,
                 HWND_BOTTOM,
                 monitor.work.left + 16,
                 monitor.work.top + 16,
                 std::min(rect.right - rect.left, monitor.work.right - monitor.work.left - 32),
                 std::min(rect.bottom - rect.top, monitor.work.bottom - monitor.work.top - 32),
                 SWP_NOACTIVATE);
    ShowWindow(hwnd, SW_SHOWNOACTIVATE);
    RedrawWindow(hwnd, nullptr, nullptr, RDW_INVALIDATE | RDW_UPDATENOW | RDW_ALLCHILDREN);
}

[[nodiscard]] bool SaveFileOpsGallerySurface(HWND hwnd,
                                             const std::filesystem::path& output,
                                             std::ofstream& manifest,
                                             std::wstring_view name,
                                             size_t display,
                                             std::wstring_view fixture,
                                             std::span<RedSalamander::TestSupport::WindowScreenshotColorProbe> probes = {})
{
    RedrawWindow(hwnd, nullptr, nullptr, RDW_INVALIDATE | RDW_UPDATENOW | RDW_ALLCHILDREN);
    const UINT dpi         = GetDpiForWindow(hwnd);
    const auto file        = std::format(L"{}__dark__fr-FR__display{}__{}dpi.png", name, display + 1u, dpi);
    const HRESULT captured = RedSalamander::TestSupport::SaveWindowScreenshot(hwnd, output / file, probes);
    if (! probes.empty())
        AppendLog(std::format(L"VisualGalleryWitness: {} matched={} required={} hr=0x{:08X}",
                              file,
                              probes.front().matchedPixels,
                              probes.front().minimumMatchingPixels,
                              static_cast<unsigned long>(captured)));
    if (FAILED(captured))
        return false;
    manifest << Common::Strings::Utf8FromUtf16ReplacingInvalid(file) << '\t' << Common::Strings::Utf8FromUtf16ReplacingInvalid(name) << "\tdark\t0\t" << dpi
             << '\t' << Common::Strings::Utf8FromUtf16ReplacingInvalid(fixture) << '\n';
    manifest.flush();
    AppendLog(std::format(L"VisualGallery: {}", file));
    return manifest.good();
}

struct FileOpsGalleryConfirmationCapture final
{
    std::filesystem::path output;
    std::ofstream* manifest = nullptr;
    std::wstring name;
    size_t display        = 0u;
    bool succeeded        = false;
    unsigned attempts     = 0u;
    COLORREF closeColor   = 0;
    HWND window           = nullptr;
    bool requireScrolling = false;
};
FileOpsGalleryConfirmationCapture* g_fileOpsGalleryConfirmation = nullptr;

void CALLBACK CaptureFileOpsGalleryConfirmation(HWND, UINT, UINT_PTR timerId, DWORD) noexcept
{
    auto* capture = g_fileOpsGalleryConfirmation;
    if (! capture)
        return;
    HWND overlay = nullptr;
    EnumThreadWindows(GetCurrentThreadId(),
                      [](HWND candidate, LPARAM context) noexcept -> BOOL
    {
        wchar_t name[128]{};
        GetClassNameW(candidate, name, static_cast<int>(std::size(name)));
        if (std::wstring_view(name) == L"RedSalamander.AlertOverlayWindow" && IsWindowVisible(candidate))
            *reinterpret_cast<HWND*>(context) = candidate;
        return TRUE;
    },
                      reinterpret_cast<LPARAM>(&overlay));
    RedSalamander::Ui::AlertOverlayWindowDebugSnapshot snapshot{};
    if (! overlay || ! RedSalamander::Ui::DebugGetAlertOverlayWindowSnapshot(overlay, snapshot) || ! snapshot.visible || ! snapshot.hasLayout ||
        snapshot.paintCount == 0u || snapshot.lastDrawOpacity < 0.999f)
    {
        if (++capture->attempts > 100u)
        {
            KillTimer(nullptr, timerId);
            HostBeginPromptShutdown(); // Bounded failure: leave the nested test prompt without accepting it.
        }
        return;
    }
    capture->window = overlay;
    KillTimer(nullptr, timerId);
    // A completed paint can precede WGC's first usable frame. Require the actual
    // semantic close glyph, not merely a successful PNG encoder. The shared
    // capture helper repaints and retries bounded stale frames without activation.
    std::array<RedSalamander::TestSupport::WindowScreenshotColorProbe, 1> probes{};
    probes[0].clientBounds  = snapshot.closeRectPx;
    probes[0].expectedColor = capture->closeColor;
    // The thin diagonal glyph is antialiased at 96 DPI. Allow up to one quarter
    // of the 8-bit channel range, while still requiring four core pixels in its
    // interior rectangle. The retained blank frame matches zero for all three colors.
    probes[0].tolerance             = 64u;
    probes[0].minimumMatchingPixels = 4u;
    capture->succeeded =
        snapshot.usesCompositionPresentation && snapshot.hasPreparedFrame && (! capture->requireScrolling || snapshot.scrollMaximumDip > 0.0f) &&
        SaveFileOpsGallerySurface(
            overlay, capture->output, *capture->manifest, capture->name, capture->display, L"production-confirmation-synthetic-request", probes);
    if (capture->succeeded && snapshot.scrollMaximumDip > 0.0f)
    {
        SendMessageW(overlay, WM_KEYDOWN, VK_END, 0);
        RedSalamander::Ui::AlertOverlayWindowDebugSnapshot scrolled{};
        if (! RedSalamander::Ui::DebugGetAlertOverlayWindowSnapshot(overlay, scrolled) || scrolled.scrollOffsetDip < scrolled.scrollMaximumDip - 0.5f)
            capture->succeeded = false;
        else
            capture->succeeded = SaveFileOpsGallerySurface(overlay,
                                                           capture->output,
                                                           *capture->manifest,
                                                           capture->name + L"-scrolled",
                                                           capture->display,
                                                           L"production-confirmation-synthetic-scroll",
                                                           probes);
        SendMessageW(overlay, WM_KEYDOWN, VK_HOME, 0);
        if (! RedSalamander::Ui::DebugGetAlertOverlayWindowSnapshot(overlay, scrolled) || scrolled.scrollOffsetDip != 0.0f)
            capture->succeeded = false;
        AppendLog(std::format(L"ConfirmationScroll: {} extent={:.1f} end/home dispatch and close-glyph capture verified={}",
                              capture->name,
                              snapshot.scrollMaximumDip,
                              capture->succeeded));
    }
    RedSalamander::Ui::AlertOverlayWindowDebugSnapshot afterCapture{};
    if (RedSalamander::Ui::DebugGetAlertOverlayWindowSnapshot(overlay, afterCapture))
        AppendLog(std::format(L"ConfirmationPaint: {} paints={}/{} opacity={:.3f}/{:.3f} endDraw=0x{:08X} windowState={} color={} close=({},{},{},{})",
                              capture->name,
                              snapshot.paintCount,
                              afterCapture.paintCount,
                              snapshot.lastDrawOpacity,
                              afterCapture.lastDrawOpacity,
                              static_cast<unsigned long>(afterCapture.lastEndDrawResult),
                              afterCapture.renderTargetWindowState,
                              capture->closeColor,
                              snapshot.closeRectPx.left,
                              snapshot.closeRectPx.top,
                              snapshot.closeRectPx.right,
                              snapshot.closeRectPx.bottom));
    SendMessageW(overlay, WM_KEYDOWN, VK_ESCAPE, 0);
}

// Geometry/model checks use the same overlay without taking native foreground focus.
// Native keyboard/UIA acceptance remains a separate governed desktop gate.
[[nodiscard]] bool ValidateFileOpsConfirmationLayout()
{
    using namespace RedSalamander::Ui;
    wil::com_ptr<IDWriteFactory> factory;
    if (FAILED(DWriteCreateFactory(DWRITE_FACTORY_TYPE_SHARED, __uuidof(IDWriteFactory), reinterpret_cast<IUnknown**>(factory.put()))))
        return false;
    for (const float width : {320.0f, 480.0f, 920.0f})
    {
        AlertOverlay overlay;
        overlay.SetScrollingEnabled(true);
        AlertModel model;
        model.title   = L"Confirmer la copie des photographies";
        model.message = L"Vérifiez les emplacements avant de poursuivre cette opération.\n"
                        L"C:\\Photographies\\Archives familiales de plusieurs générations\\Sélection définitive pour archivage.\n"
                        L"D:\\Sauvegardes\\Photographies retouchées en très haute résolution.";
        for (uint32_t id = 1u; id <= 4u; ++id)
            model.options.push_back({id,
                                     L"Vérification après la copie",
                                     {{0u, L"Désactivée"}, {1u, L"Vérifier intégralement chaque fichier après la copie avant de supprimer le fichier source"}},
                                     1u});
        model.buttons = {{10u, L"Copier les fichiers sélectionnés", true}, {11u, L"Annuler cette opération", false}};
        overlay.SetModel(std::move(model));
        if (! overlay.EnsureLayout(factory.get(), width, 480.0f) || ! overlay.DebugAllControlLabelsFit() ||
            (width < 920.0f && overlay.GetScrollMaximum() <= 0.0f))
        {
            Fail(std::format(L"Confirmation layout lost/clipped a control at width {}.", width));
            return false;
        }
        const auto initialLastOption = overlay.GetVisibleControlRect({AlertHitTest::Part::Option, 4u});
        const uint64_t layoutPasses  = overlay.DebugGetLayoutPassCount();
        for (uint32_t repeat = 0u; repeat < 32u; ++repeat)
            static_cast<void>(overlay.EnsureLayout(factory.get(), width, 480.0f));
        if (overlay.DebugGetLayoutPassCount() != layoutPasses)
        {
            Fail(L"Confirmation rebuilt unchanged text/geometry during warm paints.");
            return false;
        }
        for (uint32_t id = 1u; id <= 4u; ++id)
        {
            const AlertHitTest focus{AlertHitTest::Part::Option, id};
            static_cast<void>(overlay.FocusControl(focus));
            const auto rect = overlay.GetVisibleControlRect(focus);
            const auto hit  = overlay.HitTest(D2D1::Point2F((rect.left + rect.right) * 0.5f, (rect.top + rect.bottom) * 0.5f));
            if (rect.bottom <= rect.top || hit.part != focus.part || hit.buttonId != id)
            {
                Fail(L"Confirmation focus did not reveal the correct option hit target.");
                return false;
            }
        }
        static_cast<void>(overlay.ScrollBy(-overlay.GetScrollMaximum()));
        static_cast<void>(overlay.EnsureLayout(factory.get(), width, 480.0f));
        const auto resetLastOption = overlay.GetVisibleControlRect({AlertHitTest::Part::Option, 4u});
        if (overlay.GetScrollOffset() != 0.0f || std::abs(resetLastOption.top - initialLastOption.top) > 0.5f ||
            std::abs(resetLastOption.bottom - initialLastOption.bottom) > 0.5f)
        {
            Fail(std::format(L"Confirmation scroll reset changed visible geometry at width {}: offset {:.1f}, original ({:.1f},{:.1f}), reset ({:.1f},{:.1f}).",
                             width,
                             overlay.GetScrollOffset(),
                             initialLastOption.top,
                             initialLastOption.bottom,
                             resetLastOption.top,
                             resetLastOption.bottom));
            return false;
        }
        const auto footer    = overlay.GetVisibleControlRect({AlertHitTest::Part::Button, 11u});
        const auto footerHit = overlay.HitTest(D2D1::Point2F((footer.left + footer.right) * 0.5f, (footer.top + footer.bottom) * 0.5f));
        if (footerHit.part != AlertHitTest::Part::Button || footerHit.buttonId != 11u)
        {
            Fail(L"Confirmation option hit testing overlapped the pinned footer.");
            return false;
        }
        static_cast<void>(overlay.FocusControl({AlertHitTest::Part::Option, 4u}));
        static_cast<void>(overlay.ActivateOption(4u));
        static_cast<void>(overlay.EnsureLayout(factory.get(), width, 480.0f));
        overlay.RevealFocusedControl();
        if (! overlay.DebugAllControlLabelsFit() || overlay.GetScrollOffset() > overlay.GetScrollMaximum())
            return false;
        static_cast<void>(overlay.EnsureLayout(factory.get(), 920.0f, 1600.0f));
        if (overlay.GetScrollOffset() != 0.0f || overlay.GetScrollMaximum() != 0.0f || ! overlay.DebugAllControlLabelsFit())
        {
            Fail(L"Confirmation resize did not clamp scrolling after content fit.");
            return false;
        }
        static_cast<void>(overlay.EnsureLayout(factory.get(), 320.0f, 80.0f));
        static_cast<void>(overlay.FocusControl({AlertHitTest::Part::Button, 11u}));
        const auto shortAction = overlay.GetVisibleControlRect({AlertHitTest::Part::Button, 11u});
        if (shortAction.bottom <= shortAction.top || shortAction.top < 0.0f || shortAction.bottom > 80.0f || ! overlay.DebugAllControlLabelsFit() ||
            overlay.GetVisibleTextRect().bottom != 0.0f)
        {
            Fail(L"Short confirmation lost its final action or reported invisible text as visible.");
            return false;
        }
        AppendLog(std::format(
            L"ConfirmationLayout: width {} wrapped labels, all options reachable, footer isolation, resize clamping, 32 warm layout hits passed", width));
    }
    return true;
}

[[nodiscard]] bool CaptureFileOpsGallerySupplemental(SelfTestState& state,
                                                     const std::filesystem::path& output,
                                                     std::ofstream& manifest,
                                                     const std::vector<FileOpsGalleryMonitor>& monitors)
{
    if (! ValidateFileOpsConfirmationLayout())
        return false;
    auto* folder                  = TryGetFolderWindow(state.mainWindow);
    auto* operations              = TryGetFileOps(folder);
    auto lifetime                 = std::make_shared<int>(0);
    const bool previousAutoAccept = HostGetAutoAcceptPrompts();
    const auto previousOverride   = HostGetTestPromptResultOverride();
    HostSetAutoAcceptPrompts(false);
    HostClearTestPromptResultOverride();
    auto restorePrompts = wil::scope_exit([&]
    {
        HostSetAutoAcceptPrompts(previousAutoAccept);
        HostSetTestPromptResultOverride(previousOverride);
        HostResetPromptShutdown();
    });
    for (size_t display = 0u; display < monitors.size(); ++display)
    {
        const auto& monitor = monitors[display];
        wil::unique_hwnd anchor(FileOperationsPopup::Create(operations, folder, state.mainWindow, lifetime));
        if (! anchor)
            return false;
        auto* renderer = reinterpret_cast<FileOperationsPopupInternal::FileOperationsPopupState*>(GetWindowLongPtrW(anchor.get(), GWLP_USERDATA));
        renderer->DebugSetVisualScenario({}, false, true);
        PlaceFileOpsGalleryWindow(anchor.get(), monitor, 920, 620);
        for (uint32_t captureIndex = 0u; captureIndex < 30u; ++captureIndex)
        {
            const uint32_t variant = captureIndex % 10u;
            const int widthDip     = captureIndex < 10u ? 920 : 480;
            const int heightDip    = captureIndex < 20u ? 620 : 320;
            PlaceFileOpsGalleryWindow(anchor.get(), monitor, widthDip, heightDip);
            HostFileOperationPromptOptions options{};
            options.sizeBytes = sizeof(options);
            const std::array availability{HOST_FILE_OPERATION_VERIFICATION_SUPPORTED,
                                          HOST_FILE_OPERATION_VERIFICATION_SUPPORTED,
                                          HOST_FILE_OPERATION_VERIFICATION_UNSUPPORTED,
                                          HOST_FILE_OPERATION_VERIFICATION_CHECK_DURING_OPERATION};
            options.verifyAfterCopy              = variant == 7u ? 0u : variant % 2u;
            options.verificationAvailability     = variant == 7u ? HOST_FILE_OPERATION_VERIFICATION_NOT_APPLICABLE : availability[variant % 4u];
            options.clipboardMoveConsumesCutList = variant >= 4u ? 1u : 0u;
            options.linkPolicy                   = variant % 2u ? HOST_FILE_OPERATION_LINK_SKIP : HOST_FILE_OPERATION_LINK_PRESERVE;
            options.executionMode                = variant % 2u ? HOST_FILE_OPERATION_EXECUTION_PARALLEL : HOST_FILE_OPERATION_EXECUTION_QUEUE;
            options.bandwidthLimitBytesPerSecond = variant % 2u ? 64ull * 1024ull * 1024ull : 0u;
            HostPromptRequest request{};
            request.sizeBytes     = sizeof(request);
            request.scope         = HOST_ALERT_SCOPE_WINDOW;
            request.targetWindow  = anchor.get();
            request.severity      = HOST_ALERT_WARNING;
            request.buttons       = HOST_PROMPT_BUTTONS_OK_CANCEL;
            request.defaultResult = HOST_PROMPT_RESULT_CANCEL;
            request.presentation  = variant < 4u   ? HOST_PROMPT_PRESENTATION_COPY
                                    : variant < 8u ? HOST_PROMPT_PRESENTATION_MOVE
                                                   : HOST_PROMPT_PRESENTATION_DELETE;
            request.message =
                L"128 fichiers sélectionnés dans C:\\Photographies\\Vacances en famille — septembre 2026\\Sélection définitive pour impression et archivage.\n"
                L"Destination : D:\\Sauvegardes\\Archives photographiques personnelles\\Exposition annuelle\\Photographies retouchées en très haute "
                L"résolution.\n"
                L"Veuillez vérifier les emplacements et les options avant de poursuivre cette opération.";
            if (variant == 8u)
                request.message = L"Supprimer définitivement les 128 fichiers sélectionnés dans C:\\Photographies\\Archives familiales ?\n"
                                  L"Les fichiers ne seront pas envoyés à la Corbeille. Cette action ne peut pas être annulée.";
            else if (variant == 9u)
                request.message = L"Supprimer définitivement le dossier Photographies originales de la réunion familiale et tout son contenu ?\n"
                                  L"Emplacement : C:\\Photographies\\Archives de plusieurs générations.\n"
                                  L"Le dossier ne sera pas envoyé à la Corbeille. Cette action ne peut pas être annulée.";
            request.fileOperationOptions = variant < 8u ? &options : nullptr;
            FileOpsGalleryConfirmationCapture capture{output,
                                                      &manifest,
                                                      std::format(L"102-confirmation-{}{}",
                                                                  variant,
                                                                  captureIndex < 10u   ? L""
                                                                  : captureIndex < 20u ? L"-narrow"
                                                                                       : L"-short"),
                                                      display};
            const auto& colors           = folder->GetTheme().folderView;
            capture.requireScrolling     = captureIndex >= 20u && variant < 8u;
            capture.closeColor           = ColorToCOLORREF(variant < 4u ? colors.infoText : variant < 8u ? colors.warningText : colors.errorText);
            g_fileOpsGalleryConfirmation = &capture;
            const UINT_PTR timer         = SetTimer(nullptr, 0u, 100u, CaptureFileOpsGalleryConfirmation);
            if (! timer)
                return false;
            auto reset              = wil::scope_exit([&]
            {
                KillTimer(nullptr, timer);
                g_fileOpsGalleryConfirmation = nullptr;
            });
            HostPromptResult result = HOST_PROMPT_RESULT_NONE;
            const HRESULT hr        = HostShowPrompt(request, nullptr, &result);
            if (FAILED(hr) || ! capture.succeeded || result != HOST_PROMPT_RESULT_CANCEL)
            {
                Fail(std::format(L"Confirmation capture failed: variant {}, display {}, width {}, hr 0x{:08X}, result {}",
                                 variant,
                                 display + 1u,
                                 widthDip,
                                 static_cast<unsigned long>(hr),
                                 static_cast<unsigned>(result)));

            RedSalamander::Ui::AlertOverlayWindowDebugSnapshot hidden{};
            if (capture.window && IsWindow(capture.window) &&
                (! RedSalamander::Ui::DebugGetAlertOverlayWindowSnapshot(capture.window, hidden) || hidden.visible || hidden.usesCompositionPresentation ||

            return false;

            operations->DismissCompletedTask(90001u);
            operations->DismissCompletedTask(90002u);
        });
        for (uint64_t id : {90001ull, 90002ull})
        {
            FolderWindow::FileOperationState::CompletedTaskSummary summary{};
            summary.taskId         = id;
            summary.operation      = id == 90001u ? FILESYSTEM_COPY : FILESYSTEM_MOVE;
            summary.resultHr       = E_ACCESSDENIED;
            summary.errorCount     = 14u;
            summary.totalItems     = 128u;
            summary.completedItems = 114u;
            summary.completedTick  = GetTickCount64();
            for (unsigned row = 0u; row < 14u; ++row)
            {
                FolderWindow::FileOperationState::TaskDiagnosticEntry issue{};
                GetLocalTime(&issue.localTime);
                issue.taskId    = id;
                issue.operation = summary.operation;
                issue.severity  = FolderWindow::FileOperationState::DiagnosticSeverity::Error;
                issue.status    = E_ACCESSDENIED;
                issue.category  = L"Vérification et conservation de la source";
                issue.message   = L"Impossible de terminer la vérification : le fichier source a été conservé. Vérifiez les autorisations du dossier de "
                                  L"destination avant de réessayer.";
                issue.sourcePath =
                    std::format(L"C:\\Photographies\\Vacances en famille — septembre 2026\\Photographie panoramique retouchée numéro {:02}.raw", row);

        FileOperationsIssuesPane::SelfTestSnapshot issueSnapshot{};
        if (! FileOperationsIssuesPane::TryGetSelfTestSnapshot(issues.get(), issueSnapshot) || issueSnapshot.rowCount != 28u)
            return false;
        if (! SaveFileOpsGallerySurface(issues.get(), output, manifest, L"104-issues-populated", display, L"production-issues-synthetic-diagnostics"))
            return false;
        FileOperationsIssuesPane::SelfTestSelectTask(issues.get(), 90001u);
        if (! SaveFileOpsGallerySurface(issues.get(), output, manifest, L"105-issues-selected", display, L"production-issues-selection"))
            return false;
        FileOperationsIssuesPane::SelfTestScrollByWheelDetents(issues.get(), -8);
        if (! SaveFileOpsGallerySurface(issues.get(), output, manifest, L"106-issues-scrolled", display, L"production-issues-scroll"))
            return false;

        // Retain 103-106 unchanged; use a fresh narrow, nonactivating production pane
        // for the typed states on each native gallery display.
        issues.reset();
        operations->DismissCompletedTask(90001u);
        operations->DismissCompletedTask(90002u);
        wil::unique_hwnd typedIssues(FileOperationsIssuesPane::Create(operations, folder, state.mainWindow, lifetime));
        if (! typedIssues)
            return false;
        PlaceFileOpsGalleryWindow(typedIssues.get(), monitor, 620, 620);
        auto clearTypedIssues  = wil::scope_exit([&]
        {
            for (uint64_t taskId : {91001ull, 91002ull, 91003ull})
                operations->DebugRemoveDiagnosticsForTask(taskId);
        });
        const auto selectIssue = [&](std::wstring_view name, uint64_t taskId, uint64_t generation, FileOperationsIssuesPane::SelfTestSnapshot& snapshot)
        {
            const bool selected = FileOperationsIssuesPane::SelfTestRefresh(typedIssues.get(), true) &&
                                  FileOperationsIssuesPane::SelfTestSelectTask(typedIssues.get(), taskId, generation);
            if (selected)
                static_cast<void>(FileOperationsIssuesPane::SelfTestScrollByWheelDetents(typedIssues.get(), -8));
            if (! selected || ! FileOperationsIssuesPane::TryGetSelfTestSnapshot(typedIssues.get(), snapshot, true) || snapshot.selectionCount != 1u ||
                snapshot.primarySelectedTaskId != taskId || snapshot.primarySelectedRowId == 0u)
            {
                Fail(std::format(L"Issues gallery {} could not select task {} generation {}.", name, taskId, generation));
                return false;
            }
            return true;
        };
        const auto unknownDecision    = LoadStringResource(nullptr, IDS_FILEOPS_ISSUES_DECISION_UNKNOWN);
        const auto unknownResult      = LoadStringResource(nullptr, IDS_FILEOPS_ISSUES_RESULT_UNKNOWN);
        const auto failedResult       = LoadStringResource(nullptr, IDS_FILEOPS_STATUS_FAILED_SHORT);
        using GalleryDiagnostic       = FolderWindow::FileOperationState::TaskDiagnosticEntry;
        const auto semanticDetailFits = [&](std::wstring_view scene,
                                            const FileOperationsIssuesPane::SelfTestSnapshot& snapshot,
                                            std::array<wchar_t, 3> fluent,
                                            std::array<wchar_t, 3> fallback)
        {
            constexpr std::array<size_t, 3> labels{{0u, 2u, 3u}};
            for (size_t slot = 0u; slot < labels.size(); ++slot)
            {
                const size_t index     = labels[slot];
                const wchar_t expected = fluent[slot] != 0 && snapshot.detailIconsUseFluentFont ? fluent[slot] : fallback[slot];
                const auto icon        = snapshot.detailIconBounds[index];
                const auto text        = snapshot.detailTextBounds[index];
                if (snapshot.detailIconGlyphs[index] != expected || icon.right <= icon.left || icon.left < snapshot.detailViewport.left ||
                    icon.right + 4.0f > text.left || text.right > snapshot.detailViewport.right || snapshot.detailRequiredHeights[index] <= 0.0f ||
                    snapshot.detailRequiredHeights[index] > text.bottom - text.top + 0.5f)
                {
                    Fail(std::format(L"Issues gallery {} semantic icon/text bounds or state failed for detail section {}.", scene, index));
                    return false;
                }
            }

                {{L"subject", 0u, 240.0f}, {L"problem", 1u, 240.0f}, {L"summary-operation", 2u, 90.0f}, {L"decision", 3u, 140.0f}, {L"result", 4u, 140.0f}}};
            return std::ranges::all_of(legacyLayout, unchanged) && std::ranges::all_of(summaryDefaults, unchanged);
        };
        for (unsigned reopen = 0u; reopen < 2u; ++reopen)
        {
            wil::unique_hwnd migrated(FileOperationsIssuesPane::Create(&retention, folder, state.mainWindow, retentionLifetime));
            if (! migrated)
                return false;
            PlaceFileOpsGalleryWindow(migrated.get(), monitor, 1000, 620);
            FileOperationsIssuesPane::SelfTestSnapshot migratedSnapshot{};
            const std::vector<std::wstring> summaryIds{L"subject", L"problem", L"summary-operation", L"decision", L"result"};
            if (! FileOperationsIssuesPane::SelfTestRefresh(migrated.get(), true) ||
                ! FileOperationsIssuesPane::TryGetSelfTestSnapshot(migrated.get(), migratedSnapshot, true) || migratedSnapshot.technicalExpanded ||
                migratedSnapshot.columnIds != summaryIds || ! migratedSnapshot.hasActiveSort || ! migratedSnapshot.sortDescending ||
                migratedSnapshot.hiddenSortText.empty())
            {
                Fail(L"Legacy Issues migration lost the default summary or hidden source sort.");
                return false;
            }
            if (reopen == 0u && ! SaveFileOpsGallerySurface(
                                    migrated.get(), output, manifest, L"133a-issues-legacy-summary", display, L"production-issues-legacy-layout-migration"))
                return false;
            if (! FileOperationsIssuesPane::SelfTestSetTechnicalExpanded(migrated.get(), true))
                return false;
            RedrawWindow(migrated.get(), nullptr, nullptr, RDW_INVALIDATE | RDW_UPDATENOW | RDW_ALLCHILDREN);
            if (! FileOperationsIssuesPane::TryGetSelfTestSnapshot(migrated.get(), migratedSnapshot, true) || ! migratedSnapshot.technicalExpanded ||
                migratedSnapshot.columnIds.size() != 10u || migratedSnapshot.columnBounds.size() != 10u || ! migratedSnapshot.hasActiveSort ||
                migratedSnapshot.sortColumnIndex != 8u || ! migratedSnapshot.sortDescending || ! migratedSnapshot.hiddenSortText.empty())
            {
                Fail(L"Legacy Issues migration lost the technical source sort or columns.");
                return false;
            }
            // Snapshot IDs/bounds use model order, independent of saved display order.
            // Resolve by ID; the first three saved columns fit without clipped widths.
            const auto boundsById = [&](std::wstring_view id) noexcept
            {
                const auto found = std::ranges::find(migratedSnapshot.columnIds, id);
                return found == migratedSnapshot.columnIds.end()
                           ? D2D1::RectF()
                           : migratedSnapshot.columnBounds[static_cast<size_t>(found - migratedSnapshot.columnIds.begin())];
            };
            const auto operationBounds = boundsById(L"operation");
            const auto taskBounds      = boundsById(L"task");
            const auto sourceBounds    = boundsById(L"source");
            if (std::abs(operationBounds.right - operationBounds.left - 123.0f) > 0.5f || std::abs(taskBounds.right - taskBounds.left - 91.0f) > 0.5f ||
                std::abs(sourceBounds.right - sourceBounds.left - 311.0f) > 0.5f || operationBounds.right > taskBounds.left + 0.5f ||
                taskBounds.right > sourceBounds.left + 0.5f)
            {
                Fail(L"Legacy Issues migration changed visible saved widths or display order.");
                return false;
            }
            if (reopen == 1u &&
                ! SaveFileOpsGallerySurface(
                    migrated.get(), output, manifest, L"133b-issues-legacy-technical-reopened", display, L"production-issues-legacy-layout-reopen"))
                return false;
            migrated.reset();
            if (! verifyPersistedMigration())
            {
                Fail(L"Saving migrated Issues settings discarded legacy preferences or changed newly appended summary defaults.");
                return false;
            }
        }
        AppendLog(std::format(L"IssuesLegacyMigration: display {} preserved ten legacy columns and source sort, appended five defaults, and reopened.",
                              display + 1u));
    }
    return true;
}
