[[nodiscard]] bool ValidatePendingConflictInspectionSelection(FileOperationsPopupTestFixture&amp; fixture) noexcept
{
    using namespace std::chrono_literals;
    using Status          = FolderView::InspectionSelectionStatus;
    CaseState&amp; state      = fixture.state;
    const auto pane       = FolderWindow::Pane::Left;
    const HWND paneWindow = g_folderWindow.GetFolderViewHwnd(pane);
    auto* view            = reinterpret_cast&lt;FolderView*&gt;(GetWindowLongPtrW(paneWindow, GWLP_USERDATA));
    state.Require(view &amp;&amp; view-&gt;GetHWND() == paneWindow, L"Inspection selection requires the owned live pane.");
    if (! view)
        return false;
    const auto inspection = fixture.taskSnapshot.conflict.inspection;
    FileOperations::ConflictInspectionTarget sourceTarget;
    FileOperations::ConflictInspectionTarget destinationTarget;
    state.Require(SUCCEEDED(fixture.fileOps-&gt;ResolvePendingConflictInspectionTarget(fixture.taskId.value(), inspection, false, sourceTarget)) &amp;&amp;
                      SUCCEEDED(fixture.fileOps-&gt;ResolvePendingConflictInspectionTarget(fixture.taskId.value(), inspection, true, destinationTarget)),
                  L"Both pending roles must resolve their admitted live provider independently of current pane location.");
    if (! state.failure.empty())
        return false;
    state.Require(sourceTarget.context == inspection &amp;&amp; destinationTarget.context == inspection &amp;&amp; ! sourceTarget.destination &amp;&amp;
                      destinationTarget.destination &amp;&amp; sourceTarget.fileSystem &amp;&amp; destinationTarget.fileSystem &amp;&amp; sourceTarget.instanceContext.empty() &amp;&amp;
                      destinationTarget.instanceContext.empty(),
                  L"Inspection must retain the role adapter and raw default context, not host/default as a pane context.");
    state.Require(
        SUCCEEDED(fixture.fileOps-&gt;ResolvePendingConflictInspectionTarget(fixture.taskId.value(), destinationTarget.context, true, destinationTarget)) &amp;&amp;
            destinationTarget.context == inspection,
        L"Requalifying an output using its own context must retain the token before clearing old output.");
    const auto&amp; originalLocation = inspection-&gt;destination.value();
    const auto qualify           = [&amp;](const FileOperations::ConflictInspectionLocation&amp; location, std::wstring_view plugin, std::wstring_view instance)
    { return FileOperations::QualifyConflictInspectionLocation(location, inspection-&gt;operation, destinationTarget.fileSystem.get(), plugin, instance); };
    for (unsigned int changed = 0u; changed &lt; 5u; ++changed)
    {
        auto location = originalLocation;
        switch (changed)
        {
            case 0u: location.endpoint.profileId.append(L"-changed"); break;
            case 1u: location.endpoint.rootId.append(L"-changed"); break;
            case 2u: location.endpoint.pathIdentity-&gt;componentComparison = FileSystemPathComponentComparison::OrdinalCaseSensitive; break;
            case 3u: location.endpoint.pathIdentity-&gt;acceptedSeparators.append(L"!"); break;
            case 4u: location.endpoint.instanceId = L"host/context/changed"; break;
        }
        state.Require(qualify(location, destinationTarget.pluginId, destinationTarget.instanceContext) == HRESULT_FROM_WIN32(ERROR_REVISION_MISMATCH),
                      L"Changed profile/root/path rules/instance must fail closed before pane navigation.");
    }
    auto invalidLocation     = originalLocation;
    invalidLocation.itemName = L"another-item.txt";
    state.Require(
        qualify(invalidLocation, destinationTarget.pluginId, destinationTarget.instanceContext) == E_INVALIDARG &amp;&amp;
            qualify(originalLocation, L"builtin/different-provider", destinationTarget.instanceContext) == HRESULT_FROM_WIN32(ERROR_REVISION_MISMATCH) &amp;&amp;
            qualify(originalLocation, destinationTarget.pluginId, originalLocation.endpoint.instanceId) == HRESULT_FROM_WIN32(ERROR_REVISION_MISMATCH),
        L"Inconsistent item identity, another provider and canonical-as-raw context must not produce a navigable target.");
    invalidLocation = originalLocation;
    invalidLocation.endpoint.pathIdentity.reset();
    state.Require(qualify(invalidLocation, destinationTarget.pluginId, destinationTarget.instanceContext) == HRESULT_FROM_WIN32(ERROR_NOT_SUPPORTED) &amp;&amp;
                      FileOperations::QualifyConflictInspectionLocation(
                          originalLocation, inspection-&gt;operation, nullptr, destinationTarget.pluginId, destinationTarget.instanceContext) ==
                          HRESULT_FROM_WIN32(ERROR_NOT_SUPPORTED),
                  L"Unknown qualification and absent providers must remain unavailable.");
    auto staleContext  = std::make_shared&lt;const FileOperations::ConflictInspectionContext&gt;(*inspection);
    auto clearedTarget = destinationTarget;
    state.Require(fixture.fileOps-&gt;ResolvePendingConflictInspectionTarget(fixture.taskId.value(), staleContext, true, clearedTarget) == E_ABORT &amp;&amp;
                      ! clearedTarget.context &amp;&amp; ! clearedTarget.fileSystem &amp;&amp; clearedTarget.pluginId.empty(),
                  L"A stale equal-path prompt must clear previously qualified output without navigation.");
    const auto identity     = FileSystemPathIdentity::OrdinalIgnoreCaseForLocalFileSystem();
    size_t settledSuccesses = 0u;
    size_t settledFailures  = 0u;
    FolderView::InspectionSelectionResult selectionFromEvent{};
    bool eventOnUiThread       = true;
    bool failureReceiptMatches = true;
    const DWORD owningThread   = GetCurrentThreadId();
    g_folderWindow.SetPaneEnumerationSettledCallback(pane,
                                                     [&amp;](const FolderView::InspectionVisit&amp; eventVisit, const std::filesystem::path&amp; folder, HRESULT status)
    {
        eventOnUiThread = eventOnUiThread &amp;&amp; GetCurrentThreadId() == owningThread;
        if (FAILED(status))
        {
            ++settledFailures;
            const auto failed     = view-&gt;SelectItemForInspection(eventVisit, identity, fixture.dest.native());
            failureReceiptMatches = failureReceiptMatches &amp;&amp; failed.status == Status::EnumerationFailed &amp;&amp; failed.enumerationStatus == status;
        }
        else
        {
            ++settledSuccesses;
            if (folder == fixture.destDir)
                selectionFromEvent = view-&gt;SelectItemForInspection(eventVisit, identity, fixture.dest.native());
        }
    });
    const auto unsubscribe   = wil::scope_exit([&amp;] { g_folderWindow.SetPaneEnumerationSettledCallback(pane, {}); });
    const auto prompt        = fixture.taskSnapshot.conflict.inspection;
    const auto originalVisit = view-&gt;CaptureInspectionVisit();
    g_folderWindow.SetFolderPath(pane, fixture.destDir);
    auto visit = view-&gt;CaptureInspectionVisit();
    state.Require(originalVisit &amp;&amp; visit, L"Both navigation visits must be capturable on the owning UI thread.");
    if (! originalVisit || ! visit)
        return false;
    state.Require(view-&gt;SelectItemForInspection(originalVisit.value(), identity, fixture.dest.native()).status == Status::Stale,
                  L"Navigation must invalidate the previous visit before any late inspection selection.");
    state.Require(view-&gt;SelectItemForInspection(visit.value(), identity, fixture.dest.native()).status == Status::Pending,
                  L"Navigation acceptance must not claim enumeration/selection completion.");
    state.Require(WaitForPaneItems(pane, {fixture.dest.filename().wstring()}, SelfTest::Scale(3000ms)),
                  L"Inspection destination did not enumerate its conflict item.");
    if (! state.failure.empty())
        return false;
    state.Require(settledSuccesses &gt; 0u &amp;&amp; selectionFromEvent.status == Status::Selected &amp;&amp; eventOnUiThread,
                  L"The owning pane settlement event must support exact selection after all enumeration work without polling.");
    // Capture the settled visit; a legitimate cache refresh can supersede the requested enumeration.
    visit = view-&gt;CaptureInspectionVisit();
    if (! visit)
        return false;
    const HWND focusBefore      = GetFocus();
    const HWND foregroundBefore = GetForegroundWindow();
    state.Require(view-&gt;SelectItemForInspection(visit.value(), identity, fixture.dest.native()).status == Status::Selected,
                  L"Inspection must select the exact conflicting item after enumeration settles.");
    state.Require(view-&gt;GetFocusedPath() == fixture.dest &amp;&amp; view-&gt;GetSelectedPaths() == std::vector&lt;std::filesystem::path&gt;{fixture.dest},
                  L"Inspection must establish both one selected item and the exact current item.");
    state.Require(GetFocus() == focusBefore &amp;&amp; GetForegroundWindow() == foregroundBefore,
                  L"Selecting an inspection item must not activate a window or acquire OS focus.");
    visit = view-&gt;CaptureInspectionVisit();
    if (! visit)
        return false;
    const auto selectedBefore = view-&gt;GetSelectedPaths();
    state.Require(view-&gt;SelectItemForInspection(visit.value(), identity, (fixture.destDir / L"missing-inspection-item.txt").native()).status ==
                      Status::MissingOrFiltered,
                  L"A missing/filtered conflict item must not select a neighbor.");
    state.Require(view-&gt;SelectItemForInspection(visit.value(), identity, fixture.source.native()).status == Status::Unavailable,
                  L"An item in another parent must not be selected by its coincident leaf name.");
    auto unstable                   = identity;
    unstable.pathTextStableIdentity = false;
    state.Require(view-&gt;SelectItemForInspection(visit.value(), unstable, fixture.dest.native()).status == Status::Unavailable,
                  L"Unknown path identity must fail before changing pane selection.");
    std::wstring upperName = fixture.dest.filename().wstring();
    CharUpperBuffW(upperName.data(), static_cast&lt;DWORD&gt;(upperName.size()));
    auto sensitive                = identity;
    sensitive.componentComparison = FileSystemPathComponentComparison::OrdinalCaseSensitive;
    state.Require(upperName != fixture.dest.filename().native(), L"Case fixture must differ from the existing exact spelling.");
    state.Require(view-&gt;SelectItemForInspection(visit.value(), sensitive, (fixture.destDir / upperName).native()).status == Status::MissingOrFiltered,
                  L"Inspection must not use case-insensitive fallback under a case-sensitive provider contract.");
    state.Require(view-&gt;GetSelectedPaths() == selectedBefore &amp;&amp; view-&gt;GetFocusedPath() == fixture.dest,
                  L"Rejected targets must leave current and selection unchanged.");
    state.Require(view-&gt;SelectItemForInspection(visit.value(), identity, (fixture.destDir / upperName).native()).status == Status::Selected,
                  L"A uniquely matching local case-insensitive identity should resolve its actual item spelling.");
    const HWND otherWindow = g_folderWindow.GetFolderViewHwnd(FolderWindow::Pane::Right);
    auto* other            = reinterpret_cast&lt;FolderView*&gt;(GetWindowLongPtrW(otherWindow, GWLP_USERDATA));
    state.Require(other &amp;&amp; other-&gt;SelectItemForInspection(visit.value(), identity, fixture.dest.native()).status == Status::Stale,
                  L"A visit token from the other pane must never authorize selection.");
    view-&gt;ForceRefresh();
    state.Require(view-&gt;SelectItemForInspection(visit.value(), identity, fixture.dest.native()).status == Status::Stale,
                  L"A newer same-folder enumeration must supersede an inspection completion.");
    state.Require(WaitForPaneItems(pane, {fixture.dest.filename().wstring()}, SelfTest::Scale(3000ms)), L"Refreshed conflict destination did not settle.");
    const auto beforeFailure = view-&gt;CaptureInspectionVisit();
    g_folderWindow.SetFolderPath(pane, fixture.destDir / L"missing-inspection-directory");
    const auto failedVisit = view-&gt;CaptureInspectionVisit();
    FolderView::InspectionSelectionResult result{};
    const auto deadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
    if (failedVisit)
    {
        do
        {
            PumpPendingMessages();
            result = view-&gt;SelectItemForInspection(failedVisit.value(), identity, fixture.dest.native());
            if (result.status != Status::Pending)
                break;
            std::this_thread::sleep_for(20ms);
        } while (std::chrono::steady_clock::now() &lt; deadline);
    }
    state.Require(failedVisit &amp;&amp; result.status == Status::EnumerationFailed &amp;&amp; FAILED(result.enumerationStatus),
                  L"Failed navigation must expose its enumeration failure instead of selecting the retained old model.");
    state.Require(settledFailures == 1u &amp;&amp; failureReceiptMatches &amp;&amp; eventOnUiThread,
                  L"The owner must receive one failed settlement with the exact status and no old-model selection.");
    const size_t successesBeforeRestore = settledSuccesses;
    g_folderWindow.SetFolderPath(pane, fixture.destDir);
    // A failed enumeration retains the old item model. Finding that same leaf
    // cannot establish that this new navigation has settled. Wait on the existing
    // owner completion event before installing the reentrant-selection witness.
    const auto restoreDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
    while ((settledSuccesses == successesBeforeRestore || ! view-&gt;IsCurrentFolderEnumerated()) &amp;&amp; std::chrono::steady_clock::now() &lt; restoreDeadline)
    {
        PumpPendingMessages();
        std::this_thread::sleep_for(20ms);
    }
    state.Require(settledSuccesses &gt; successesBeforeRestore &amp;&amp; view-&gt;IsCurrentFolderEnumerated() &amp;&amp; selectionFromEvent.status == Status::Selected,
                  L"Restoring the conflict folder must complete a new successful enumeration and exact selection, not reuse retained cached items.");
    state.Require(WaitForPaneItems(pane, {fixture.dest.filename().wstring()}, SelfTest::Scale(3000ms)), L"Restored conflict folder did not enumerate.");
    if (! state.failure.empty())
        return false;
    state.Require(beforeFailure &amp;&amp; view-&gt;SelectItemForInspection(beforeFailure.value(), identity, fixture.dest.native()).status == Status::Stale,
                  L"Returning to identical path text must not resurrect an earlier navigation visit.");
    state.Require(fixture.fileOps-&gt;IsPendingConflictInspectionCurrent(fixture.taskId.value(), prompt),
                  L"Pane inspection selection must leave the same prompt generation pending.");
    state.Require(TextFileEqualsForFileOpsPrompt(fixture.dest, "existing payload"),
                  L"Inspection must preserve the exact original destination bytes while the conflict remains pending.");
    state.Require(view-&gt;DebugValidateInspectionSelectionReentryForSelfTest(identity, fixture.dest.native()),
                  L"Reentrant owner navigation must invalidate selection without reusing its old item index.");
    size_t selfUnsubscribeCount = 0u;
    g_folderWindow.SetPaneEnumerationSettledCallback(pane,
                                                     [&amp;](const FolderView::InspectionVisit&amp;, const std::filesystem::path&amp;, HRESULT)
    {
        g_folderWindow.SetPaneEnumerationSettledCallback(pane, {});
        ++selfUnsubscribeCount;
    });
    g_folderWindow.SetFolderPath(pane, fixture.sourceDir);
    state.Require(WaitForPaneItems(pane, {fixture.source.filename().wstring()}, SelfTest::Scale(3000ms)), L"Inspection fixture must restore its source pane.");
    view-&gt;ForceRefresh();
    state.Require(WaitForPaneItems(pane, {fixture.source.filename().wstring()}, SelfTest::Scale(3000ms)) &amp;&amp; selfUnsubscribeCount == 1u,
                  L"A settled callback must safely unsubscribe itself and receive no later event.");
    size_t obsoleteSettlements        = 0u;
    size_t currentSettlements         = 0u;
    bool redirectedFromLegacyCallback = false;
    g_folderWindow.SetPaneEnumerationSettledCallback(pane,
                                                     [&amp;](const FolderView::InspectionVisit&amp;, const std::filesystem::path&amp; folder, HRESULT)
    {
        if (folder == fixture.destDir)
            ++obsoleteSettlements;
        if (folder == fixture.sourceDir)
            ++currentSettlements;
    });
    g_folderWindow.SetPaneEnumerationCompletedCallback(pane,
                                                       [&amp;](const std::filesystem::path&amp; folder)
    {
        if (folder == fixture.destDir &amp;&amp; ! redirectedFromLegacyCallback)
        {
            redirectedFromLegacyCallback = true;
            g_folderWindow.SetFolderPath(pane, fixture.sourceDir);
        }
    });
    const auto restoreCompletedCallback = wil::scope_exit([&amp;] { g_folderWindow.SetPaneEnumerationCompletedCallback(pane, {}); });
    g_folderWindow.SetFolderPath(pane, fixture.destDir);
    state.Require(WaitForPanePath(pane, fixture.sourceDir, SelfTest::Scale(3000ms)) &amp;&amp;
                      WaitForPaneItems(pane, {fixture.source.filename().wstring()}, SelfTest::Scale(3000ms)) &amp;&amp; redirectedFromLegacyCallback &amp;&amp;
                      obsoleteSettlements == 0u &amp;&amp; currentSettlements &gt; 0u,
                  L"Reentrant navigation from legacy completion must suppress the old settlement and notify only the new visit.");
    return state.failure.empty();
}

[[nodiscard]] bool ValidatePendingConflictInspectionNavigation(FileOperationsPopupTestFixture&amp; fixture) noexcept
{
    using namespace std::chrono_literals;
    using Status          = FileOperations::ConflictInspectionNavigationStatus;
    using Result          = FileOperations::ConflictInspectionNavigationResult;
    auto&amp; owner           = g_folderWindow;
    auto&amp; state           = fixture.state;
    auto* operations      = fixture.fileOps;
    const auto context    = fixture.taskSnapshot.conflict.inspection;
    const auto activePane = owner.GetActivePane();
    const auto leftPath   = owner.GetCurrentPath(FolderWindow::Pane::Left);
    const auto rightPath  = owner.GetCurrentPath(FolderWindow::Pane::Right);
    const auto restore    = wil::scope_exit([&amp;]
    {
        operations-&gt;CancelPendingConflictInspection();
        if (leftPath)
            owner.SetFolderPath(FolderWindow::Pane::Left, leftPath.value());
        if (rightPath)
            owner.SetFolderPath(FolderWindow::Pane::Right, rightPath.value());
        owner.SetActivePane(activePane);
    });
    std::optional&lt;Result&gt; result;
    const auto complete = [&amp;](const Result&amp; value) { result = value; };
    const auto wait     = [&amp;]()
    {
        const auto deadline = std::chrono::steady_clock::now() + SelfTest::Scale(4000ms);
        while (! result &amp;&amp; std::chrono::steady_clock::now() &lt; deadline)
        {
            PumpPendingMessages();
            std::this_thread::sleep_for(10ms);
        }
        return result.has_value();
    };
    for (const auto pane : {FolderWindow::Pane::Left, FolderWindow::Pane::Right})
    {
        owner.SetActivePane(pane);
        const bool destination = pane == FolderWindow::Pane::Left;
        result.reset();
        const HWND focus      = GetFocus();
        const HWND foreground = GetForegroundWindow();
        state.Require(SUCCEEDED(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(), context, destination, complete)) &amp;&amp; wait() &amp;&amp;
                          result-&gt;status == Status::Selected,
                      L"Inspection must navigate the captured active pane and select the exact pending role item.");
        const auto expected = destination ? fixture.dest : fixture.source;
        const auto selected = owner.DebugGetPaneCommandTargetPathsForSelfTest(pane);
        state.Require(owner.GetCurrentPath(pane) == expected.parent_path() &amp;&amp; selected.size() == 1u &amp;&amp; selected.front() == expected &amp;&amp; GetFocus() == focus &amp;&amp;
                          GetForegroundWindow() == foreground,
                      L"The non-activating owner transaction must select only its exact item and preserve OS focus.");
        if (! state.failure.empty())
            return false;
    }
    owner.SetActivePane(FolderWindow::Pane::Left);
    result.reset();
    size_t completionCount = 0u;
    HRESULT chainedStatus  = E_PENDING;
    state.Require(SUCCEEDED(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(),
                                                                       context,
                                                                       true,
                                                                       [&amp;](const Result&amp; first)
    {
        ++completionCount;
        state.Require(first.status == Status::Selected, L"First chained inspection must select its target.");
        chainedStatus = operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(),
                                                                   context,
                                                                   false,
                                                                   [&amp;](const Result&amp; second)
        {
            ++completionCount;
            complete(second);
        });
    })) &amp;&amp; wait() &amp;&amp; SUCCEEDED(chainedStatus) &amp;&amp;
                      completionCount == 2u &amp;&amp; result-&gt;status == Status::Selected,
                  L"A completion callback must be able to begin the next inspection without retaining the old request or listener.");
    if (! state.failure.empty())
        return false;
    result.reset();
    state.Require(SUCCEEDED(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(), context, true, complete)),
                  L"Inspection cancellation fixture must begin navigation.");
    if (! result)
    {
        state.Require(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(), context, false, complete) == HRESULT_FROM_WIN32(ERROR_BUSY),
                      L"An in-flight inspection must reject a duplicate without replacing its completion owner.");
        operations-&gt;CancelPendingConflictInspection();
        state.Require(result &amp;&amp; result-&gt;status == Status::Stale, L"Explicit cancellation must complete once as stale without waiting for enumeration.");
    }
    PumpPendingMessages();
    owner.SetFolderPath(FolderWindow::Pane::Left, fixture.sourceDir);
    state.Require(WaitForPaneItems(FolderWindow::Pane::Left, {fixture.source.filename().wstring()}, SelfTest::Scale(3000ms)), L"Source reset failed.");
    result.reset();
    state.Require(SUCCEEDED(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(), context, true, complete)), L"Superseded inspection must start.");
    owner.SetFolderPath(FolderWindow::Pane::Left, fixture.sourceDir);
    state.Require(wait() &amp;&amp; result-&gt;status == Status::Stale &amp;&amp; owner.GetCurrentPath(FolderWindow::Pane::Left) == fixture.sourceDir,
                  L"A newer pane navigation must supersede an earlier inspection without pulling the pane back.");
    if (! state.failure.empty())
        return false;

    // Change only immutable display-inspection facts in the blocked harness prompt.
    // Engine paths, bindings, decision, repeat scope and source/destination bytes stay unchanged.
    auto* task = operations-&gt;FindTask(fixture.taskId.value());
    if (! task)
        return false;
    const auto restoreContext = wil::scope_exit([&amp;]
    {
        std::scoped_lock lock(task-&gt;_conflictArbiter.mutex);
        task-&gt;_conflictArbiter.prompt.pathDisplay.inspection = context;
    });
    for (const bool missingParent : {false, true})
    {
        auto altered                       = std::make_shared&lt;FileOperations::ConflictInspectionContext&gt;(*context);
        const auto parent                  = missingParent ? fixture.destDir / L"inspection-parent-that-does-not-exist" : fixture.destDir;
        const auto path                    = parent / L"inspection-item-that-does-not-exist.txt";
        altered-&gt;destination-&gt;parentPath   = parent.native();
        altered-&gt;destination-&gt;providerPath = path.native();
        altered-&gt;destination-&gt;itemName     = path.filename().native();
        {
            std::scoped_lock lock(task-&gt;_conflictArbiter.mutex);
            task-&gt;_conflictArbiter.prompt.pathDisplay.inspection = altered;
        }
        const auto previous = owner.GetCurrentPath(FolderWindow::Pane::Left);
        result.reset();
        state.Require(SUCCEEDED(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(), altered, true, complete)) &amp;&amp; wait(),
                      L"Missing inspection target must produce an event-driven terminal result.");
        state.Require(result &amp;&amp; result-&gt;status == (missingParent ? Status::NavigationFailed : Status::MissingOrFiltered),
                      L"Missing file and failed folder enumeration must have distinct truthful inspection outcomes.");
        if (missingParent)
            state.Require(owner.GetCurrentPath(FolderWindow::Pane::Left) == previous,
                          L"Failed folder navigation must settle restoration of the previous pane location.");
        if (! state.failure.empty())
            return false;
    }
    {
        std::scoped_lock lock(task-&gt;_conflictArbiter.mutex);
        task-&gt;_conflictArbiter.prompt.pathDisplay.inspection = context;
    }
    result.reset();
    auto stale = std::make_shared&lt;const FileOperations::ConflictInspectionContext&gt;(*context);
    state.Require(operations-&gt;BeginPendingConflictInspection(fixture.taskId.value(), stale, true, complete) == E_ABORT &amp;&amp; ! result,
                  L"A stale equal-path prompt must be refused before navigation and must not claim request acceptance.");
    state.Require(operations-&gt;IsPendingConflictInspectionCurrent(fixture.taskId.value(), context),
                  L"Inspection must leave the original engine prompt unresolved and current.");
    return state.failure.empty();
}

[[nodiscard]] bool ValidatePendingConflictInspectionControls(FileOperationsPopupTestFixture&amp; fixture) noexcept
{
    using namespace std::chrono_literals;
    Localization::LanguagePreference previousLanguage{};
    if (g_settings.ui.has_value() &amp;&amp; ! g_settings.ui-&gt;language.empty() &amp;&amp; g_settings.ui-&gt;language != L"system")
        previousLanguage = {Localization::LanguagePreferenceKind::Culture, g_settings.ui-&gt;language};
    const auto restoreLanguage = wil::scope_exit([&amp;] { static_cast&lt;void&gt;(Localization::ApplyLanguagePreference(previousLanguage)); });
    fixture.state.Require(SUCCEEDED(Localization::ApplyLanguagePreference({Localization::LanguagePreferenceKind::Culture, L"fr-FR"})),
                          L"Inspection results require the French satellite.");
    // ApplyCurrentSettingsToRunningApp routes a live language change through
    // WM_THEMECHANGED; exercise that same notification on the existing popup.
    std::wstring englishTitle;
    for (const auto* culture : {L"en-US", L"fr-FR"})
    {
        fixture.state.Require(SUCCEEDED(Localization::ApplyLanguagePreference({Localization::LanguagePreferenceKind::Culture, culture})),
                              L"Live caption coverage requires the English and French satellites.");
        SendMessageW(fixture.popup, WM_THEMECHANGED, 0, 0);
        const std::wstring title = Win32Text::GetWindowTextString(fixture.popup);
        fixture.state.Require(title == LoadStringResource(nullptr, IDS_FILEOPS_POPUP_TITLE),
                              L"An existing popup must refresh its native title when the language changes.");
        if (englishTitle.empty())
            englishTitle = title;
        else
            fixture.state.Require(title != englishTitle, L"The French popup title must differ from the English title.");
    }
    const auto captures = SelfTest::GetSuiteArtifactPath(SelfTest::SelfTestSuite::Commands, L"conflict-inspection");
    fixture.state.Require(! captures.empty() &amp;&amp; SelfTest::EnsureDirectory(captures), L"Inspection capture directory is unavailable.");
    auto&amp; state                   = fixture.state;
    auto&amp; owner                   = g_folderWindow;
    const auto activePane         = owner.GetActivePane();
    const auto leftPath           = owner.GetCurrentPath(FolderWindow::Pane::Left);
    const auto rightPath          = owner.GetCurrentPath(FolderWindow::Pane::Right);
    const auto context            = fixture.taskSnapshot.conflict.inspection;
    const HWND originalForeground = GetForegroundWindow();
    const HWND originalFocus      = GetFocus();
    POINT originalCursor{};
    const bool cursorSaved = GetCursorPos(&amp;originalCursor) != FALSE;
    const auto restore     = wil::scope_exit([&amp;]
    {
        fixture.fileOps-&gt;CancelPendingConflictInspection();
        if (leftPath)
            owner.SetFolderPath(FolderWindow::Pane::Left, leftPath.value());
        if (rightPath)
            owner.SetFolderPath(FolderWindow::Pane::Right, rightPath.value());
        owner.SetActivePane(activePane);
        if (cursorSaved)
            SetCursorPos(originalCursor.x, originalCursor.y);
        if (originalForeground &amp;&amp; IsWindow(originalForeground))
            SetForegroundWindow(originalForeground);
        if (originalFocus &amp;&amp; IsWindow(originalFocus))
            SetFocus(originalFocus);
    });
    const auto redraw      = [&amp;] { RedrawWindow(fixture.popup, nullptr, nullptr, RDW_INVALIDATE | RDW_UPDATENOW | RDW_ALLCHILDREN); };
    const auto capture     = [&amp;](std::wstring_view name, UINT messageId)
    {
        redraw();
        FileOperationsPopupInternal::PopupLayoutDebugSnapshot layout{};
        layout.taskId = fixture.taskId.value();
        state.Require(DebugGetFileOperationsPopupLayoutSnapshot(fixture.popup, layout) &amp;&amp; layout.taskInspectionResultFits,
                      L"The painted inspection result must retain its complete wrapped text height, not just a nonempty clipped semantic rectangle.");
        wil::com_ptr_nothrow&lt;IRawElementProviderSimple&gt; resultProvider;
        wil::com_ptr_nothrow&lt;IRawElementProviderSimple&gt; scopeProvider;
        wil::com_ptr_nothrow&lt;IRawElementProviderFragment&gt; resultFragment;
        wil::com_ptr_nothrow&lt;IRawElementProviderFragment&gt; scopeFragment;

[[nodiscard]] bool TestFileOperationsConflictResultRetention(CaseState&amp; state) noexcept
{
    using State      = FolderWindow::FileOperationState;
    auto* operations = g_folderWindow.DebugGetFileOperationState();
    state.Require(operations != nullptr, L"File operations state is required.");
    if (! operations)
        return false;
    const auto previous = g_settings.fileOperations;
    auto restore        = wil::scope_exit([&amp;] { g_settings.fileOperations = previous; });
    if (! g_settings.fileOperations)
        g_settings.fileOperations.emplace();
    g_settings.fileOperations-&gt;diagnosticsInfoEnabled = false;
    State::Task task(*operations);
    task._taskId = 0xF500000000000000ull + GetTickCount64();
    task._sourcePaths.resize(5u);
    auto cleanup = wil::scope_exit([&amp;] { operations-&gt;DebugRemoveDiagnosticsForTask(task._taskId); });
    FileOperations::FileOperationItemResult result{};
    result.completion          = FileOperations::ItemCompletion::Completed;
    result.publication         = FileOperations::PublicationState::Published;
    result.status              = S_OK;
    result.conflictObservation = FileOperations::ConflictObservationReference{task._taskId, 27u};
    state.Require(task.StoreTypedItemResult(result), L"First typed result must be accepted.");
    result.completion = FileOperations::ItemCompletion::Failed;
    state.Require(! task.StoreTypedItemResult(result), L"A second terminal result cannot replace the first observation.");
    result.sourceIndex = 1u;
    result.conflictObservation-&gt;taskId += 1u;
    state.Require(task.StoreTypedItemResult(result), L"A foreign observational reference must not change terminal result acceptance.");
    result.sourceIndex         = 2u;
    result.conflictObservation = FileOperations::ConflictObservationReference{task._taskId, 0u};
    state.Require(task.StoreTypedItemResult(result), L"A missing generation must not change terminal result acceptance.");
    result.sourceIndex = 3u;
    result.conflictObservation.reset();
    state.Require(task.StoreTypedItemResult(result), L"Uncorrelated executors must retain their ordinary terminal result.");
    result.sourceIndex         = 99u;
    result.conflictObservation = FileOperations::ConflictObservationReference{task._taskId, 28u};
    state.Require(! task.StoreTypedItemResult(result), L"An invalid source index cannot publish an observation.");
    std::vector&lt;State::TaskDiagnosticEntry&gt; entries;
    operations-&gt;CollectDiagnostics(entries);
    size_t observations = 0u;
    for (const auto&amp; entry : entries)
    {
        if (entry.taskId != task._taskId || ! entry.conflictResult)
            continue;
        ++observations;
        state.Require(entry.conflictResult-&gt;reference == FileOperations::ConflictObservationReference{task._taskId, 27u} &amp;&amp;
                          entry.conflictResult-&gt;completion == FileOperations::ItemCompletion::Completed &amp;&amp; entry.severity == State::DiagnosticSeverity::Info,
                      L"The exact first result must survive disabled informational diagnostics without becoming a warning.");
    }
    state.Require(observations == 1u, L"Only one exact first-truth conflict observation may be retained.");
    return state.failure.empty();
}

[[nodiscard]] bool TestFileOperationsAcceptedDecisionRecords(HWND mainWindow, CaseState&amp; state, bool requireDisplayFacts = false) noexcept
{
    using namespace std::chrono_literals;
    using Task                  = FolderWindow::FileOperationState::Task;
    using Entry                 = FolderWindow::FileOperationState::TaskDiagnosticEntry;

    const auto restoreSettings  = wil::scope_exit([&amp;] { g_settings.fileOperations = previousSettings; });
    if (! g_settings.fileOperations)
        g_settings.fileOperations.emplace();
    g_settings.fileOperations-&gt;diagnosticsInfoEnabled = false;
    struct DecisionScenario
    {
        Task::ConflictAction action;
        bool cachedScope;
    };
    constexpr std::array&lt;DecisionScenario, 8&gt; scenarios{{{Task::ConflictAction::Overwrite, false},
                                                         {Task::ConflictAction::Skip, false},
                                                         {Task::ConflictAction::KeepBoth, false},
                                                         {Task::ConflictAction::Cancel, false},
                                                         {Task::ConflictAction::None, false},
                                                         {Task::ConflictAction::Retry, false},
                                                         {Task::ConflictAction::Overwrite, true},
                                                         {Task::ConflictAction::SkipAll, true}}};
    for (const auto&amp; scenario : scenarios)
    {
        const auto action         = scenario.action;
        const auto acceptedAction = action == Task::ConflictAction::SkipAll ? Task::ConflictAction::Skip : action;
        FileOperationsPopupTestFixture fixture(state);
        if (! InitializeFileOperationsPopupTestFixture(mainWindow, fixture, false, scenario.cachedScope))
            return false;
        const auto taskId = fixture.taskId.value();
        auto* task        = fixture.fileOps-&gt;FindTask(taskId);
        state.Require(task != nullptr, L"The decision fixture requires its admitted task.");
        if (! task)
            return false;
        Task::ConflictDecisionScope expectedScope;
        HRESULT expectedProblem     = S_OK;
        uint64_t expectedGeneration = 0u;
        std::optional&lt;size_t&gt; expectedSourceIndex;
        std::wstring expectedSourcePath;
        std::wstring expectedDestinationPath;
        std::array&lt;std::wstring, 4&gt; expectedDisplay;
        {
            std::scoped_lock lock(task-&gt;_conflictArbiter.mutex);
            expectedScope           = task-&gt;_conflictArbiter.prompt.decisionScope;
            expectedProblem         = task-&gt;_conflictArbiter.prompt.status;
            expectedGeneration      = task-&gt;_conflictArbiter.prompt.generation;
            expectedSourceIndex     = task-&gt;_conflictArbiter.prompt.sourceItemIndex;
            expectedSourcePath      = task-&gt;_conflictArbiter.prompt.sourcePath;
            expectedDestinationPath = task-&gt;_conflictArbiter.prompt.destinationPath;
            const auto&amp; display     = task-&gt;_conflictArbiter.prompt.pathDisplay;
            expectedDisplay         = {{display.sourceName, display.sourceParentPath, display.destinationName, display.destinationParentPath}};
        }
        // This fixture admits the builtin local provider. Either worker may reach the first prompt.
        if (requireDisplayFacts)
            state.Require(expectedDisplay[0] == std::filesystem::path(expectedSourcePath).filename().wstring() &amp;&amp;
                              expectedDisplay[1] == std::filesystem::path(expectedSourcePath).parent_path().wstring() &amp;&amp;
                              expectedDisplay[2] == std::filesystem::path(expectedDestinationPath).filename().wstring() &amp;&amp;
                              expectedDisplay[3] == std::filesystem::path(expectedDestinationPath).parent_path().wstring(),
                          L"Local fixture paths must be qualified by the admitted provider before they are observed.");
        const auto requireOriginalDisplay = [&amp;](const Entry&amp; entry) noexcept
        {
            if (! requireDisplayFacts)
                return;
            state.Require(entry.conflictPrompt.has_value() &amp;&amp; entry.conflictPrompt-&gt;sourceName == expectedDisplay[0] &amp;&amp;
                              entry.conflictPrompt-&gt;sourceParentPath == expectedDisplay[1] &amp;&amp; entry.conflictPrompt-&gt;destinationName == expectedDisplay[2] &amp;&amp;
                              entry.conflictPrompt-&gt;destinationParentPath == expectedDisplay[3],
                          L"Original problem and accepted decision must preserve qualified names/parents after the prompt closes.");
        };
        std::vector&lt;Entry&gt; entries;
        fixture.fileOps-&gt;CollectDiagnostics(entries);
        const auto isDecision = [taskId](const Entry&amp; entry) noexcept { return entry.taskId == taskId &amp;&amp; entry.acceptedDecision.has_value(); };
        state.Require(std::ranges::none_of(entries, isDecision), L"A waiting prompt must not publish a fabricated accepted choice.");
        const auto problemReady = RedSalamander::TestSupport::PumpMessagesUntil(
            [&amp;]() noexcept
        {
            fixture.fileOps-&gt;CollectDiagnostics(entries);
            return std::ranges::any_of(entries,
                                       [taskId](const Entry&amp; entry) noexcept { return entry.taskId == taskId &amp;&amp; entry.category == L"item.conflict.prompt"; });
        },
            {.timeout = SelfTest::Scale(2000ms), .operationName = L"original conflict diagnostic publication"});
        state.Require(problemReady.conditionMet, problemReady.timeoutDiagnostic);
        for (const auto&amp; entry : entries)
        {
            if (entry.taskId != taskId || entry.category != L"item.conflict.prompt")
                continue;
            state.Require(entry.conflictPrompt.has_value() &amp;&amp; entry.conflictPrompt-&gt;generation == expectedGeneration &amp;&amp;
                              entry.conflictPrompt-&gt;bucket == expectedScope.bucket &amp;&amp; entry.conflictPrompt-&gt;sourceItemIndex == expectedSourceIndex &amp;&amp;
                              ! entry.acceptedDecision.has_value(),
                          L"The original problem must carry its exact typed prompt reference without inventing a decision.");
            requireOriginalDisplay(entry);
        }
        const auto startedAt     = std::chrono::steady_clock::now();
        const bool expectsRecord = action != Task::ConflictAction::None &amp;&amp; action != Task::ConflictAction::Retry;
        if (action == Task::ConflictAction::None)
            task-&gt;RequestCancel();
        else
            task-&gt;SubmitConflictDecision(action, scenario.cachedScope);
        const auto completed = RedSalamander::TestSupport::PumpMessagesUntil(
            [&amp;]() noexcept
        {
            FileOperationsPopupInternal::TaskSnapshot snapshot;
            return DebugGetFileOperationsPopupTaskSnapshot(fixture.popup, taskId, snapshot) &amp;&amp; snapshot.finished;
        },
            {.timeout = SelfTest::Scale(5000ms), .operationName = L"accepted conflict decision completion"});
        Debug::Perf::Emit(L"FileOps.DecisionFixture.AcceptToFinishUs",
                          L"",
                          static_cast&lt;uint64_t&gt;(std::chrono::duration_cast&lt;std::chrono::microseconds&gt;(std::chrono::steady_clock::now() - startedAt).count()),
                          static_cast&lt;uint64_t&gt;(action),
                          scenario.cachedScope ? 1u : 0u,
                          completed.conditionMet ? S_OK : HRESULT_FROM_WIN32(ERROR_TIMEOUT));
        state.Require(completed.conditionMet, completed.timeoutDiagnostic);
        fixture.fileOps-&gt;CollectDiagnostics(entries);
        const auto count = std::ranges::count_if(entries, isDecision);
        state.Require(
            count == (expectsRecord ? (scenario.cachedScope ? 2 : 1) : 0),
            std::format(L"Only offered, worker-accepted actions produce a typed record; cancellation and invalid ingress do not: action={}, records={}.",
                        static_cast&lt;unsigned&gt;(action),
                        count));
        const auto found = std::ranges::find_if(entries, [&amp;](const Entry&amp; entry) noexcept {
            return isDecision(entry) &amp;&amp; entry.acceptedDecision-&gt;origin == FileOperations::ConflictDecisionOrigin::User;
        });
        if (found != entries.end())
        {
            const auto&amp; decision = found-&gt;acceptedDecision.value();
            state.Require(found-&gt;conflictPrompt.has_value() &amp;&amp; found-&gt;conflictPrompt-&gt;generation == expectedGeneration &amp;&amp;
                              found-&gt;conflictPrompt-&gt;bucket == expectedScope.bucket &amp;&amp; found-&gt;conflictPrompt-&gt;sourceItemIndex == expectedSourceIndex,
                          L"A decision must retain the same original-problem key despite diagnostic enqueue ordering.");
            state.Require(decision.taskId == taskId &amp;&amp; decision.promptGeneration == expectedGeneration &amp;&amp; expectedGeneration != 0u &amp;&amp;
                              decision.sourceItemIndex == expectedSourceIndex &amp;&amp; decision.action == acceptedAction &amp;&amp; decision.scope == expectedScope &amp;&amp;
                              decision.applyToAll == scenario.cachedScope &amp;&amp; decision.origin == FileOperations::ConflictDecisionOrigin::User &amp;&amp;
                              decision.originatingPromptGeneration == decision.promptGeneration,
                          L"The record must retain the accepted action, exact prompt generation, selected root and scope, with explicit user origin.");
            state.Require(found-&gt;status == expectedProblem &amp;&amp; FAILED(expectedProblem),
                          L"Accepted-choice diagnostics must retain the original conflict, never manufacture success.");
            state.Require(found-&gt;sourcePath == expectedSourcePath &amp;&amp; found-&gt;destinationPath == expectedDestinationPath,
                          L"Decision diagnostics must retain the prompt's exact paths, not later task progress.");
            requireOriginalDisplay(*found);
        }
        if (scenario.cachedScope)
        {
            const auto cached = std::ranges::find_if(entries, [&amp;](const Entry&amp; entry) noexcept {
                return isDecision(entry) &amp;&amp; entry.acceptedDecision-&gt;origin == FileOperations::ConflictDecisionOrigin::CachedScope;
            });
            state.Require(cached != entries.end(), L"The second conflict must record cached scope rather than new user consent.");
            if (cached != entries.end())
            {
                const auto&amp; decision = cached-&gt;acceptedDecision.value();
                state.Require(cached-&gt;conflictPrompt.has_value() &amp;&amp; cached-&gt;conflictPrompt-&gt;generation == decision.promptGeneration &amp;&amp;
                                  cached-&gt;conflictPrompt-&gt;bucket == decision.scope.bucket &amp;&amp; decision.taskId == taskId &amp;&amp;
                                  decision.promptGeneration &gt; expectedGeneration &amp;&amp; decision.originatingPromptGeneration == expectedGeneration &amp;&amp;
                                  decision.action == acceptedAction &amp;&amp; decision.scope == expectedScope &amp;&amp; decision.applyToAll &amp;&amp;
                                  decision.sourceItemIndex.has_value() &amp;&amp; decision.sourceItemIndex != expectedSourceIndex &amp;&amp;
                                  cached-&gt;sourcePath != expectedSourcePath &amp;&amp; cached-&gt;destinationPath != expectedDestinationPath,
                              L"Cached decisions must preserve their grant, exact scope and distinct prompt/item identity.");
                if (requireDisplayFacts)
                    state.Require(cached-&gt;conflictPrompt.has_value() &amp;&amp; cached-&gt;conflictPrompt-&gt;sourceName != expectedDisplay[0] &amp;&amp;
                                      cached-&gt;conflictPrompt-&gt;sourceName == std::filesystem::path(cached-&gt;sourcePath).filename().wstring() &amp;&amp;
                                      cached-&gt;conflictPrompt-&gt;destinationName == std::filesystem::path(cached-&gt;destinationPath).filename().wstring() &amp;&amp;
                                      cached-&gt;conflictPrompt-&gt;sourceParentPath == expectedDisplay[1] &amp;&amp;
                                      cached-&gt;conflictPrompt-&gt;destinationParentPath == expectedDisplay[3],
                                  L"Cached observations must retain this item's qualified name, not the original grant's filename.");
            }
            state.Require(TextFileEqualsForFileOpsPrompt(fixture.destDir / L"second-payload.txt",
                                                         action == Task::ConflictAction::Overwrite ? "second new payload" : "second existing payload"),
                          L"Cached observations must agree with the second file's actual outcome.");
        }
        std::vector&lt;Entry&gt; outcomeDiagnostics;
        fixture.fileOps-&gt;CollectDiagnostics(outcomeDiagnostics);
        state.Require(
            std::ranges::none_of(outcomeDiagnostics, [&amp;](const Entry&amp; entry) noexcept { return entry.taskId == taskId &amp;&amp; entry.conflictResult.has_value(); }),
            L"Copy outcomes cannot be attached to a decision through matching paths, root index, or task completion.");
        if (expectsRecord)
        {
            std::filesystem::path exportPath;
            state.Require(fixture.fileOps-&gt;ExportTaskIssuesReport(taskId, &amp;exportPath, false),
                          L"Accepted decisions must remain available through completed task export.");
            wil::unique_handle exportFile(CreateFileW(exportPath.c_str(),
                                                      GENERIC_READ,
                                                      FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
                                                      nullptr,
                                                      OPEN_EXISTING,
                                                      FILE_ATTRIBUTE_NORMAL,
                                                      nullptr));
            LARGE_INTEGER size{};
            const bool bounded = exportFile &amp;&amp; GetFileSizeEx(exportFile.get(), &amp;size) &amp;&amp; size.QuadPart &gt; 0 &amp;&amp; size.QuadPart &lt;= 1024 * 1024 &amp;&amp;
                                 size.QuadPart % sizeof(wchar_t) == 0;
            state.Require(bounded, L"Decision export must be a bounded UTF-16 document.");
            if (bounded)
            {
                std::wstring text(static_cast&lt;size_t&gt;(size.QuadPart) / sizeof(wchar_t), L'\0');
                state.Require(SUCCEEDED(Common::HandleIo::ReadExact(exportFile.get(), text.data(), static_cast&lt;size_t&gt;(size.QuadPart))) &amp;&amp;
                                  text.contains(std::format(L"\"promptGeneration\":{}", expectedGeneration)) &amp;&amp; text.contains(L"\"origin\":\"user\"") &amp;&amp;
                                  (! scenario.cachedScope || text.contains(L"\"origin\":\"cachedScope\"")),
                              L"Export must preserve typed identity and explicit/cached origin without implying success.");
                if (requireDisplayFacts)
                    state.Require(text.contains(L"\"sourceName\":\"payload.txt\"") &amp;&amp; text.contains(L"\"destinationName\":\"payload.txt\"") &amp;&amp;
                                      text.contains(L"\"sourceParentPath\":") &amp;&amp; text.contains(L"\"destinationParentPath\":"),
                                  L"Conflict exports must retain qualified display names and parent folders separately from exact paths.");
            }
        }
        state.Require(TextFileEqualsForFileOpsPrompt(fixture.dest, action == Task::ConflictAction::Overwrite ? "new payload" : "existing payload"),
                      L"Decision records must not change the authoritative operation outcome.");
        state.Require(fixture.Cleanup(), L"Accepted-decision fixture must clean its owned task.");
        if (! state.failure.empty())
            return false;
    }
    return state.failure.empty();
}

[[nodiscard]] bool TestFileOperationsPopupConflictCloseAndEscape(HWND mainWindow, CaseState&amp; state) noexcept
{
    using namespace std::chrono_literals;
    FileOperationsPopupTestFixture fixture(state);
    if (InitializeFileOperationsPopupTestFixture(mainWindow, fixture) &amp;&amp; ValidateFileOperationsConflictAndMetadata(fixture) &amp;&amp;
        ValidatePendingConflictInspectionSelection(fixture) &amp;&amp; ValidatePendingConflictInspectionNavigation(fixture) &amp;&amp;
        ValidatePendingConflictInspectionControls(fixture))
    {
        const HWND popup = fixture.popup;
        SendMessageW(popup, WM_CLOSE, 0, 0);
        PumpPendingMessages();
        state.Require(IsWindow(popup) != FALSE &amp;&amp; IsWindowVisible(popup) == FALSE,
                      L"Closing an actionable conflict popup should hide it without destroying the task surface.");

        FileOperationsPopupInternal::TaskSnapshot hiddenSnapshot{};
        state.Require(DebugGetFileOperationsPopupTaskSnapshot(popup, fixture.taskId.value(), hiddenSnapshot) &amp;&amp; hiddenSnapshot.conflict.active,
                      L"Closing an actionable conflict popup must not resolve its immutable decision snapshot.");
        const auto inspection = fixture.taskSnapshot.conflict.inspection;
        state.Require(hiddenSnapshot.conflict.inspection == inspection &amp;&amp;
                          fixture.fileOps-&gt;IsPendingConflictInspectionCurrent(fixture.taskId.value(), inspection),
                      L"Hiding a pending popup must preserve its inspection generation and decision scope.");

        SendMessageW(mainWindow, WM_COMMAND, MAKEWPARAM(IDM_VIEW_FILE_OPERATIONS, 0), 0);
        fixture.popup = WaitForWindow(
            [&amp;]() noexcept
        {
            const HWND candidate = fixture.fileOps-&gt;GetPopupHwndForSelfTest();
            return candidate &amp;&amp; IsWindowVisible(candidate) != FALSE ? candidate : nullptr;
        },
            SelfTest::Scale(5000ms));
        state.Require(fixture.popup == popup, L"Show File Operations should reopen the same hidden conflict surface without changing its decision.");

        SetFocus(fixture.popup);
        SendMessageW(fixture.popup, WM_KEYDOWN, VK_ESCAPE, 0);
        PumpPendingMessages();

        const auto finishedDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(5000ms);
        bool canceledByEscape       = false;
        while (std::chrono::steady_clock::now() &lt; finishedDeadline)
        {
            PumpPendingMessages();
            FileOperationsPopupInternal::TaskSnapshot resolved{};
            if (DebugGetFileOperationsPopupTaskSnapshot(fixture.popup, fixture.taskId.value(), resolved) &amp;&amp; (! resolved.conflict.active || resolved.finished))
            {
                canceledByEscape = true;
                break;
            }
            std::this_thread::sleep_for(20ms);
        }
        state.Require(canceledByEscape, L"Escape should invoke the engine-published Cancel action for the current conflict.");
        state.Require(! fixture.fileOps-&gt;IsPendingConflictInspectionCurrent(fixture.taskId.value(), inspection),
                      L"A resolved/cancelled prompt must reject an inspection completion retaining its old generation.");
        state.Require(IsWindow(fixture.popup) != FALSE, L"Escape should resolve the current conflict decision without closing the stable popup surface.");
        state.Require(TextFileEqualsForFileOpsPrompt(fixture.dest, "existing payload"),
                      L"Escape/Cancel should leave the existing destination payload untouched.");
    }
    state.Require(fixture.Cleanup(), L"Conflict close/Escape test did not clean every case-owned task and completion.");
    return state.failure.empty();
}

// C4: the graphics-independent failure surface. One popup HWND, three native children (text, Cancel all,
// Close), no hosted control, no legacy painter, keyboard and UIA through the native classes, and no
// mutation action reachable except Cancel all.
[[nodiscard]] bool ValidateFileOperationsPopupFailureSurface(FileOperationsPopupTestFixture&amp; fixture, std::wstring_view label) noexcept
{
    using namespace std::chrono_literals;
    CaseState&amp; state = fixture.state;
    HWND popup       = fixture.popup;
    PumpPendingMessages();

    FileOperationsPopupInternal::PopupLayoutDebugSnapshot layout{};
    layout.taskId = fixture.taskId.value_or(0u);
    state.Require(DebugGetFileOperationsPopupLayoutSnapshot(popup, layout), std::format(L"{}: failed to capture the layout snapshot.", label));
    state.Require(layout.failureSurfaceActive, std::format(L"{}: the popup should be on its failure surface.", label));
    state.Require(! layout.usesDxUiHost &amp;&amp; layout.hostedActionControlCount == 0u &amp;&amp; layout.hostedProgressControlCount == 0u &amp;&amp;
                      layout.hostedGraphControlCount == 0u,
                  std::format(L"{}: the failure surface must not retain hosted controls.", label));
    state.Require(layout.failureSurfaceChildCount == 3u,
                  std::format(L"{}: expected text, Cancel all, and Close; saw {} native children.", label, layout.failureSurfaceChildCount));
    state.Require(! layout.failureSurfaceText.empty() &amp;&amp; ! layout.globalSummaryText.empty() &amp;&amp;
                      layout.failureSurfaceText.find(layout.globalSummaryText) != std::wstring::npos,
                  std::format(L"{}: the failure text must explain the situation and carry the live summary; text='{}' summary='{}'.",
                              label,
                              layout.failureSurfaceText,
                              layout.globalSummaryText));

    const std::wstring cancelAllText = LoadStringResource(nullptr, IDS_FILEOPS_BTN_CANCEL_ALL);
    const std::wstring closeText     = LoadStringResource(nullptr, IDS_FILEOPS_BTN_CLOSE);
    const HWND cancelAll             = FindFailureSurfaceButton(popup, cancelAllText);
    const HWND close                 = FindFailureSurfaceButton(popup, closeText);
    state.Require(cancelAll != nullptr &amp;&amp; close != nullptr &amp;&amp; IsWindowVisible(cancelAll) != FALSE &amp;&amp; IsWindowVisible(close) != FALSE,
                  std::format(L"{}: native Cancel all and Close buttons must exist and be visible.", label));
    if (! cancelAll || ! close)
    {
        return false;
    }

    const auto uiaStats = CollectVisibleUiaDescendantPatternStats(popup);
    state.Require(uiaStats.has_value() &amp;&amp; uiaStats-&gt;namedButtonControlCount &gt;= 2u &amp;&amp; uiaStats-&gt;invokePatternCount &gt;= 2u,
                  std::format(L"{}: UIA must expose two named, invokable buttons; {}.", label, DescribeFileOpsPatternStats(uiaStats)));

    SetFocus(close);
    PumpPendingMessages();
    state.Require(GetFocus() == close, std::format(L"{}: Close should take keyboard focus.", label));
    SendMessageW(close, WM_KEYDOWN, VK_TAB, 0);
    PumpPendingMessages();
    state.Require(GetFocus() == cancelAll, std::format(L"{}: Tab from Close should focus Cancel all.", label));
    SendMessageW(cancelAll, WM_KEYDOWN, VK_TAB, 0);
    PumpPendingMessages();
    state.Require(GetFocus() == close, std::format(L"{}: Tab from Cancel all should return to Close.", label));

    if (fixture.taskId.has_value())
    {
        FileOperationsPopupInternal::PopupSelfTestInvoke conflictInvoke{};
        conflictInvoke.kind   = FileOperationsPopupInternal::PopupHitTest::Kind::TaskConflictAction;
        conflictInvoke.taskId = fixture.taskId.value();
        conflictInvoke.data   = 1u;
        state.Require(! DebugInvokeFileOperationsPopup(popup, conflictInvoke),
                      std::format(L"{}: a conflict action must not be reachable while the failure surface is shown.", label));
        state.Require(fixture.fileOps &amp;&amp; fixture.fileOps-&gt;FindTask(fixture.taskId.value()) != nullptr,
                      std::format(L"{}: the refused action must leave the task untouched.", label));
    }

    AppTheme highContrast     = fixture.previousTheme;
    highContrast.highContrast = true;
    g_folderWindow.ApplyTheme(highContrast);
    PumpPendingMessages();
    RECT windowRect{};
    static_cast&lt;void&gt;(GetWindowRect(popup, &amp;windowRect));
    SendMessageW(popup, WM_DPICHANGED, MAKEWPARAM(144, 144), reinterpret_cast&lt;LPARAM&gt;(&amp;windowRect));
    PumpPendingMessages();
    FileOperationsPopupInternal::PopupLayoutDebugSnapshot afterTheme{};
    afterTheme.taskId = fixture.taskId.value_or(0u);
    state.Require(DebugGetFileOperationsPopupLayoutSnapshot(popup, afterTheme) &amp;&amp; afterTheme.failureSurfaceActive &amp;&amp;
                      afterTheme.failureSurfaceChildCount == 3u &amp;&amp; IsWindowVisible(cancelAll) != FALSE &amp;&amp; IsWindowVisible(close) != FALSE,
                  std::format(L"{}: the failure surface must survive a high-contrast theme change and a DPI change.", label));

    SetFocus(close);
    SendMessageW(close, WM_KEYDOWN, VK_ESCAPE, 0);
    PumpPendingMessages();
    state.Require(IsWindowVisible(popup) == FALSE, std::format(L"{}: Escape should hide the popup without resolving anything.", label));
    state.Require(fixture.fileOps &amp;&amp; fixture.taskId.has_value() &amp;&amp; fixture.fileOps-&gt;FindTask(fixture.taskId.value()) != nullptr,
                  std::format(L"{}: hiding must not cancel the task.", label));

    if (fixture.fileOps)
    {
        fixture.fileOps-&gt;DebugEnsurePopupVisibleForSelfTest();
        PumpPendingMessages();
        state.Require(IsWindowVisible(popup) != FALSE, std::format(L"{}: the hidden failure surface must show again on request.", label));
        SetFocus(cancelAll);
        SendMessageW(cancelAll, WM_KEYDOWN, VK_RETURN, 0);
        const auto deadline = std::chrono::steady_clock::now() + SelfTest::Scale(5000ms);
        while (fixture.fileOps-&gt;HasActiveOperations() &amp;&amp; std::chrono::steady_clock::now() &lt; deadline)
        {
            PumpPendingMessages();
            std::this_thread::sleep_for(20ms);
        }
        state.Require(! fixture.fileOps-&gt;HasActiveOperations(), std::format(L"{}: Enter on Cancel all must cancel every operation.", label));
    }
    return state.failure.empty();
}

[[nodiscard]] bool TestFileOperationsSpeedLimitPromptUsesDxUiSurface(HWND mainWindow, CaseState&amp; state) noexcept
{
    using namespace std::chrono_literals;
    constexpr uint64_t kInitialLimitBytesPerSecond = 64ull * 1024ull;

    Localization::LanguagePreference previousLanguage{};
    if (g_settings.ui.has_value() &amp;&amp; ! g_settings.ui-&gt;language.empty() &amp;&amp; g_settings.ui-&gt;language != L"system")
        previousLanguage = {Localization::LanguagePreferenceKind::Culture, g_settings.ui-&gt;language};
    const auto restoreLanguage = wil::scope_exit([&amp;] { static_cast&lt;void&gt;(Localization::ApplyLanguagePreference(previousLanguage)); });

        std::wstring lastSnapshotText;

        {

            FileOperationsSpeedLimitPromptDebugSnapshot snapshot{};

            {
                lastSnapshotText = snapshot.text;
            }
            if (snapshot.text == requestedText)
            {
                updated = true;
                break;
            }
            std::this_thread::sleep_for(20ms);
        }
        state.Require(updated,
                      std::format(L"Custom speed-limit prompt text did not update after the debug setter during {}; requested='{}' value='{}' snapshot='{}'.",

                                  requestedText,
                                  lastValueText,

                                  requestedText,
                                  failureContext,
                                  editedSnapshot.text));
        return state.failure.empty();
    };

    const bool shutdownInvoked               = RunFileOperationsSpeedLimitPromptModalCycle(popup,
                                                                                           openPrompt,
                                                                                           [&amp;](const HWND prompt) noexcept
    {
        shutdownPrompt = prompt;
        shutdownPromptObserved.store(prompt != nullptr &amp;&amp; IsWindow(prompt) != FALSE, std::memory_order_release);
        if (! prompt || IsWindow(prompt) == FALSE)
        {
            return;
        }

        state.Require(folderWindow != nullptr &amp;&amp; IsWindow(folderWindow) != FALSE,
                      L"Folder-window handle unavailable for nested speed-limit prompt shutdown validation.");
        state.Require(PostMessageW(folderWindow, WndMsg::kFileOperationShutdownForSelfTest, taskId.value(), 0) != FALSE,
                      L"Failed to queue File Operations shutdown inside the custom speed-limit prompt pump.");
        state.Require(WaitForWindowClosed(prompt, SelfTest::Scale(3000ms)), L"Custom speed-limit prompt did not unwind after nested File Operations shutdown.");
    });
    HostResetPromptShutdown();
    const bool retainedDuringShutdown = g_folderWindow.DebugWasFileOperationStateRetainedDuringNestedPromptShutdown();
    auto* const freshFileOps          = g_folderWindow.DebugGetFileOperationState();
    Debug::Perf::Emit(L"FileOps.SelfTest.SpeedLimitPromptNestedShutdownUs",
                      L"popup-control-prompt-lifetime",
                      Debug::Perf::ElapsedUs(shutdownStartedAt),
                      retainedDuringShutdown ? 1u : 0u,
                      GetFileOperationsSpeedLimitPromptHandle() == nullptr ? 1u : 0u,
                      freshFileOps &amp;&amp; ! freshFileOps-&gt;FindTask(taskId.value()) ? S_OK : E_UNEXPECTED);
    state.Require(shutdownInvoked &amp;&amp; shutdownPromptObserved.load(std::memory_order_acquire),
                  L"Failed to open the custom speed-limit prompt for nested File Operations shutdown validation.");
    state.Require(shutdownPrompt != nullptr &amp;&amp; IsWindow(shutdownPrompt) == FALSE,
                  L"Custom speed-limit prompt remained alive after nested File Operations shutdown.");
    state.Require(retainedDuringShutdown, L"Nested speed-limit prompt shutdown must retain FileOperationState until the modal frame unwinds.");
    state.Require(freshFileOps != nullptr &amp;&amp; ! freshFileOps-&gt;FindTask(taskId.value()),
                  L"Nested speed-limit prompt shutdown must leave no stale task for post-pump speed-limit submission.");

    return state.failure.empty();
}

    {
        return false;
    }

    bool taskStarted           = false;
    while (std::chrono::steady_clock::now() &lt; startedDeadline)

        auto* task = fileOps-&gt;FindTask(taskId.value());
        FileOperationsPopupInternal::TaskSnapshot popupTaskSnapshot{};
        const bool havePopupTaskSnapshot = DebugGetFileOperationsPopupTaskSnapshot(popup, taskId.value_or(0), popupTaskSnapshot);

            taskId = popupTaskSnapshot.taskId;
            task   = fileOps-&gt;FindTask(taskId.value());
        }

        if (task &amp;&amp; task-&gt;HasStarted())
        {
            taskStarted = true;
            break;
        }

        if (havePopupTaskSnapshot)
        {
            taskStarted = true;
            break;
        }

        std::this_thread::sleep_for(20ms);
    }
    state.Require(taskStarted, L"File-operations speed-limit live interaction task did not start in time.");
    if (! taskStarted)
    {
        return false;
    }
    if (auto* task = fileOps-&gt;FindTask(taskId.value()))
    {
        task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
    }

                      std::format(L"Failed to set custom speed-limit prompt DX edit '{}' during {}.", editName, context));

        {

            while (std::chrono::steady_clock::now() &lt; deadline)
            {
                PumpPendingMessages();
                const auto valueState = CollectVisibleDescendantValuePatternStateByName(prompt, UIA_EditControlTypeId, editName);
                if (valueState.has_value() &amp;&amp; valueState-&gt;value == value)
                {
                    return true;
                }
                std::this_thread::sleep_for(20ms);
            }

            const auto valueState = CollectVisibleDescendantValuePatternStateByName(prompt, UIA_EditControlTypeId, editName);
            return valueState.has_value() &amp;&amp; valueState-&gt;value == value;
        };

        state.Require(waitForEditValue(expectedValue),
                      std::format(L"Custom speed-limit prompt DX edit did not update after live UIA interaction during {}.", context));

        FileOperationsSpeedLimitPromptDebugSnapshot editedSnapshot{};
        state.Require(DebugGetFileOperationsSpeedLimitPromptSnapshot(editedSnapshot),
                      std::format(L"Failed to recapture custom speed-limit prompt snapshot during {}.", context));
        state.Require(
            ! editedSnapshot.text.empty(),
            std::format(L"Custom speed-limit prompt should keep non-empty normalized text after the live DX edit during {}; saw empty text.", context));
        return state.failure.empty();
    };

    state.Require(! okButtonText.empty(), L"Failed to resolve OK button caption for custom speed-limit prompt.");
    const std::wstring cancelButtonText = LoadStringResource(nullptr, IDS_BTN_CANCEL);
    state.Require(! cancelButtonText.empty(), L"Failed to resolve Cancel button caption for custom speed-limit prompt.");
    if (! state.failure.empty())
    {
        return false;
    }

    FileOperationsSpeedLimitPromptDebugSnapshot initialSnapshot{};
    HWND confirmPrompt        = nullptr;
    const bool confirmInvoked = RunFileOperationsSpeedLimitPromptModalCycle(popup,

        {
            return;
        }

        state.Require(InvokeVisibleDescendantByName(prompt, UIA_ButtonControlTypeId, okButtonText),

    if (! state.failure.empty())
    {
        return false;
    }

    const auto limitDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
    while (std::chrono::steady_clock::now() &lt; limitDeadline)
    {
        PumpPendingMessages();
        if (trackedTask-&gt;_desiredSpeedLimitBytesPerSecond.load(std::memory_order_acquire) == 128ull * 1024ull)
        {
            break;
        }
        std::this_thread::sleep_for(20ms);
    }
    state.Require(trackedTask-&gt;_desiredSpeedLimitBytesPerSecond.load(std::memory_order_acquire) == 128ull * 1024ull,
                  L"Live confirm interaction did not update the task speed limit.");

    return state.failure.empty();
}

                  L"Failed to set local file-system plugin for speed-limit churn test (left pane).");
    state.Require(SUCCEEDED(g_folderWindow.SetFileSystemPluginForPane(FolderWindow::Pane::Right, L"builtin/file-system")),
                  L"Failed to set local file-system plugin for speed-limit churn test (right pane).");
    g_folderWindow.SetFolderPath(FolderWindow::Pane::Left, sourceDir);
    g_folderWindow.SetFolderPath(FolderWindow::Pane::Right, destDir);
    state.Require(WaitForPanePath(FolderWindow::Pane::Left, sourceDir, SelfTest::Scale(3000ms)), L"Failed to set left pane path for speed-limit churn test.");
    state.Require(WaitForPanePath(FolderWindow::Pane::Right, destDir, SelfTest::Scale(3000ms)), L"Failed to set right pane path for speed-limit churn test.");
    if (! state.failure.empty())
    {
        return false;
    }

    const wil::com_ptr&lt;IFileSystem&gt; leftFileSystem  = dummyFileSystem;
    const wil::com_ptr&lt;IFileSystem&gt; rightFileSystem = dummyFileSystem;
    state.Require(leftFileSystem &amp;&amp; rightFileSystem, L"Failed to resolve dummy file-system interfaces for speed-limit churn test.");
    if (! leftFileSystem || ! rightFileSystem)
    {
        return false;
    }

    std::vector&lt;FolderWindow::FileOperationState::Task*&gt; beforeTasks;
    fileOps-&gt;CollectTasks(beforeTasks);
    std::unordered_set&lt;uint64_t&gt; existingTaskIds;
    existingTaskIds.reserve(beforeTasks.size());
    for (auto* task : beforeTasks)
    {
        if (task)
        {
            existingTaskIds.insert(task-&gt;GetId());
        }
    }

    const HRESULT startHr = fileOps-&gt;AdmitOperation(FILESYSTEM_COPY,
                                                    FolderWindow::Pane::Left,
                                                    FolderWindow::Pane::Right,
                                                    leftFileSystem,
                                                    sourceFiles,
                                                    dummyDestDir,
                                                    static_cast&lt;FileSystemFlags&gt;(FILESYSTEM_FLAG_ALLOW_OVERWRITE),
                                                    false,
                                                    kInitialLimitBytesPerSecond,
                                                    FolderWindow::FileOperationState::ExecutionMode::BulkItems,
                                                    false,
                                                    rightFileSystem);
    state.Require(SUCCEEDED(startHr),
                  std::format(L"Failed to start file-operations speed-limit churn test copy (hr=0x{:08X}).", static_cast&lt;unsigned long&gt;(startHr)));
    if (! state.failure.empty())
    {
        return false;
    }

    std::optional&lt;uint64_t&gt; taskId = ResolveAndPauseNewFileOperationsTaskForSelfTest(fileOps, existingTaskIds, kInitialLimitBytesPerSecond);

    HWND popup = WaitForWindow([&amp;]() noexcept { return fileOps-&gt;GetPopupHwndForSelfTest(); }, SelfTest::Scale(5000ms));
    state.Require(popup != nullptr &amp;&amp; IsWindow(popup) != FALSE, L"File-operations popup did not open for speed-limit churn validation.");
    if (! popup || IsWindow(popup) == FALSE)
    {
        return false;
    }

    const auto readyDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(5000ms);
    if (! taskId.has_value())
    {
        taskId = ResolveNewFileOperationsTaskIdForSelfTest(fileOps, existingTaskIds, SelfTest::Scale(250ms));
    }
    bool taskReady = false;
    if (auto* task = taskId.has_value() ? fileOps-&gt;FindTask(taskId.value()) : nullptr)
    {
        task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
    }
    while (std::chrono::steady_clock::now() &lt; readyDeadline)
    {
        PumpPendingMessages();

        auto* task = taskId.has_value() ? fileOps-&gt;FindTask(taskId.value()) : nullptr;
        FileOperationsPopupInternal::TaskSnapshot popupTaskSnapshot{};
        const bool havePopupTaskSnapshot = DebugGetFileOperationsPopupTaskSnapshot(popup, taskId.value_or(0), popupTaskSnapshot);
        if (havePopupTaskSnapshot &amp;&amp; popupTaskSnapshot.taskId != 0)
        {
            taskId = popupTaskSnapshot.taskId;
            task   = fileOps-&gt;FindTask(taskId.value());
            if (task)
            {
                task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
            }
        }

        if (task)
        {
            taskReady = true;
            task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
            break;
        }

        if (havePopupTaskSnapshot &amp;&amp; popupTaskSnapshot.taskId != 0 &amp;&amp; ! popupTaskSnapshot.finished)
        {
            taskReady = true;
            if (task)
            {
                task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
            }
            break;
        }

        std::this_thread::sleep_for(20ms);
    }
    state.Require(taskReady, L"File-operations speed-limit churn task did not become available in time.");
    if (! taskReady)
    {
        return false;
    }
    state.Require(taskId.has_value(), L"Failed to identify the file-operations task for speed-limit churn validation.");
    if (! taskId.has_value())
    {
        return false;
    }

    FileOperationsPopupInternal::TaskSnapshot readyTaskSnapshot{};
    state.Require(RefreshTrackedFileOperationsTaskForSelfTest(fileOps, popup, taskId, &amp;readyTaskSnapshot),
                  L"Failed to resolve the queued speed-limit churn task before prompt cycling.");
    if (! state.failure.empty())
    {
        return false;
    }

    if (auto* task = fileOps-&gt;FindTask(taskId.value()))
    {
        task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
        if (! task-&gt;IsPaused())
        {
            task-&gt;TogglePause();
        }
    }
    const auto holdDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
    bool taskHeld           = false;
    while (std::chrono::steady_clock::now() &lt; holdDeadline)
    {
        PumpPendingMessages();
        if (auto* task = fileOps-&gt;FindTask(taskId.value()); task)
        {
            taskHeld = true;
            break;
        }
        std::this_thread::sleep_for(20ms);
    }
    state.Require(taskHeld, L"Speed-limit churn task did not remain available before prompt cycling.");
    if (! taskHeld)
    {
        return false;
    }

    FileOperationsPopupInternal::PopupSelfTestInvoke openPrompt{};
    openPrompt.kind   = FileOperationsPopupInternal::PopupHitTest::Kind::TaskSpeedLimit;
    openPrompt.taskId = taskId.value();
    openPrompt.data   = 1u;

    auto ensureChurnTaskForCycle = [&amp;](const size_t cycle, const uint64_t expectedLimit, FileOperationsPopupInternal::TaskSnapshot&amp; cycleSnapshot) -&gt; bool
    {
        if (popup &amp;&amp; IsWindow(popup) != FALSE &amp;&amp; RefreshTrackedFileOperationsTaskForSelfTest(fileOps, popup, taskId, &amp;cycleSnapshot))
        {
            return true;
        }

        std::vector&lt;FolderWindow::FileOperationState::Task*&gt; beforeRestartTasks;
        fileOps-&gt;CollectTasks(beforeRestartTasks);
        std::unordered_set&lt;uint64_t&gt; restartExistingTaskIds;
        restartExistingTaskIds.reserve(beforeRestartTasks.size());
        for (auto* task : beforeRestartTasks)
        {
            if (task)
            {
                restartExistingTaskIds.insert(task-&gt;GetId());
            }
        }

        const HRESULT restartHr = fileOps-&gt;AdmitOperation(FILESYSTEM_COPY,
                                                          FolderWindow::Pane::Left,
                                                          FolderWindow::Pane::Right,
                                                          leftFileSystem,
                                                          sourceFiles,
                                                          dummyDestDir,
                                                          static_cast&lt;FileSystemFlags&gt;(FILESYSTEM_FLAG_ALLOW_OVERWRITE),
                                                          false,
                                                          expectedLimit,
                                                          FolderWindow::FileOperationState::ExecutionMode::BulkItems,
                                                          false,
                                                          rightFileSystem);
        state.Require(SUCCEEDED(restartHr),
                      std::format(L"Failed to restart file-operations speed-limit churn copy before cycle {} (hr=0x{:08X}).",
                                  cycle,
                                  static_cast&lt;unsigned long&gt;(restartHr)));
        if (! state.failure.empty())
        {
            return false;
        }

        taskId = ResolveAndPauseNewFileOperationsTaskForSelfTest(fileOps, restartExistingTaskIds, expectedLimit);

        popup = WaitForWindow([&amp;]() noexcept { return fileOps-&gt;GetPopupHwndForSelfTest(); }, SelfTest::Scale(5000ms));
        state.Require(popup != nullptr &amp;&amp; IsWindow(popup) != FALSE,
                      std::format(L"File-operations popup did not reopen for speed-limit churn cycle {}.", cycle));
        if (! popup || IsWindow(popup) == FALSE)
        {
            return false;
        }

        if (! taskId.has_value())
        {
            taskId = ResolveNewFileOperationsTaskIdForSelfTest(fileOps, restartExistingTaskIds, SelfTest::Scale(5000ms));
        }
        state.Require(taskId.has_value(), std::format(L"Failed to identify restarted speed-limit churn task before cycle {}.", cycle));
        if (! taskId.has_value())
        {
            return false;
        }

        if (auto* task = fileOps-&gt;FindTask(taskId.value()))
        {
            task-&gt;SetDesiredSpeedLimit(expectedLimit);
            if (! task-&gt;IsPaused())
            {
                task-&gt;TogglePause();
            }
        }

        return RefreshTrackedFileOperationsTaskForSelfTest(fileOps, popup, taskId, &amp;cycleSnapshot);
    };

    uint64_t expectedLimit   = kInitialLimitBytesPerSecond;
    constexpr size_t kCycles = 3u;
    for (size_t cycle = 0; cycle &lt; kCycles; ++cycle)
    {
        closePrompt();
        const auto promptCloseDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
        while (std::chrono::steady_clock::now() &lt; promptCloseDeadline)
        {
            PumpPendingMessages();
            const HWND lingeringPrompt = GetFileOperationsSpeedLimitPromptHandle();
            if (! lingeringPrompt || IsWindow(lingeringPrompt) == FALSE)
            {
                break;
            }
            std::this_thread::sleep_for(20ms);
        }
        state.Require(GetFileOperationsSpeedLimitPromptHandle() == nullptr,
                      std::format(L"Custom speed-limit prompt should close before churn cycle {} starts.", cycle));
        // Each churn cycle creates a new provider surface. Release the previous
        // cycle's cached UIA provider before opening the next prompt so a reused
        // HWND cannot return stale edit text.
        ReleaseThreadUiAutomationForSelfTest();
        FileOperationsPopupInternal::TaskSnapshot cycleSnapshot{};
        const bool haveChurnTask = ensureChurnTaskForCycle(cycle, expectedLimit, cycleSnapshot);
        state.Require(haveChurnTask &amp;&amp; taskId.has_value(), std::format(L"Failed to resolve the queued speed-limit churn task before cycle {}.", cycle));
        if (! haveChurnTask || ! taskId.has_value())
        {
            closePrompt();
            return false;
        }

        if (taskId.has_value())
        {
            openPrompt.taskId = taskId.value();
        }
        if (auto* task = fileOps-&gt;FindTask(taskId.value()))
        {
            task-&gt;SetDesiredSpeedLimit(expectedLimit);
            if (! task-&gt;IsPaused())
            {
                task-&gt;TogglePause();
            }
        }
        state.Require(fileOps-&gt;GetPopupHwndForSelfTest() == popup,
                      std::format(L"File-operations popup should stay open for speed-limit churn cycle {}.", cycle));
        if (! state.failure.empty())
        {
            return false;
        }

        const bool accept                = false;
        const uint64_t requestedLimit    = (96ull + 32ull * static_cast&lt;uint64_t&gt;(cycle)) * 1024ull;
        const std::wstring requestedText = std::format(L"{}KB", requestedLimit / 1024ull);

        HWND promptHandle  = nullptr;
        const bool invoked = RunFileOperationsSpeedLimitPromptModalCycle(popup,
                                                                         openPrompt,
                                                                         [&amp;](const HWND prompt) noexcept
        {
            promptHandle = prompt;
            state.Require(prompt != nullptr &amp;&amp; IsWindow(prompt) != FALSE, std::format(L"Custom speed-limit prompt did not open during cycle {}.", cycle));
            if (! prompt || IsWindow(prompt) == FALSE)
            {
                return;
            }

            state.Require(IsOwnedBy(prompt, popup), std::format(L"Custom speed-limit prompt should be owned by the popup during cycle {}.", cycle));
            state.Require(WindowExposesUiaProvider(prompt), std::format(L"Custom speed-limit prompt should answer WM_GETOBJECT during cycle {}.", cycle));

            FileOperationsSpeedLimitPromptDebugSnapshot snapshot{};
            state.Require(DebugGetFileOperationsSpeedLimitPromptSnapshot(snapshot),
                          std::format(L"Failed to capture custom speed-limit prompt snapshot during cycle {}.", cycle));
            state.Require(snapshot.usesDxUiHost, std::format(L"Custom speed-limit prompt should use a DxUi host during cycle {}.", cycle));
            state.Require(snapshot.visibleChildWindowCount &lt;= 1u,
                          std::format(L"Custom speed-limit prompt should not expose more than one visible child window during cycle {}; saw {}.",
                                      cycle,
                                      snapshot.visibleChildWindowCount));
            state.Require(snapshot.initialLimitBytesPerSecond == expectedLimit,
                          std::format(L"Custom speed-limit prompt initial limit mismatch during cycle {}.", cycle));
            state.Require(snapshot.text == FormatBytesCompact(expectedLimit),
                          std::format(L"Custom speed-limit prompt should show the live limit text during cycle {}.", cycle));

            const auto uiaPatternStats = CollectVisibleUiaDescendantPatternStats(prompt);
            state.Require(uiaPatternStats.has_value(),
                          std::format(L"Failed to collect UI Automation stats for custom speed-limit prompt during cycle {}.", cycle));
            if (uiaPatternStats.has_value())
            {
                state.Require(uiaPatternStats-&gt;visibleElementCount &gt; 0u,
                              std::format(L"Custom speed-limit prompt should expose visible UI Automation descendants during cycle {}.", cycle));
                state.Require(uiaPatternStats-&gt;editControlCount &gt; 0u,
                              std::format(L"Custom speed-limit prompt should expose a visible edit descendant during cycle {}.", cycle));
                state.Require(uiaPatternStats-&gt;valuePatternCount &gt; 0u,
                              std::format(L"Custom speed-limit prompt should expose ValuePattern during cycle {}.", cycle));
                state.Require(uiaPatternStats-&gt;buttonControlCount &gt; 0u,
                              std::format(L"Custom speed-limit prompt should expose visible command buttons during cycle {}.", cycle));
            }

            std::optional&lt;UiaValuePatternState&gt; valueState;
            const bool valueStateMatchesSnapshot =
                WaitForVisibleDescendantValuePatternState(prompt, UIA_EditControlTypeId, [&amp;](const UiaValuePatternState&amp; state) noexcept {
                return state.value == snapshot.text;
            }, valueState, std::format(L"File Operations speed-limit churn cycle {} initial ValuePattern read", cycle));
            const auto editValueStates = CollectVisibleDescendantControlValueStates(prompt, UIA_EditControlTypeId);
            const std::wstring valueDiagnostics =
                std::format(L"snapshotText='{}' snapshotInitial={} observed={} stats={} edits=[{}] prompt=0x{:X} focus=0x{:X} foreground=0x{:X}",
                            snapshot.text,
                            snapshot.initialLimitBytesPerSecond,
                            DescribeFileOpsValuePatternState(valueState),
                            DescribeFileOpsPatternStats(uiaPatternStats),
                            DescribeFileOpsControlValueStates(editValueStates),
                            reinterpret_cast&lt;UINT_PTR&gt;(prompt),
                            reinterpret_cast&lt;UINT_PTR&gt;(GetFocus()),
                            reinterpret_cast&lt;UINT_PTR&gt;(GetForegroundWindow()));
            Trace(std::format(L"SpeedLimitChurn: cycle {} value diagnostics: {}", cycle, valueDiagnostics));
            state.Require(valueState.has_value(),
                          std::format(L"Custom speed-limit prompt should expose a visible editable ValuePattern field during cycle {}.", cycle));
            if (valueState.has_value())
            {
                state.Require(! valueState-&gt;isReadOnly, std::format(L"Custom speed-limit prompt field should remain editable during cycle {}.", cycle));
                state.Require(
                    valueStateMatchesSnapshot,
                    std::format(L"Custom speed-limit prompt ValuePattern should settle to '{}' during cycle {}. {}", snapshot.text, cycle, valueDiagnostics));
                state.Require(! valueState-&gt;name.empty(),
                              std::format(L"Custom speed-limit prompt field should expose a stable accessible name during cycle {}.", cycle));
            }

            const auto buttonState = CollectVisibleDescendantNamedElementState(prompt, UIA_ButtonControlTypeId);
            state.Require(buttonState.has_value(), std::format(L"Custom speed-limit prompt should expose a visible DX command button during cycle {}.", cycle));
            if (buttonState.has_value())
            {
                state.Require(! buttonState-&gt;name.empty(),
                              std::format(L"Custom speed-limit prompt command button should expose a stable accessible name during cycle {}.", cycle));
            }

            state.Require(DebugSetFileOperationsSpeedLimitPromptText(requestedText),
                          std::format(L"Failed to set custom speed-limit prompt text during cycle {}.", cycle));
            state.Require(accept ? DebugConfirmFileOperationsSpeedLimitPrompt() : DebugCancelFileOperationsSpeedLimitPrompt(),
                          std::format(L"Failed to close the custom speed-limit prompt during cycle {}.", cycle));
            state.Require(WaitForWindowClosed(prompt, SelfTest::Scale(3000ms)), std::format(L"Custom speed-limit prompt did not close after cycle {}.", cycle));
        });
        state.Require(invoked, std::format(L"Failed to request the custom speed-limit prompt during cycle {}.", cycle));
        state.Require(promptHandle != nullptr &amp;&amp; IsWindow(promptHandle) == FALSE,
                      std::format(L"Custom speed-limit prompt should not remain open after cycle {}.", cycle));

        if (accept)
        {
            expectedLimit = requestedLimit;
        }

        const auto limitDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
        while (std::chrono::steady_clock::now() &lt; limitDeadline)
        {
            PumpPendingMessages();
            if (auto* task = fileOps-&gt;FindTask(taskId.value()); task &amp;&amp; task-&gt;_desiredSpeedLimitBytesPerSecond.load(std::memory_order_acquire) == expectedLimit)
            {
                break;
            }
            std::this_thread::sleep_for(20ms);
        }

        FileOperationsPopupInternal::TaskSnapshot postCycleSnapshot{};
        const bool havePostCycleSnapshot = DebugGetFileOperationsPopupTaskSnapshot(popup, taskId.value_or(0), postCycleSnapshot);
        if (taskId.has_value())
        {
            openPrompt.taskId = taskId.value();
        }
        auto* trackedTask = taskId.has_value() ? fileOps-&gt;FindTask(taskId.value()) : nullptr;
        if (trackedTask != nullptr || (havePostCycleSnapshot &amp;&amp; ! postCycleSnapshot.finished))
        {
            state.Require((trackedTask != nullptr &amp;&amp; trackedTask-&gt;_desiredSpeedLimitBytesPerSecond.load(std::memory_order_acquire) == expectedLimit) ||
                              (havePostCycleSnapshot &amp;&amp; ! postCycleSnapshot.finished &amp;&amp; postCycleSnapshot.desiredSpeedLimitBytesPerSecond == expectedLimit),
                          std::format(L"Speed-limit churn cycle {} did not leave the expected live limit (taskId={}, tracked={}, trackedLimit={}, "
                                      L"snapshotFound={}, snapshotFinished={}, snapshotTaskId={}, snapshotLimit={}, expected={}).",
                                      cycle,
                                      taskId.value_or(0),
                                      trackedTask ? L"yes" : L"no",
                                      trackedTask ? trackedTask-&gt;_desiredSpeedLimitBytesPerSecond.load(std::memory_order_acquire) : 0ull,
                                      havePostCycleSnapshot ? L"yes" : L"no",
                                      havePostCycleSnapshot &amp;&amp; postCycleSnapshot.finished ? L"yes" : L"no",
                                      havePostCycleSnapshot ? postCycleSnapshot.taskId : 0ull,
                                      havePostCycleSnapshot ? postCycleSnapshot.desiredSpeedLimitBytesPerSecond : 0ull,
                                      expectedLimit));
        }

        const HWND lingeringPrompt = GetFileOperationsSpeedLimitPromptHandle();
        state.Require(lingeringPrompt == nullptr || IsWindow(lingeringPrompt) == FALSE,
                      std::format(L"Custom speed-limit prompt should not remain open after cycle {}.", cycle));
        if (! state.failure.empty())
        {
            closePrompt();
            return false;
        }
    }

    return state.failure.empty();
}

[[nodiscard]] bool TestFileOperationsSpeedLimitPromptKeepsNavigationShellStable(HWND mainWindow, CaseState&amp; state) noexcept
{
    using namespace std::chrono_literals;
    Trace(L"SpeedLimitNavShell: begin");

    if (! mainWindow || IsWindow(mainWindow) == FALSE)
    {
        state.Require(false, L"Main window handle invalid.");
        return false;
    }

    auto* fileOps = g_folderWindow.DebugGetFileOperationState();
    state.Require(fileOps != nullptr, L"File-operations state unavailable for speed-limit shell-stability validation.");
    if (! fileOps)
    {
        return false;
    }

    const std::filesystem::path suiteRoot = SelfTest::GetTempRoot(SelfTest::SelfTestSuite::Commands);
    state.Require(! suiteRoot.empty(), L"SelfTest temp root unavailable.");
    if (suiteRoot.empty())
    {
        return false;
    }

    const std::filesystem::path root           = suiteRoot / L"work" / (L"fileops_speed_limit_nav_shell_" + NewGuidText());
    const std::filesystem::path sourceDir      = root / L"src";
    const std::filesystem::path destDir        = root / L"dst";
    const std::filesystem::path sourceFile     = sourceDir / L"payload.bin";
    const std::filesystem::path dummyRoot      = std::filesystem::path(L"/cmd-speedlimit-nav-shell") / SanitizeDummyPathSegmentForFileOpsPrompt(NewGuidText());
    const std::filesystem::path dummySourceDir = dummyRoot / L"src";
    const std::filesystem::path dummyDestDir   = dummyRoot / L"dst";
    constexpr uint64_t kInitialLimitBytesPerSecond = 64ull * 1024ull;

    std::error_code ec;
    std::filesystem::remove_all(root, ec);
    state.Require(SelfTest::EnsureDirectory(sourceDir), L"Failed to create speed-limit shell-stability source directory.");
    state.Require(SelfTest::EnsureDirectory(destDir), L"Failed to create speed-limit shell-stability destination directory.");
    if (! state.failure.empty())
    {
        return false;
    }

    state.Require(SelfTest::WriteTextFile(sourceFile, "payload"), L"Failed to create visible shell-stability payload.");
    if (! state.failure.empty())
    {
        return false;
    }

    FileSystemPluginManager&amp; pluginManager = FileSystemPluginManager::GetInstance();
    static_cast&lt;void&gt;(pluginManager.EnablePlugin(kBuiltinDummyFileSystemIdForFileOpsPrompt.data(), g_settings));
    const FileSystemPluginManager::PluginEntry* dummyEntry = FindFileSystemPluginById(kBuiltinDummyFileSystemIdForFileOpsPrompt);
    state.Require(dummyEntry &amp;&amp; dummyEntry-&gt;fileSystem &amp;&amp; dummyEntry-&gt;informations, L"Loaded dummy file system unavailable for speed-limit shell stability.");
    const wil::com_ptr&lt;IFileSystem&gt; dummyFileSystem = dummyEntry ? dummyEntry-&gt;fileSystem : nullptr;
    const wil::com_ptr&lt;IInformations&gt; dummyInfo     = dummyEntry ? dummyEntry-&gt;informations : nullptr;
    wil::com_ptr&lt;IFileSystemIO&gt; dummyIo;
    state.Require(CreateFileSystemIoForFileOpsPrompt(dummyFileSystem, dummyIo),
                  L"Loaded dummy file system missing IFileSystemIO for speed-limit shell stability.");

    constexpr std::string_view kSeedDummyConfig =
        R"json({"maxChildrenPerDirectory":42,"maxDepth":10,"seed":42,"latencyMs":0,"streamChunkLatencyMs":0,"virtualSpeedLimit":"1048576"})json";
    constexpr std::string_view kSlowDummyConfig =
        R"json({"maxChildrenPerDirectory":42,"maxDepth":10,"seed":42,"latencyMs":0,"streamChunkLatencyMs":1000,"virtualSpeedLimit":"1048576"})json";
    constexpr uint64_t kDummyPayloadBytes                  = 8ull * 1024ull * 1024ull;
    const std::wstring leftPluginBefore                    = std::wstring(g_folderWindow.GetFileSystemPluginId(FolderWindow::Pane::Left));
    const std::wstring rightPluginBefore                   = std::wstring(g_folderWindow.GetFileSystemPluginId(FolderWindow::Pane::Right));
    const std::optional&lt;std::filesystem::path&gt; leftBefore  = g_folderWindow.GetCurrentPath(FolderWindow::Pane::Left);
    const std::optional&lt;std::filesystem::path&gt; rightBefore = g_folderWindow.GetCurrentPath(FolderWindow::Pane::Right);
    const std::optional&lt;Common::Settings::FileOperationsSettings&gt; previousFileOperations = g_settings.fileOperations;
    std::string previousDummyConfig;
    state.Require(BackupPluginConfigurationForFileOpsPrompt(dummyInfo.get(), previousDummyConfig),
                  L"Failed to snapshot dummy configuration for speed-limit shell stability.");
    if (! state.failure.empty())
    {
        return false;
    }
    const auto restoreSettings = wil::scope_exit([&amp;] noexcept
    {
        g_settings.fileOperations = previousFileOperations;
        static_cast&lt;void&gt;(SetPluginConfigurationForFileOpsPrompt(dummyInfo.get(), previousDummyConfig));
    });

    state.Require(SetPluginConfigurationForFileOpsPrompt(dummyInfo.get(), kSeedDummyConfig),
                  L"Failed to apply deterministic dummy seed configuration for speed-limit shell stability.");
    state.Require(EnsureDummyFolderExistsForFileOpsPrompt(dummyFileSystem.get(), ToPluginPathTextForFileOpsPrompt(dummyRoot)),
                  L"Failed to create dummy root folder for speed-limit shell stability.");
    state.Require(EnsureDummyFolderExistsForFileOpsPrompt(dummyFileSystem.get(), ToPluginPathTextForFileOpsPrompt(dummySourceDir)),
                  L"Failed to create dummy source folder for speed-limit shell stability.");
    state.Require(EnsureDummyFolderExistsForFileOpsPrompt(dummyFileSystem.get(), ToPluginPathTextForFileOpsPrompt(dummyDestDir)),
                  L"Failed to create dummy destination folder for speed-limit shell stability.");
    if (! state.failure.empty())
    {
        return false;
    }

    std::vector&lt;std::filesystem::path&gt; sourceFiles;
    sourceFiles.reserve(16u);
    for (size_t index = 0; index &lt; 16u; ++index)
    {
        const std::filesystem::path filePath = index == 0u ? (dummySourceDir / L"payload.bin") : (dummySourceDir / std::format(L"payload_{:02}.bin", index));
        state.Require(WritePatternFileFsIoForFileOpsPrompt(dummyIo, filePath, kDummyPayloadBytes),
                      std::format(L"Failed to seed dummy shell-stability payload {}.", filePath.filename().native()));
        if (! state.failure.empty())
        {
            return false;
        }
        sourceFiles.push_back(filePath);
    }

    state.Require(SetPluginConfigurationForFileOpsPrompt(dummyInfo.get(), kSlowDummyConfig),
                  L"Failed to apply deterministic dummy latency for speed-limit shell stability.");
    if (! state.failure.empty())
    {
        return false;
    }

    const auto closePrompt = []() noexcept
    {
        if (const HWND prompt = GetFileOperationsSpeedLimitPromptHandle(); prompt &amp;&amp; IsWindow(prompt) != FALSE)
        {
            if (! DebugCancelFileOperationsSpeedLimitPrompt())
            {
                PostMessageW(prompt, WM_CLOSE, 0, 0);
            }
            static_cast&lt;void&gt;(WaitForWindowClosed(prompt, SelfTest::Scale(3000ms)));
        }
    };

            {

                                                        FolderWindow::Pane::Left,

            if (task)

                task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);
                break;

            if (havePopupTaskSnapshot &amp;&amp; popupTaskSnapshot.taskId != 0 &amp;&amp; ! popupTaskSnapshot.finished)
            {
                taskReady = true;
                if (task)
                {
                    task-&gt;SetDesiredSpeedLimit(kInitialLimitBytesPerSecond);

        state.Require(taskId.has_value(),

        FileOperationsPopupInternal::TaskSnapshot heldTaskSnapshot{};
        if (RefreshTrackedFileOperationsTaskForSelfTest(fileOps, popup, taskId, &amp;heldTaskSnapshot))

    if (! startTrackedSpeedLimitTask(L"cancel pass"))

    const auto runOwnedPromptCycle = [&amp;](std::wstring_view context, FileOperationsSpeedLimitPromptDebugSnapshot&amp; snapshot, auto&amp;&amp; workerFunc) noexcept -&gt; bool
    {
        const bool requestSpecificTaskId = openPrompt.taskId != 0;
        FileOperationsPopupInternal::TaskSnapshot liveTaskSnapshot{};
        static_cast&lt;void&gt;(RefreshTrackedFileOperationsTaskForSelfTest(fileOps, popup, taskId, &amp;liveTaskSnapshot));
        if (requestSpecificTaskId &amp;&amp; taskId.has_value())
        {
            openPrompt.taskId = taskId.value();
        }

        Trace(std::format(L"SpeedLimitNavShell: invoking prompt for {}", context));

                                                           openPrompt,
                                                           [&amp;](const HWND prompt) noexcept
        {
            FileOperationsPopupInternal::TaskSnapshot taskSnapshot{};

            const bool haveTrackedTask  = taskId.has_value() &amp;&amp; fileOps-&gt;FindTask(taskId.value()) != nullptr;
            state.Require(prompt != nullptr &amp;&amp; IsWindow(prompt) != FALSE,
                          std::format(L"Custom speed-limit prompt did not open during {} (invokeTaskId={}, trackedTaskId={}, trackedLive={}, "
                                      L"popupTaskFound={}, popupTaskId={}, popupTaskFinished={}, popupTaskLimit={}, popupCurrent=0x{:X}, livePopup=0x{:X}).",
                                      context,
                                      openPrompt.taskId,
                                      taskId.value_or(0),
                                      haveTrackedTask ? L"yes" : L"no",
                                      haveTaskSnapshot ? L"yes" : L"no",
                                      haveTaskSnapshot ? taskSnapshot.taskId : 0ull,
                                      haveTaskSnapshot &amp;&amp; taskSnapshot.finished ? L"yes" : L"no",
                                      haveTaskSnapshot ? taskSnapshot.desiredSpeedLimitBytesPerSecond : 0ull,
                                      static_cast&lt;unsigned long long&gt;(reinterpret_cast&lt;uintptr_t&gt;(popup)),
                                      static_cast&lt;unsigned long long&gt;(reinterpret_cast&lt;uintptr_t&gt;(fileOps-&gt;GetPopupHwndForSelfTest()))));
            state.Require(prompt == nullptr || IsOwnedBy(prompt, popup),
                          std::format(L"Custom speed-limit prompt should be owned by the file-operations popup during {}.", context));
            if (! prompt || IsWindow(prompt) == FALSE || ! state.failure.empty())
            {
                return;
            }

            Trace(std::format(L"SpeedLimitNavShell: prompt=0x{:X} opened for {}", reinterpret_cast&lt;uintptr_t&gt;(prompt), context));

            bool settled              = false;
            const auto promptDeadline = std::chrono::steady_clock::now() + SelfTest::Scale(3000ms);
            while (std::chrono::steady_clock::now() &lt; promptDeadline)
            {
                snapshot = {};
                if (DebugGetFileOperationsSpeedLimitPromptSnapshot(snapshot) &amp;&amp; snapshot.usesDxUiHost &amp;&amp; snapshot.visibleChildWindowCount &lt;= 1u &amp;&amp;
                    ! snapshot.text.empty())
                {
                    settled = true;
                    break;
                }
                Sleep(10);
            }

            state.Require(settled,
                          std::format(L"Custom speed-limit prompt did not settle during {}; usesDxUiHost={}, visibleChildren={}, text='{}'.",
                                      context,
                                      snapshot.usesDxUiHost ? L"yes" : L"no",
                                      snapshot.visibleChildWindowCount,
                                      snapshot.text));
            if (! settled || ! state.failure.empty())
            {
                return;
            }

            Trace(std::format(L"SpeedLimitNavShell: prompt settled for {} text='{}'", context, snapshot.text));
            workerFunc(prompt);
        });
    };

    FileOperationsSpeedLimitPromptDebugSnapshot cancelSnapshot{};
    const bool cancelInvoked = runOwnedPromptCycle(L"cancel pass",
                                                   cancelSnapshot,
                                                   [&amp;](const HWND prompt) noexcept
    {
        state.Require(DebugCancelFileOperationsSpeedLimitPrompt(), L"Failed to cancel the custom speed-limit prompt during shell-stability validation.");
        state.Require(WaitForWindowClosed(prompt, SelfTest::Scale(3000ms)),
                      L"Custom speed-limit prompt did not close after cancel during shell-stability validation.");
    });
    state.Require(cancelInvoked, L"Failed to request the custom speed-limit prompt during the cancel shell-stability pass.");
    if (! cancelInvoked || ! state.failure.empty())
    {
        return false;
    }
    Trace(L"SpeedLimitNavShell: cancel pass closed");

    requireAnchoredNavigationShells(L"speed-limit cancel");
    if (! state.failure.empty())
    {
        return false;
    }
    Trace(L"SpeedLimitNavShell: cancel pass shell stable");

    state.Require(CloseFileOperationsPopupForSelfTest(fileOps), L"File-operations popup did not close after cancel shell-stability validation.");
    state.Require(GetFileOperationsSpeedLimitPromptHandle() == nullptr,
                  L"Custom speed-limit prompt should not remain open after cancel shell-stability validation.");
    if (! state.failure.empty())
    {
        return false;
    }
    Trace(L"SpeedLimitNavShell: popup cleanup closed");

    requireStableNavigationShells(L"speed-limit cleanup", false);
    if (state.failure.empty())
    {
        Trace(L"SpeedLimitNavShell: done");
    }
    return state.failure.empty();
}

[[nodiscard]] bool TestLocalBlockedReaderCancellationIsBounded(CaseState&amp; state) noexcept
{
    using PfnRunLocalReaderCancellationTest =
        HRESULT(__stdcall*)(const wchar_t*, uint64_t*, uint64_t*, uint64_t*, unsigned int*, unsigned long*, HRESULT*, BOOL*, BOOL*, BOOL*);

    const std::filesystem::path suiteRoot   = SelfTest::GetTempRoot(SelfTest::SelfTestSuite::Commands);
    const std::filesystem::path root        = suiteRoot / L"work" / (L"local_reader_cancel_" + NewGuidText());
    const std::filesystem::path payloadPath = root / L"payload.bin";
    state.Require(SelfTest::EnsureDirectory(root), L"Failed to create the Local reader cancellation fixture directory.");
    if (! state.failure.empty())
    {
        return false;
    }

    constexpr size_t payloadBytes = 8u * 1024u * 1024u;
    std::vector&lt;std::byte&gt; payload(payloadBytes);
    for (size_t index = 0u; index &lt; payload.size(); ++index)
    {
