param([Parameter(Mandatory)][string]$ExpectedHead)
$ErrorActionPreference='Stop'
$records=[Collections.Generic.List[object]]::new()
$records.Add([pscustomobject]@{configuration='ASan Debug';attempt=1;record=(Join-Path $PSScriptRoot 'owned-residual-seed-memory-baseline-asan-1.json');exitCode=0})
function Save-BaselineState([string]$Phase,[string]$Outcome) {
    [ordered]@{head=$ExpectedHead;ownerPid=$PID;phase=$Phase;outcome=$Outcome;scope='Three retained native NTFS seed baselines per profile; never Full qualification.';updatedUtc=[DateTime]::UtcNow.ToString('o');entries=$records.ToArray()} | ConvertTo-Json -Depth 6 | Set-Content (Join-Path $PSScriptRoot 'ntfs-seed-baselines.json')
}
$first=Get-Content $records[0].record -Raw | ConvertFrom-Json
if($first.source -ne $ExpectedHead -or $first.outcome -ne 'focused_pass_only' -or $first.memorySampleCount -lt 2) {throw 'The first retained baseline is incomplete.'}
try {
    foreach($profile in @('ASan Debug','Release')) {
        $start=if($profile -eq 'ASan Debug'){2}else{1}
        foreach($attempt in $start..3) {
            $label='seed-memory-baseline-'+$(if($profile -eq 'ASan Debug'){'asan'}else{'release'})+'-'+$attempt
            Save-BaselineState $label running
            $log=Join-Path $PSScriptRoot ($label+'-driver.log')
            & pwsh -NoProfile -File (Join-Path $PSScriptRoot 'Invoke-OwnedResidualProbe.ps1') -ExpectedHead $ExpectedHead -Label $label -Suite Compare -Configuration $profile -CaseFilter local_index_core_sqlite_cold_start_stale_root_refreshes_before_query -SampleOwnedMemory *> $log
            $code=$LASTEXITCODE
            $recordPath=Join-Path $PSScriptRoot ('owned-residual-'+$label+'.json')
            $records.Add([pscustomobject]@{configuration=$profile;attempt=$attempt;record=$recordPath;log=$log;exitCode=$code})
            Save-BaselineState $label running
            if($code -ne 0) {throw "Retained NTFS seed baseline failed: $label ($code)."}
            $record=Get-Content $recordPath -Raw | ConvertFrom-Json
            if($record.source -ne $ExpectedHead -or $record.outcome -ne 'focused_pass_only' -or $record.memorySampleCount -lt 2) {throw 'Incomplete retained native baseline.'}
        }
    }
    Save-BaselineState complete diagnostic_baselines_complete
}
catch { Save-BaselineState failed failed; throw }
