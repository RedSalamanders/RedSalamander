function Get-FunctionText([string]$Text,[string]$Name) {
    $tokens=$null;$errors=$null
    $ast=[Management.Automation.Language.Parser]::ParseInput($Text,[ref]$tokens,[ref]$errors)
    if($errors.Count){throw "Parse failed: $Name"}
    $definition=$ast.Find({param($n) $n -is [Management.Automation.Language.FunctionDefinitionAst] -and $n.Name -ceq $Name},$true)
    if($null -eq $definition){throw "Function not found: $Name"}
    return $definition.Extent.Text
}
