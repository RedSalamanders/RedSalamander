param([Parameter(Mandatory)][string]$Executable,[Parameter(Mandatory)][string]$Case)
$ErrorActionPreference='Stop'
if([IO.Path]::GetFullPath($Executable) -ne 'E:\RedSalamander.Perf\integration-20261024\.build\x64\Debug\ViewerPETests.exe'){throw 'Unexpected diagnostic executable.'}
$process=[Diagnostics.Process]::GetCurrentProcess()
$mask=$process.ProcessorAffinity.ToInt64()
$single=0L
foreach($bit in 0..62){$candidate=1L -shl $bit;if($mask -band $candidate){$single=$candidate;break}}
if(!$single){throw 'No diagnostic CPU affinity available.'}
$process.ProcessorAffinity=[IntPtr]$single
Write-Output ('Owned diagnostic wrapper affinity: '+$process.ProcessorAffinity)
& $Executable '--group=noninteractive' $Case
exit $LASTEXITCODE
