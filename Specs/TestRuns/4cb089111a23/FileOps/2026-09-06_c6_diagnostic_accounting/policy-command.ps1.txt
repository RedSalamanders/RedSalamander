$ErrorActionPreference = 'Stop'
$env:REDSALAMANDER_TEST_ROOT = 'C:\RedSalamander.Perf'
$env:REDSALAMANDER_TEST_RUN_ID = 'c6-diagnostics-policy-20260906'
Import-Module .\Tools\Modules\Testing\TestSandbox.psm1
$policyScratch = New-RSTestSandboxScratchDirectory -RepoRoot (Get-Location).Path -Harness policy -Case diagnostics
$env:TEMP = $policyScratch
$env:TMP = $policyScratch
$policyTests = @(
 'Tools/Tests/MSBuildInvocation.Tests.ps1',
 'Tools/Tests/ProcessStreaming.Tests.ps1',
 'Tools/Tests/BuildOutputProcess.Tests.ps1',
 'Tools/Tests/ToolInventory.Tests.ps1',
 'Tools/Tests/ToolingGovernance.Tests.ps1'
)
$policyInputs = @(
 'Tools/Modules/Build/MSBuildInvocation.psm1',
 'Tools/Modules/Build/ProcessStreaming.psm1',
 'Tools/Modules/Build/SanitizedEnvironment.psm1',
 'Tools/Modules/Tooling/ToolInventory.psm1',
 'Tools/Tests/TestSupport.psm1',
 'Tools/tool-inventory.json',
 'Tools/README.md',
 'Specs/Build/Build_Toolchain.md',
 'Specs/NormativeConsistency.json'
) + $policyTests
$inputManifest = foreach ($inputPath in $policyInputs) {
 [ordered]@{ path=$inputPath; sha256=(Get-FileHash -LiteralPath $inputPath).Hash.ToLowerInvariant() }
}
[IO.File]::WriteAllText((Join-Path $policyScratch 'inputs.json'), (@($inputManifest) | ConvertTo-Json -Depth 4), [Text.UTF8Encoding]::new($false))
$policyResult = Invoke-Pester -Script $policyTests -PassThru -OutputFormat NUnitXml -OutputFile (Join-Path $policyScratch 'results.xml')
$policyRecord = [ordered]@{ run_id=$env:REDSALAMANDER_TEST_RUN_ID; git_head=(git rev-parse HEAD); tests=$policyTests; passed=$policyResult.PassedCount; failed=$policyResult.FailedCount; skipped=$policyResult.SkippedCount; input_manifest_sha256=(Get-FileHash -LiteralPath (Join-Path $policyScratch 'inputs.json')).Hash.ToLowerInvariant() }
[IO.File]::WriteAllText((Join-Path $policyScratch 'provenance.json'), ($policyRecord | ConvertTo-Json -Depth 4), [Text.UTF8Encoding]::new($false))
$policyRecord | ConvertTo-Json -Depth 4
exit $policyResult.FailedCount
