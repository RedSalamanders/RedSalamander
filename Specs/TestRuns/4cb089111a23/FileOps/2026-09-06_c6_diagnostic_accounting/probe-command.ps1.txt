$ErrorActionPreference = 'Stop'
$env:REDSALAMANDER_TEST_ROOT = 'C:\RedSalamander.Perf'
$env:REDSALAMANDER_TEST_RUN_ID = 'c6-diagnostics-invariant-candidate-20260906'
Import-Module .\Tools\Modules\Testing\TestSandbox.psm1
$diagScratch = New-RSTestSandboxScratchDirectory -RepoRoot (Get-Location).Path -Harness policy -Case diagnostics
$env:TEMP = $diagScratch
$env:TMP = $diagScratch
Copy-Item -LiteralPath 'Tools/Modules/Build/MSBuildInvocation.psm1' -Destination (Join-Path $diagScratch 'tested-module.psm1')
Copy-Item -LiteralPath 'Tools/Tests/MSBuildInvocation.Tests.ps1' -Destination (Join-Path $diagScratch 'tested-tests.ps1')
$result = Invoke-Pester .\Tools\Tests\MSBuildInvocation.Tests.ps1 -PassThru -OutputFormat NUnitXml -OutputFile (Join-Path $diagScratch 'results.xml')
Import-Module .\Tools\Modules\Build\MSBuildInvocation.psm1 -Force
$diagWitness = 'Specs/TestRuns/4cb089111a23/FileOps/2026-09-06_c0_curl_parser_baseline_debug/build.txt'
$diagSummary = Get-RSMSBuildDiagnosticSummary -LogPath $diagWitness
$diagLines = [IO.File]::ReadAllLines((Join-Path (Get-Location).Path $diagWitness))
$diagMetrics = [Collections.Generic.List[object]]::new()
foreach ($diagMode in @('summary', 'colors')) {
    for ($sample = 0; $sample -lt 6; $sample++) {
        $diagClock = [Diagnostics.Stopwatch]::StartNew()
        for ($iteration = 0; $iteration -lt 10; $iteration++) {
            if ($diagMode -eq 'summary') { $null = Get-RSMSBuildDiagnosticSummary -LogPath $diagWitness }
            else { foreach ($diagLine in $diagLines) { $null = Get-RSMSBuildLineForegroundColor -Line $diagLine } }
        }
        $diagClock.Stop()
        if ($sample -gt 0) {
            $diagMetrics.Add([ordered]@{ metric="build.diagnostics.$diagMode.ms"; sample=$sample; iterations=10; lines=$diagLines.Count; elapsed_ms=$diagClock.Elapsed.TotalMilliseconds; warning_count=$diagSummary.WarningCount; error_count=$diagSummary.ErrorCount })
        }
    }
}
$diagProvenance = [ordered]@{ run_id=$env:REDSALAMANDER_TEST_RUN_ID; git_head=(git rev-parse HEAD); witness=$diagWitness; witness_sha256=(Get-FileHash -LiteralPath $diagWitness).Hash.ToLowerInvariant(); module_sha256=(Get-FileHash 'Tools/Modules/Build/MSBuildInvocation.psm1').Hash.ToLowerInvariant(); tests_sha256=(Get-FileHash 'Tools/Tests/MSBuildInvocation.Tests.ps1').Hash.ToLowerInvariant(); passed=$result.PassedCount; failed=$result.FailedCount; skipped=$result.SkippedCount; real_warning_count=$diagSummary.WarningCount; real_error_count=$diagSummary.ErrorCount }
[IO.File]::WriteAllText((Join-Path $diagScratch 'provenance.json'), ($diagProvenance | ConvertTo-Json -Depth 4), [Text.UTF8Encoding]::new($false))
[IO.File]::WriteAllLines((Join-Path $diagScratch 'perf_metrics.jsonl'), [string[]]@($diagMetrics | ForEach-Object { $_ | ConvertTo-Json -Compress }), [Text.UTF8Encoding]::new($false))
$diagProvenance | ConvertTo-Json
exit $result.FailedCount
