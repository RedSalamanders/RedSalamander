$ErrorActionPreference = 'Stop'
$transitionRepo = (Get-Location).Path
$env:REDSALAMANDER_TEST_ROOT = 'C:\RedSalamander.Perf'
$env:REDSALAMANDER_TEST_RUN_ID = 'c6-production-transition-verify-20260906'
Import-Module .\Tools\Modules\Testing\TestSandbox.psm1
$transitionScratch = New-RSTestSandboxScratchDirectory -RepoRoot $transitionRepo -Harness policy -Case production-transition
$env:TEMP = $transitionScratch
$env:TMP = $transitionScratch
Import-Module .\Tools\Modules\Build\ArtifactOperationLock.psm1
Import-Module .\Tools\Modules\Build\BuildEvidence.psm1
Import-Module .\Tools\Modules\Build\ProcessStreaming.psm1
$transitionOutput = Join-Path $transitionRepo '.build/x64/Release'
$transitionLock = Enter-RSArtifactOperationLock -RepoRoot $transitionRepo -Operation 'Verify production Release transition' -Scope @{configuration='Release';platform='x64';suite='C6 production transition'}
try {
 $afterReceipt = Read-RSBuildReceipt -BuildOutputDir $transitionOutput -SchemaPath (Join-Path $transitionRepo 'Specs/Build/BuildReceipt.schema.json')
 if ($null -eq $afterReceipt -or $afterReceipt.tests_enabled -or $afterReceipt.target -ne 'solution' -or $afterReceipt.git_head -ne (git rev-parse HEAD)) { throw 'Expected a current production full-solution receipt.' }
 foreach ($artifact in $afterReceipt.outputs) {
  $artifactPath = [IO.Path]::GetFullPath((Join-Path $transitionRepo $artifact.relative_path))
  if (-not $artifactPath.StartsWith($transitionOutput + '\', [StringComparison]::OrdinalIgnoreCase)) { throw 'Artifact outside selected output.' }
  $artifactFile = Get-Item -LiteralPath $artifactPath -Force
  if ($artifactFile.PSIsContainer -or ($artifactFile.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0 -or $artifactFile.Length -ne $artifact.size_bytes -or (Get-FileHash -LiteralPath $artifactPath).Hash.ToLowerInvariant() -ne $artifact.sha256) { throw "Production artifact mismatch: $($artifact.relative_path)" }
 }
 Copy-Item -LiteralPath (Join-Path $transitionOutput 'build-receipt.json') -Destination (Join-Path $transitionScratch 'after-receipt.json')
 $beforeReceipt = Get-Content -LiteralPath 'C:\RedSalamander.Perf\runs\c6-production-transition-20260906\scratch\policy\production-transition\before-receipt.json' -Raw | ConvertFrom-Json
 $checks = [Collections.Generic.List[object]]::new()
 $retiredNames = @('CrashHandlingTests.exe','DxUiTests.exe','FileSystemCurlTests.exe','LocalizationTests.exe','MonitorTest.exe','PerformanceTests2.dll','RedConfigureTests.exe','SettingsSchemaTests.exe','ViewerPETests.exe','ViewerSqliteTests.exe','Lang/LocalizationTests-cs-CZ.dll','Lang/LocalizationTests-fr-FR.dll','Lang/LocalizationTests-sk-SK.dll')
 foreach ($name in $retiredNames) {
  $relative = '.build/x64/Release/' + $name
  $before = @($beforeReceipt.outputs | Where-Object relative_path -EQ $relative)
  $after = @($afterReceipt.outputs | Where-Object relative_path -EQ $relative)
  $checks.Add([ordered]@{ name="retired:$name"; passed=($before.Count -eq 1 -and $after.Count -eq 0 -and -not (Test-Path -LiteralPath (Join-Path $transitionOutput $name))); baseline_count=$before.Count; production_count=$after.Count })
 }
 foreach ($name in @('RedSalamander.exe','RedSalamanderSearchService.exe','sqlite3.dll','Plugins/sqlite3.dll','PluginContractTests.exe')) {
  $relative = '.build/x64/Release/' + $name
  $before = @($beforeReceipt.outputs | Where-Object relative_path -EQ $relative)
  $after = @($afterReceipt.outputs | Where-Object relative_path -EQ $relative)
  $checks.Add([ordered]@{ name="retained:$name"; passed=($before.Count -eq 1 -and $after.Count -eq 1); baseline_count=$before.Count; production_count=$after.Count })
 }
 $service = @($afterReceipt.outputs | Where-Object relative_path -EQ '.build/x64/Release/RedSalamanderSearchService.exe')
 $checks.Add([ordered]@{name='service SQLite receipt closure';passed=($service.Count -eq 1 -and '.build/x64/Release/sqlite3.dll' -in $service[0].runtime_closure)})
 foreach ($smoke in @(
  @{name='search-service-help';exe='RedSalamanderSearchService.exe';args=@('--help');timeout=10000},
  @{name='plugin-package-smoke';exe='PluginContractTests.exe';args=@('--package-smoke');timeout=120000}
 )) {
  $smokeClock = [Diagnostics.Stopwatch]::StartNew()
  $smokeExit = Invoke-RSStreamingProcess -FilePath (Join-Path $transitionOutput $smoke.exe) -Arguments $smoke.args -WorkingDirectory $transitionOutput -LogPath (Join-Path $transitionScratch ($smoke.name + '.txt')) -TimeoutMilliseconds $smoke.timeout
  $smokeClock.Stop()
  $checks.Add([ordered]@{name=$smoke.name;passed=($smokeExit -eq 0);exit_code=$smokeExit;elapsed_ms=$smokeClock.Elapsed.TotalMilliseconds})
 }
 $verification = [ordered]@{ git_head=(git rev-parse HEAD); receipt_id=$afterReceipt.receipt_id; source_snapshot_id=$afterReceipt.source_snapshot_id; tests_enabled=$afterReceipt.tests_enabled; artifacts_verified=@($afterReceipt.outputs).Count; checks=@($checks); passed=@($checks | Where-Object passed -EQ $true).Count; failed=@($checks | Where-Object passed -NE $true).Count }
 [IO.File]::WriteAllText((Join-Path $transitionScratch 'verification.json'), ($verification | ConvertTo-Json -Depth 6), [Text.UTF8Encoding]::new($false))
 $verification | Select-Object git_head,receipt_id,source_snapshot_id,tests_enabled,artifacts_verified,passed,failed | ConvertTo-Json -Compress
 if ($verification.failed -ne 0) { throw 'Production transition checks failed; retain verification.json and both smoke logs.' }
}
finally { Exit-RSArtifactOperationLock -Lock $transitionLock }
