param(
 [Parameter(Mandatory)][string]$Repo,
 [Parameter(Mandatory)][string]$ExpectedSource,
 [Parameter(Mandatory)][ValidateSet('Debug','Release','ASan Debug')][string]$Configuration,
 [Parameter(Mandatory)][string]$Label,
 [int]$Repeat=5
)
$ErrorActionPreference='Stop'
Set-Location -LiteralPath $Repo
if((git rev-parse HEAD).Trim() -ne $ExpectedSource -or (git status --porcelain)){throw 'Diagnostic source must be frozen and clean'}
if(Get-Process -Name cl,link,vcpkg,ViewerPETests -ErrorAction SilentlyContinue){throw 'Wait for other native work'}
if(Get-CimInstance Win32_Process -Filter "Name='MSBuild.exe'" | Where-Object CommandLine -NotMatch '/nodemode:'){throw 'Wait for active MSBuild coordinator'}
$env:REDSALAMANDER_TEST_ROOT='Z:/RedSalamander.Perf'
Import-Module ./Tools/Modules/Build/ArtifactOperationLock.psm1 -Force
Import-Module ./Tools/Modules/Build/ProcessStreaming.psm1 -Force
Import-Module ./Tools/Modules/Testing/TestRunPlan.psm1 -Force
$context=New-RSTestRunContext -RepoRoot $Repo
$env:REDSALAMANDER_TEST_RUN_ID=$context.RunId
$env:REDSALAMANDER_SELFTEST_ROOT=$context.LegacySelfTestRoot
$scope=@{platform='x64';configuration=$Configuration}
$lease=Enter-RSArtifactOperationLock -RepoRoot $Repo -Operation 'I19 RedConfigure profile diagnosis' -Scope $scope
try{
 if($lease.WasAbandoned -or (Test-RSArtifactOperationContaminated -RepoRoot $Repo -Scope $scope)){throw 'Official rebuild required before diagnosis'}
 $output=Join-Path $Repo ('.build/x64/'+$Configuration)
 $exe=Join-Path $output 'RedConfigureTests.exe'
 $root=Join-Path 'Z:/RedSalamander.Perf/evidence/I19-20260909' ('redconfigure-profile-'+$Label)
 if(Test-Path -LiteralPath $root){throw 'Preserve diagnostic evidence'}
 New-Item -ItemType Directory -Path $root|Out-Null
 $rows=@()
 for($iteration=1;$iteration -le $Repeat;$iteration++){
  $log=Join-Path $root ('run-'+$iteration+'.log')
  $exitCode=Invoke-RSStreamingProcess -FilePath $exe -Arguments @() -WorkingDirectory $output -LogPath $log -TimeoutMilliseconds 120000
  $text=Get-Content -LiteralPath $log -Raw
  $metric=[regex]::Match($text,'redconfigure\.theme\.mass_preview keys=512 elapsedMicros=(\d+)')
  if(-not $metric.Success){throw 'Run did not reach the measured theme fixture; inspect original output'}
  $rows += [ordered]@{iteration=$iteration;exitCode=$exitCode;elapsedMicros=[long]$metric.Groups[1].Value;completedAllTests=($text -match 'RedConfigureTests passed\.');log=$log}
  [ordered]@{source=$ExpectedSource;configuration=$Configuration;platform='x64';runId=$context.RunId;module=$exe;moduleSha256=(Get-FileHash -LiteralPath $exe).Hash;runs=$rows;noHostedCi=$true}|ConvertTo-Json -Depth 5|Set-Content (Join-Path $root 'summary.json')
 }
 Get-Content -LiteralPath (Join-Path $root 'summary.json')
}finally{Exit-RSArtifactOperationLock -Lock $lease}
