#define REDSAL_DEFINE_TRACE_PROVIDER
#include "Helpers.h"
#include <cstdio>
struct First { unsigned value; };
struct Second { unsigned value; };
int main() {
 wil::unique_hwnd first(CreateWindowExW(0,L"STATIC",L"",0,0,0,0,0,HWND_MESSAGE,nullptr,GetModuleHandleW(nullptr),nullptr));
 wil::unique_hwnd other(CreateWindowExW(0,L"STATIC",L"",0,0,0,0,0,HWND_MESSAGE,nullptr,GetModuleHandleW(nullptr),nullptr));
 if(!first || !other) return 2;
 InitPostedPayloadWindow(first.get()); InitPostedPayloadWindow(other.get());
 constexpr UINT msg=WM_APP+201;
 const auto begin=std::chrono::steady_clock::now();
 for(unsigned i=0;i<10000;++i) {
  auto value=std::make_unique<First>(); value->value=i;
  if(!PostMessagePayload(first.get(),msg,0,std::move(value))) return 3;
  MSG message{}; if(!PeekMessageW(&message,first.get(),msg,msg,PM_REMOVE)) return 4;
  if(TakeMessagePayload<First>(other.get(),msg,message.lParam)) return 5;
  if(TakeMessagePayload<First>(first.get(),msg+1,message.lParam)) return 6;
  if(TakeMessagePayload<Second>(first.get(),msg,message.lParam)) return 7;
  const auto accepted=TakeMessagePayload<First>(first.get(),msg,message.lParam);
  if(!accepted || accepted->value!=i || TakeMessagePayload<First>(first.get(),msg,message.lParam)) return 8;
 }
 const auto us=std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::steady_clock::now()-begin).count();
 if(!PostMessagePayload(first.get(),msg,0,std::make_unique<First>())) return 9;
 MSG stale{}; if(!PeekMessageW(&stale,first.get(),msg,msg,PM_REMOVE)) return 10;
 if(DrainPostedPayloadsForWindow(first.get())!=1 || TakeMessagePayload<First>(first.get(),msg,stale.lParam)) return 11;
 if(DrainPostedPayloadsForWindow(other.get())!=0) return 12;
 std::printf("PASS destination/message/trivial-type isolation, replay, drain; count=10000 duration_us=%lld\n",us);
 return 0;
}
