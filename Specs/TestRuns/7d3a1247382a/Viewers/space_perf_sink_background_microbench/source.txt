using Clock = std::chrono::steady_clock;

constexpr std::wstring_view kMetric = L"FileSystem.DirectoryOps.Enumerate";
constexpr size_t kCallsPerCase      = 1000;
constexpr size_t kConcurrentThreads = 8;
constexpr size_t kCallsPerThread    = kCallsPerCase / kConcurrentThreads;
constexpr size_t kDetailLength      = 300;

struct SinkReset final
{
    ~SinkReset()
    {
        Debug::Perf::ClearJsonlOutput();
    }
};

struct ThreadStats final
{
    uint64_t totalCallNs{};
    uint64_t maxCallNs{};
    uint64_t calls{};
    uint32_t backgroundBeginError{};
    uint32_t backgroundRestoreError{};
    uint32_t backgroundRestoreAttempts{};
    uint32_t backgroundRestoreFailures{};
    bool backgroundBeginFailed{};
    bool backgroundEntered{};
    bool backgroundRestored{};
};

struct CaseStats final
{
    uint64_t totalCallNs{};
    uint64_t maxCallNs{};
    uint64_t wallNs{};
    uint64_t calls{};
    uint32_t backgroundEntries{};
    uint32_t backgroundRestorations{};
    uint32_t backgroundRestoreAttempts{};
    uint32_t backgroundBeginFailures{};
    uint32_t backgroundRestoreFailures{};
    uint32_t backgroundLastError{};
};

// GetCurrentThread returns a pseudo handle owned by Windows for the calling thread; it must
// not be closed. THREAD_MODE_BACKGROUND_END is attempted on the same thread, including from
// this destructor if a future normal return is added before the explicit RestoreNow call.
class BackgroundPriorityGuard final
{
public:
    explicit BackgroundPriorityGuard(ThreadStats& stats) noexcept : _stats(stats)
    {
        if (SetThreadPriority(GetCurrentThread(), THREAD_MODE_BACKGROUND_BEGIN))
        {
            _entered                 = true;
            _stats.backgroundEntered = true;
        }
        else
        {
            _stats.backgroundBeginFailed = true;
            _stats.backgroundBeginError  = GetLastError();
        }
    }

    BackgroundPriorityGuard(const BackgroundPriorityGuard&)            = delete;
    BackgroundPriorityGuard& operator=(const BackgroundPriorityGuard&) = delete;

    ~BackgroundPriorityGuard() noexcept
    {
        Restore();
    }

    [[nodiscard]] bool Entered() const noexcept
    {
        return _entered;
    }

    void RestoreNow() noexcept
    {
        Restore();
    }

private:
    void Restore() noexcept
    {
        if (! _entered || _restored)
        {
            return;
        }

        ++_stats.backgroundRestoreAttempts;
        if (SetThreadPriority(GetCurrentThread(), THREAD_MODE_BACKGROUND_END))
        {
            _restored                 = true;
            _stats.backgroundRestored = true;
        }
        else
        {
            ++_stats.backgroundRestoreFailures;
            _stats.backgroundRestoreError = GetLastError();
        }
    }

    ThreadStats& _stats;
    bool _entered{};
    bool _restored{};
};

std::wstring MakeFixtureDetail()
{
    std::wstring detail{L"root=C:\\viewer-space-fixture\\tenant-2026-10\\dataset"};
    constexpr std::wstring_view segment{L"\\nested-directory-000000000000000000000000000000000000000000000000"};
    while (detail.size() < kDetailLength)
    {
        detail.append(segment);
    }
    detail.resize(kDetailLength);
    return detail;
}

void EmitMeasuredCalls(std::wstring_view detail, ThreadStats& stats, size_t calls)
{
    for (size_t index = 0; index < calls; ++index)
    {
        const auto started = Clock::now();
        // Synthetic one-microsecond operation duration: no directory read is performed.
        Debug::detail::WritePerfJsonl(kMetric, detail, 1, 4096, 1, S_OK);
        const auto elapsed   = std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now() - started).count();
        const auto elapsedNs = static_cast<uint64_t>(elapsed);
        stats.totalCallNs += elapsedNs;
        stats.maxCallNs = std::max(stats.maxCallNs, elapsedNs);
        ++stats.calls;
    }
}

void EmitMeasured(std::wstring_view detail, ThreadStats& stats, size_t calls, bool backgroundPriority)
{
    if (! backgroundPriority)
    {
        EmitMeasuredCalls(detail, stats, calls);
        return;
    }

    BackgroundPriorityGuard priority{stats};
    if (! priority.Entered())
    {
        return;
    }

    EmitMeasuredCalls(detail, stats, calls);
    priority.RestoreNow();
}

CaseStats RunCase(std::wstring_view detail, size_t threadCount, bool backgroundPriority)
{
    std::array<ThreadStats, kConcurrentThreads> perThread{};
    std::vector<std::jthread> workers;
    if (threadCount > 1)
    {
        workers.reserve(threadCount);
    }

    const auto wallStarted = Clock::now();
    if (threadCount == 1)
    {
        EmitMeasured(detail, perThread[0], kCallsPerCase, backgroundPriority);
    }
    else
    {
        for (size_t index = 0; index < threadCount; ++index)
        {
            workers.emplace_back([detail, &perThread, index, backgroundPriority]()
            { EmitMeasured(detail, perThread[index], kCallsPerThread, backgroundPriority); });
        }
        for (auto& worker : workers)
        {
            worker.join();
        }
    }

    CaseStats result{};
    result.wallNs = static_cast<uint64_t>(std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now() - wallStarted).count());
    for (size_t index = 0; index < threadCount; ++index)
    {
        result.totalCallNs += perThread[index].totalCallNs;
        result.maxCallNs = std::max(result.maxCallNs, perThread[index].maxCallNs);
        result.calls += perThread[index].calls;
        result.backgroundEntries += static_cast<uint32_t>(perThread[index].backgroundEntered);
        result.backgroundRestorations += static_cast<uint32_t>(perThread[index].backgroundRestored);
        result.backgroundRestoreAttempts += perThread[index].backgroundRestoreAttempts;
        result.backgroundBeginFailures += static_cast<uint32_t>(perThread[index].backgroundBeginFailed);
        result.backgroundRestoreFailures += perThread[index].backgroundRestoreFailures;
        result.backgroundLastError = perThread[index].backgroundRestoreError != 0 ? perThread[index].backgroundRestoreError
                                     : perThread[index].backgroundBeginError != 0 ? perThread[index].backgroundBeginError
                                                                                  : result.backgroundLastError;
    }
    return result;
}

void PrintCase(const CaseStats& stats, size_t threadCount, bool sinkEnabled, bool backgroundPriority)
{
    std::wcout << L"{\"sink_enabled\":" << (sinkEnabled ? L"true" : L"false") << L",\"priority_mode\":\"" << (backgroundPriority ? L"background" : L"normal")
               << L"\",\"threads\":" << threadCount << L",\"requested_calls\":" << kCallsPerCase << L",\"calls\":" << stats.calls << L",\"sink_call_total_ns\":"
               << stats.totalCallNs << L",\"sink_call_max_ns\":" << stats.maxCallNs << L",\"wall_ns\":" << stats.wallNs << L",\"priority_entries\":"
               << stats.backgroundEntries << L",\"priority_restore_attempts\":" << stats.backgroundRestoreAttempts << L",\"priority_restorations\":"
               << stats.backgroundRestorations << L",\"priority_begin_failures\":" << stats.backgroundBeginFailures << L",\"priority_restore_failures\":"
               << stats.backgroundRestoreFailures << L",\"priority_last_win32_error\":" << stats.backgroundLastError << L"}";
}

int wmain(int argc, wchar_t* argv[])
{
    if (argc != 2)
    {
        std::wcerr << L"Usage: space-perf-sink-microbench.exe <sandboxed-jsonl-output-path>\n";
        return 2;
    }

    // The parent driver owns the sandbox check and supplies a fresh, unique JSONL path below
    // its exact initialized-root path; the Common sink appends rows to the configured file.
    const std::filesystem::path outputPath{argv[1]};
    const std::wstring detail = MakeFixtureDetail();
    SinkReset resetSink;

    ConfigureSink(outputPath);
    const CaseStats firstSingle     = RunCase(detail, 1, false);
    const CaseStats firstConcurrent = RunCase(detail, kConcurrentThreads, false);

    // The sink appends. Clear and reconfigure the same parent-validated path so all four cases
    // remain in one log; after all four batches the expected row count is exactly 4,000.
    Debug::Perf::ClearJsonlOutput();
    ConfigureSink(outputPath);
    const CaseStats secondSingle     = RunCase(detail, 1, false);
    const CaseStats secondConcurrent = RunCase(detail, kConcurrentThreads, false);
    Debug::Perf::ClearJsonlOutput();
    const CaseStats sinkOffSingle     = RunCase(detail, 1, false);
    const CaseStats sinkOffConcurrent = RunCase(detail, kConcurrentThreads, false);

    ConfigureSink(outputPath);
    const CaseStats backgroundSingle     = RunCase(detail, 1, true);
    const CaseStats backgroundConcurrent = RunCase(detail, kConcurrentThreads, true);
    Debug::Perf::ClearJsonlOutput();

    const CaseStats backgroundSinkOffSingle     = RunCase(detail, 1, true);
    const CaseStats backgroundSinkOffConcurrent = RunCase(detail, kConcurrentThreads, true);
    Debug::Perf::ClearJsonlOutput();

    std::wcout << L"{\"scenario\":\"viewer_space_perf_sink_microbench\",\"metric\":\"FileSystem.DirectoryOps.Enumerate\"" << L",\"detail_characters\":"
               << detail.size() << L",\"synthetic_duration_us\":1,\"warmup_calls\":0,\"per_call_clock_reads\":2"
               << L",\"expected_invocations\":10000,\"expected_jsonl_rows\":6000,\"normal_priority_rounds\":[" << L"{\"round\":1,\"single_thread\":";
    PrintCase(firstSingle, 1, true, false);
    std::wcout << L",\"eight_threads\":";
    PrintCase(firstConcurrent, kConcurrentThreads, true, false);
    std::wcout << L"},{\"round\":2,\"single_thread\":";
    PrintCase(secondSingle, 1, true, false);
    std::wcout << L",\"eight_threads\":";
    PrintCase(secondConcurrent, kConcurrentThreads, true, false);
    std::wcout << L"}],\"normal_priority_sink_off\":{\"single_thread\":";
    PrintCase(sinkOffSingle, 1, false, false);
    std::wcout << L",\"eight_threads\":";
    PrintCase(sinkOffConcurrent, kConcurrentThreads, false, false);
    std::wcout << L"},\"background_priority_sink_enabled\":{\"single_thread\":";
    PrintCase(backgroundSingle, 1, true, true);
    std::wcout << L",\"eight_threads\":";
    PrintCase(backgroundConcurrent, kConcurrentThreads, true, true);
    std::wcout << L"},\"background_priority_sink_off\":{\"single_thread\":";
    PrintCase(backgroundSinkOffSingle, 1, false, true);
    std::wcout << L",\"eight_threads\":";
    PrintCase(backgroundSinkOffConcurrent, kConcurrentThreads, false, true);
    std::wcout << L"},\"scope\":\"synthetic logger control; no filesystem reads; no performance-gain claim\"}\n";

    const auto hasPriorityFailure = [](const CaseStats& stats)
    { return stats.backgroundBeginFailures != 0 || stats.backgroundRestoreFailures != 0 || stats.backgroundEntries != stats.backgroundRestorations; };
    return hasPriorityFailure(backgroundSingle) || hasPriorityFailure(backgroundConcurrent) || hasPriorityFailure(backgroundSinkOffSingle) ||
                   hasPriorityFailure(backgroundSinkOffConcurrent)
               ? 3
               : 0;
}
